// WrappedError wraps an underlying cause with a contextual message while
// preserving errors.Is / errors.As matching through Unwrap.
type WrappedError struct {
        Msg string
        Err error
}

// Error implements the error interface.
func (e *WrappedError) Error() string {
        return e.Msg + ": " + e.Err.Error()
}

// Unwrap returns the wrapped cause.
func (e *WrappedError) Unwrap() error { return e.Err }

// ntpClient is the subset of the SDK time-synchronization client the CLI
// commands use. It is satisfied by *ntpsvc.Client and by test fakes.
type ntpClient interface {
        // Status returns the node's synchronization state.
        Status(ctx context.Context) (*platformv1.NTPStatus, error)

        // Sources returns the configured time sources.
        Sources(ctx context.Context) ([]*platformv1.NTPSource, error)

        // Step forces an immediate clock correction.
        Step(ctx context.Context) (*platformv1.NTPStatus, error)
}

// defaultCLITimeout is the context deadline for all NTP CLI operations.
const defaultCLITimeout = 30 * time.Second

// newCLIContext returns a context with the default CLI timeout and its cancel
// function. Callers must defer cancel().
func newCLIContext() (context.Context, context.CancelFunc) {
        return context.WithTimeout(context.Background(), defaultCLITimeout)
}

// clientFactory builds a time-synchronization client from the common CLI
// flags. Tests replace it with a fake to exercise command wiring without a
// daemon.
type clientFactory func(flags *cli.CommonFlags) (ntpClient, func() error, error)

// newNTPClient is the production client factory. It honors the global
// --protocol/--endpoint flags via cli.NewBaseCommand.
var newNTPClient clientFactory = func(flags *cli.CommonFlags) (ntpClient, func() error, error) {
        base, err := cli.NewBaseCommand(flags)
        if err != nil {
                return nil, nil, err
        }
        return ntpsvc.NewClientFromGRPC(base.Client().NTP()), base.Close, nil
}

// outputFunc renders a command result. Tests replace it to capture output.
type outputFunc func(flags *cli.CommonFlags, v any) error

// output is the production output renderer.
var output outputFunc = func(flags *cli.CommonFlags, v any) error {
        base, err := cli.NewBaseCommand(flags)
        if err != nil {
                return err
        }
        defer base.Close()
        return base.Output(v)
}

// runStatus executes the status subcommand.
func runStatus(flags *cli.CommonFlags) error {
        client, closeFn, err := newNTPClient(flags)
        if err != nil {
                return err
        }
        defer closeFn()

        ctx, cancel := newCLIContext()
        defer cancel()

        status, err := client.Status(ctx)
        if err != nil {
                return &WrappedError{Msg: "failed to get NTP status", Err: err}
        }
        if isStructuredOutput(flags) {
                return output(flags, status)
        }
        return output(flags, statusTable(status))
}

// isStructuredOutput reports whether the caller asked for JSON or YAML, in
// which case the raw proto message is rendered instead of a table.
func isStructuredOutput(flags *cli.CommonFlags) bool {
        if flags == nil {
                return false
        }
        return flags.OutputFormat == cli.OutputFormatJSON || flags.OutputFormat == cli.OutputFormatYAML
}

// statusTable renders a status message as a two-column table.
func statusTable(status *platformv1.NTPStatus) *cli.TableData {
        table := cli.NewTableData("FIELD", "VALUE")
        if status == nil {
                return table
        }

        table.AddRow("synced", strconv.FormatBool(status.GetSynced()))
        table.AddRow("stratum", strconv.FormatUint(uint64(status.GetStratum()), 10))
        table.AddRow("leap status", status.GetLeapStatus())
        table.AddRow("offset (s)", strconv.FormatFloat(status.GetOffsetSeconds(), 'f', 9, 64))
        table.AddRow("root dispersion (s)", strconv.FormatFloat(status.GetRootDispersion(), 'f', 9, 64))
        table.AddRow("root delay (s)", strconv.FormatFloat(status.GetRootDelay(), 'f', 9, 64))
        table.AddRow("last sync", formatUnix(status.GetLastSyncUnix()))
        table.AddRow("mode", status.GetMode())
        table.AddRow("reference", status.GetReferenceName())
        table.AddRow("sources", strconv.FormatUint(uint64(status.GetSourcesReachable()), 10)+
                "/"+strconv.FormatUint(uint64(status.GetSourcesTotal()), 10)+" reachable")
        table.AddRow("anomaly", strconv.FormatBool(status.GetAnomaly()))
        if status.GetAnomalyReason() != "" {
                table.AddRow("anomaly reason", status.GetAnomalyReason())
        }
        return table
}

// sourcesTable renders the source list as a table.
func sourcesTable(sources []*platformv1.NTPSource) *cli.TableData {
        table := cli.NewTableData("ADDRESS", "MODE", "STATE", "STRATUM", "POLL", "REACH", "LAST RX", "OFFSET")
        for _, src := range sources {
                table.AddRow(
                        src.GetAddress(),
                        src.GetMode(),
                        src.GetState(),
                        strconv.FormatUint(uint64(src.GetStratum()), 10),
                        strconv.FormatUint(uint64(src.GetPollSeconds()), 10)+"s",
                        strconv.FormatUint(uint64(src.GetReach()), 10),
                        strconv.FormatInt(src.GetLastRxSeconds(), 10)+"s",
                        strconv.FormatFloat(src.GetOffsetSeconds(), 'f', 9, 64),
                )
        }
        return table
}

// formatUnix renders a UNIX timestamp, reporting "never" for the zero value
// so an unsynchronized node is unambiguous.
func formatUnix(seconds int64) string {
        if seconds <= 0 {
                return "never"
        }
        return time.Unix(seconds, 0).UTC().Format(time.RFC3339)
}

// ManagedHeader marks a chrony.conf as platform-owned. Operators editing the
// file by hand will have their changes overwritten on the next boot.
const ManagedHeader = "# Managed by go-trusted-platform (pkg/ntp). Do not edit by hand."

// firstBootStepUpdates is the number of initial updates during which chronyd
// may STEP the clock instead of slewing it. A node whose RTC is years wrong
// cannot be slewed back in any useful time, so the first corrections must be
// steps; afterwards a large jump is an anomaly, not a correction.
const firstBootStepUpdates = 3

// firstBootStepThresholdSeconds is the offset above which those initial
// updates step rather than slew.
const firstBootStepThresholdSeconds = "1.0"

// Paths are the filesystem locations chronyd is pointed at. They are not part
// of the YAML schema: they are platform layout constants, overridable only in
// tests via Service options.
type Paths struct {
        // ConfigFile is the rendered chrony.conf destination.
        ConfigFile string

        // DriftFile persists the measured oscillator drift across reboots.
        DriftFile string

        // NTSCertFile is where the resolved NTS server certificate is written.
        NTSCertFile string

        // NTSKeyFile is where the resolved NTS server private key is written.
        NTSKeyFile string

        // NTSDumpDir holds chronyd's NTS-KE server key cache.
        NTSDumpDir string

        // CommandSocket is chronyd's Unix control socket, which is how the
        // platform reads sync state.
        CommandSocket string
}

// DefaultPaths returns the Alpine/OpenRC filesystem layout chrony ships with.
func DefaultPaths() Paths {
        return Paths{
                ConfigFile:    "/etc/chrony/chrony.conf",
                DriftFile:     "/var/lib/chrony/chrony.drift",
                NTSCertFile:   "/etc/chrony/nts-server.crt",
                NTSKeyFile:    "/etc/chrony/nts-server.key",
                NTSDumpDir:    "/var/lib/chrony",
                CommandSocket: "/run/chrony/chronyd.sock",
        }
}

// Render produces the chrony.conf body for cfg. It never emits an
// unrestricted `allow` directive: the allow-list comes straight from the
// validated serve.listen-networks entries.
func Render(cfg *Config, paths Paths) (string, error) {
        if cfg == nil {
                return "", ErrRenderFailed{Cause: ErrNilConfig{}}
        }
        if err := cfg.Validate(); err != nil {
                return "", ErrRenderFailed{Cause: err}
        }
        if !cfg.Enabled {
                return "", ErrRenderFailed{Cause: ErrDisabled{}}
        }

        var b strings.Builder
        b.WriteString(ManagedHeader)
        b.WriteString("\n# mode: ")
        b.WriteString(string(cfg.Mode))
        b.WriteString("\n\n")

        renderUpstream(&b, cfg)
        renderDiscipline(&b, cfg, paths)
        renderServer(&b, cfg, paths)
        renderControl(&b, paths)

        return b.String(), nil
}

// renderUpstream writes the server/pool directives.
func renderUpstream(b *strings.Builder, cfg *Config) {
        if !cfg.Mode.Consumes() || cfg.Consume == nil || len(cfg.Consume.Upstream) == 0 {
                return
        }

        b.WriteString("# Upstream time sources, in configuration order.\n")
        for _, src := range cfg.Consume.Upstream {
                directive := "server"
                if src.Pool {
                        directive = "pool"
                }
                b.WriteString(directive)
                b.WriteByte(' ')
                b.WriteString(strings.TrimSpace(src.Address))
                if src.IBurst {
                        b.WriteString(" iburst")
                }
                if src.Prefer {
                        b.WriteString(" prefer")
                }
                if src.NTS {
                        b.WriteString(" nts")
                }
                b.WriteByte('\n')
        }
        b.WriteByte('\n')
}

// renderDiscipline writes the clock-discipline directives shared by every
// role.
func renderDiscipline(b *strings.Builder, cfg *Config, paths Paths) {
        b.WriteString("# Clock discipline. The first ")
        b.WriteString(strconv.Itoa(firstBootStepUpdates))
        b.WriteString(" updates may STEP the clock so a node whose RTC is years\n")
        b.WriteString("# wrong converges immediately; later corrections slew and a large one is an anomaly.\n")
        b.WriteString("driftfile ")
        b.WriteString(paths.DriftFile)
        b.WriteString("\nmakestep ")
        b.WriteString(firstBootStepThresholdSeconds)
        b.WriteByte(' ')
        b.WriteString(strconv.Itoa(firstBootStepUpdates))
        b.WriteByte('\n')

        if cfg.RTCWriteBack() {
                b.WriteString("rtcsync\n")
        }
        b.WriteByte('\n')
}

// renderServer writes the server-side directives: the explicit allow-list,
// orphan-mode stratum, and NTS-KE material. Consume-only nodes get `port 0`
// so they never listen on 123 at all.
func renderServer(b *strings.Builder, cfg *Config, paths Paths) {
        if !cfg.Mode.Serves() {
                b.WriteString("# Consume-only node: do not listen on the NTP port.\nport 0\n\n")
                return
        }

        b.WriteString("# Explicit CIDR allow-list. Never 0.0.0.0/0 — an open NTP server is a\n")
        b.WriteString("# reflection-amplification source.\n")
        for _, network := range cfg.Serve.ListenNetworks {
                b.WriteString("allow ")
                b.WriteString(strings.TrimSpace(network))
                b.WriteByte('\n')
        }
        b.WriteByte('\n')

        if cfg.Consume != nil && cfg.Consume.OrphanFallback {
                b.WriteString("# Orphan fallback: keep serving from the local clock when no upstream is\n")
                b.WriteString("# reachable, at a stratum low enough that any real chain wins.\n")
                b.WriteString("local stratum ")
                b.WriteString(strconv.Itoa(cfg.LocalStratum()))
                b.WriteString("\n\n")
        }

        if cfg.NTSServing() {
                b.WriteString("# RFC 8915 Network Time Security, keyed by the platform CA.\n")
                b.WriteString("ntsservercert ")
                b.WriteString(paths.NTSCertFile)
                b.WriteString("\nntsserverkey ")
                b.WriteString(paths.NTSKeyFile)
                b.WriteString("\nntsdumpdir ")
                b.WriteString(paths.NTSDumpDir)
                b.WriteString("\n\n")
        }
}

// renderControl writes the control-socket directives. The UDP command port is
// disabled outright: the platform reads state over the local Unix socket.
func renderControl(b *strings.Builder, paths Paths) {
        b.WriteString("# Control surface: local Unix socket only, no UDP command port.\n")
        b.WriteString("bindcmdaddress ")
        b.WriteString(paths.CommandSocket)
        b.WriteString("\ncmdport 0\n")
}

// Package ntp owns the platform's time-synchronization surface. The platform
// does not implement the NTP protocol: chrony runs as a packaged OpenRC
// service and this package renders its configuration, manages its lifecycle,
// polls its control socket for sync state, and exposes that state to the
// multi-protocol API.
//
// Time is a trust primitive here. TPM attestation freshness, ACME (RFC 8555)
// issuance, DANE/DNSSEC signature windows, and JWT exp/nbf all silently
// misbehave on a node whose clock has drifted, so a node that cannot prove it
// is synchronized is flagged in health status rather than left unnoticed.
//
// Following the platform's explicit-configuration convention, an absent
// `ntp:` section means platformd never touches chrony at all.
package ntp

import (
        "net/netip"
        "strings"
        "time"
)

// Mode selects whether a node serves NTP to peers, consumes it from
// upstream, or both.
type Mode string

const (
        // ModeServe serves time to the configured listen networks only.
        ModeServe Mode = "serve"

        // ModeConsume synchronizes from upstream sources only.
        ModeConsume Mode = "consume"

        // ModeBoth consumes from upstream and serves the result to peers. This
        // is the control-plane role.
        ModeBoth Mode = "both"
)

// Serves reports whether the mode makes the node an NTP server.
func (m Mode) Serves() bool { return m == ModeServe || m == ModeBoth }

// Consumes reports whether the mode makes the node an NTP client.
func (m Mode) Consumes() bool { return m == ModeConsume || m == ModeBoth }

// TimeoutAction selects what happens when the startup sync gate expires.
type TimeoutAction string

const (
        // ActionWarnAndContinue boots the node but flags it time-unsynced in
        // health status so fleet monitoring surfaces it. This is the default.
        ActionWarnAndContinue TimeoutAction = "warn-and-continue"

        // ActionFailStartup aborts platformd startup with ErrSyncTimeout.
        ActionFailStartup TimeoutAction = "fail-startup"
)

// DefaultGateTimeout is the startup gate deadline applied when
// startup-gate.timeout is omitted.
const DefaultGateTimeout = 120 * time.Second

// DefaultLocalStratum is the orphan-mode stratum applied when
// serve.local-stratum is omitted. Stratum 10 keeps a self-referencing
// control plane clearly below any real upstream chain.
const DefaultLocalStratum = 10

// Config is the `ntp:` configuration section. A nil *Config means the
// section is absent and platformd leaves chrony completely alone.
type Config struct {
        // Enabled gates the whole subsystem. A present-but-disabled section is
        // treated exactly like an absent one.
        Enabled bool `yaml:"enabled" json:"enabled" mapstructure:"enabled"`

        // Mode selects serve, consume, or both.
        Mode Mode `yaml:"mode" json:"mode" mapstructure:"mode"`

        // Consume configures upstream synchronization and the startup gate.
        Consume *ConsumeConfig `yaml:"consume,omitempty" json:"consume,omitempty" mapstructure:"consume"`

        // Serve configures the server side (allow-list, orphan stratum, NTS).
        Serve *ServeConfig `yaml:"serve,omitempty" json:"serve,omitempty" mapstructure:"serve"`

        // RTC configures hardware-clock write-back.
        RTC *RTCConfig `yaml:"rtc,omitempty" json:"rtc,omitempty" mapstructure:"rtc"`
}

// ConsumeConfig describes how the node obtains time from upstream.
type ConsumeConfig struct {
        // Upstream lists the servers and pools chronyd polls, in order.
        Upstream []UpstreamSource `yaml:"upstream,omitempty" json:"upstream,omitempty" mapstructure:"upstream"`

        // StartupGate blocks platform bring-up until the clock is trustworthy.
        StartupGate *StartupGate `yaml:"startup-gate,omitempty" json:"startup_gate,omitempty" mapstructure:"startup-gate"`

        // OrphanFallback keeps the local clock usable as a last-resort
        // reference when no upstream is reachable. Only meaningful on a node
        // that also serves time.
        OrphanFallback bool `yaml:"orphan-fallback" json:"orphan_fallback" mapstructure:"orphan-fallback"`
}

// UpstreamSource is one `server` or `pool` line in chrony.conf.
type UpstreamSource struct {
        // Address is the host name or IP of the upstream.
        Address string `yaml:"address" json:"address" mapstructure:"address"`

        // Pool renders the source as a chrony `pool` directive (a name that
        // resolves to several addresses) instead of `server`.
        Pool bool `yaml:"pool,omitempty" json:"pool,omitempty" mapstructure:"pool"`

        // Prefer marks this source as preferred over the others.
        Prefer bool `yaml:"prefer,omitempty" json:"prefer,omitempty" mapstructure:"prefer"`

        // IBurst sends a fast initial burst so the first sync is quick. This
        // is what turns a two-year clock error into a sub-minute correction.
        IBurst bool `yaml:"iburst,omitempty" json:"iburst,omitempty" mapstructure:"iburst"`

        // NTS requires RFC 8915 Network Time Security towards this source.
        // Internal control-plane sources set this; most public upstreams and
        // nearly every router cannot do NTS, so it is best-effort there.
        NTS bool `yaml:"nts,omitempty" json:"nts,omitempty" mapstructure:"nts"`
}

// StartupGate blocks platform bring-up until chronyd reports a synchronized
// clock, so TPM/CA initialization never runs against a bogus time.
type StartupGate struct {
        // BlockUntilSynced enables the gate.
        BlockUntilSynced bool `yaml:"block-until-synced" json:"block_until_synced" mapstructure:"block-until-synced"`

        // Timeout bounds how long the gate waits.
        Timeout time.Duration `yaml:"timeout,omitempty" json:"timeout,omitempty" mapstructure:"timeout"`

        // OnTimeout selects warn-and-continue (default) or fail-startup.
        OnTimeout TimeoutAction `yaml:"on-timeout,omitempty" json:"on_timeout,omitempty" mapstructure:"on-timeout"`
}

// ServeConfig describes the server side of the node's time service.
type ServeConfig struct {
        // ListenNetworks is the explicit CIDR allow-list rendered as chrony
        // `allow` directives. It is never 0.0.0.0/0 or ::/0 — an open NTP
        // server is a reflection-amplification source.
        ListenNetworks []string `yaml:"listen-networks,omitempty" json:"listen_networks,omitempty" mapstructure:"listen-networks"`

        // LocalStratum is the stratum announced when the server is running on
        // its own clock (orphan mode).
        LocalStratum int `yaml:"local-stratum,omitempty" json:"local_stratum,omitempty" mapstructure:"local-stratum"`

        // NTS configures RFC 8915 Network Time Security for served time.
        NTS *NTSConfig `yaml:"nts,omitempty" json:"nts,omitempty" mapstructure:"nts"`
}

// NTSConfig carries the platform-CA-issued key material chronyd uses as an
// NTS-KE server. The URIs are resolved by the platform secrets engine before
// the service starts, so by the time the renderer runs these fields hold PEM
// material, not URIs.
type NTSConfig struct {
        // Enabled turns on the NTS-KE server.
        Enabled bool `yaml:"enabled" json:"enabled" mapstructure:"enabled"`

        // CertURI references the NTS server certificate (kms:// URI).
        CertURI string `yaml:"cert-uri,omitempty" json:"cert_uri,omitempty" mapstructure:"cert-uri"`

        // KeyURI references the NTS server private key (kms:// URI).
        KeyURI string `yaml:"key-uri,omitempty" json:"key_uri,omitempty" mapstructure:"key-uri"`
}

// RTCConfig controls hardware-clock write-back.
type RTCConfig struct {
        // WriteBack runs `hwclock --systohc` once the clock is confirmed
        // synchronized, so the next cold boot starts from a sane time even
        // before the network is up.
        WriteBack bool `yaml:"write-back" json:"write_back" mapstructure:"write-back"`
}

// Active reports whether platformd should manage chrony at all.
func (c *Config) Active() bool { return c != nil && c.Enabled }

// GateTimeout returns the effective startup gate deadline.
func (c *Config) GateTimeout() time.Duration {
        if c == nil || c.Consume == nil || c.Consume.StartupGate == nil ||
                c.Consume.StartupGate.Timeout <= 0 {
                return DefaultGateTimeout
        }
        return c.Consume.StartupGate.Timeout
}

// GateEnabled reports whether the startup gate should block bring-up.
func (c *Config) GateEnabled() bool {
        return c.Active() && c.Mode.Consumes() &&
                c.Consume != nil && c.Consume.StartupGate != nil &&
                c.Consume.StartupGate.BlockUntilSynced
}

// TimeoutAction returns the effective gate timeout policy.
func (c *Config) TimeoutAction() TimeoutAction {
        if c == nil || c.Consume == nil || c.Consume.StartupGate == nil ||
                c.Consume.StartupGate.OnTimeout == "" {
                return ActionWarnAndContinue
        }
        return c.Consume.StartupGate.OnTimeout
}

// RTCWriteBack reports whether the hardware clock is updated after sync.
func (c *Config) RTCWriteBack() bool {
        return c != nil && c.RTC != nil && c.RTC.WriteBack
}

// LocalStratum returns the effective orphan-mode stratum.
func (c *Config) LocalStratum() int {
        if c == nil || c.Serve == nil || c.Serve.LocalStratum == 0 {
                return DefaultLocalStratum
        }
        return c.Serve.LocalStratum
}

// NTSServing reports whether the node runs an NTS-KE server.
func (c *Config) NTSServing() bool {
        return c.Active() && c.Mode.Serves() && c.Serve != nil &&
                c.Serve.NTS != nil && c.Serve.NTS.Enabled
}

// validTimeoutActions is the map-based membership test for on-timeout.
var validTimeoutActions = map[TimeoutAction]struct{}{
        ActionWarnAndContinue: {},
        ActionFailStartup:     {},
}

// validModes is the map-based membership test for mode.
var validModes = map[Mode]struct{}{
        ModeServe:   {},
        ModeConsume: {},
        ModeBoth:    {},
}

// unrestrictedNetworks are the prefixes that would turn the node into an open
// NTP reflector. They are rejected outright rather than warned about.
var unrestrictedNetworks = map[string]struct{}{
        "0.0.0.0/0": {},
        "::/0":      {},
}

// Validate checks the configuration for internal consistency. A disabled or
// absent section always validates: it does nothing.
func (c *Config) Validate() error {
        if c == nil {
                return ErrNilConfig{}
        }
        if !c.Enabled {
                return nil
        }

        if _, ok := validModes[c.Mode]; !ok {
                return ErrInvalidMode{Mode: string(c.Mode)}
        }

        if c.Mode.Consumes() {
                if err := c.validateConsume(); err != nil {
                        return err
                }
        }

        if c.Mode.Serves() {
                if err := c.validateServe(); err != nil {
                        return err
                }
        }

        return nil
}

// validateConsume checks the client side of the configuration.
func (c *Config) validateConsume() error {
        if c.Consume == nil {
                return ErrMissingConsumeSection{Mode: string(c.Mode)}
        }

        if len(c.Consume.Upstream) == 0 && !c.Consume.OrphanFallback {
                return ErrNoUpstream{}
        }

        for i, src := range c.Consume.Upstream {
                address := strings.TrimSpace(src.Address)
                if address == "" {
                        return ErrEmptyUpstreamAddress{Index: i}
                }
                // chrony.conf is whitespace-delimited and line-oriented: an
                // address carrying whitespace or a comment marker would inject
                // additional directives.
                if strings.ContainsAny(address, " \t\r\n#") {
                        return ErrInvalidUpstreamAddress{Address: src.Address}
                }
        }

        if gate := c.Consume.StartupGate; gate != nil {
                if gate.OnTimeout != "" {
                        if _, ok := validTimeoutActions[gate.OnTimeout]; !ok {
                                return ErrInvalidTimeoutAction{Action: string(gate.OnTimeout)}
                        }
                }
                if gate.BlockUntilSynced && gate.Timeout < 0 {
                        return ErrInvalidGateTimeout{Seconds: gate.Timeout.Seconds()}
                }
        }

        return nil
}

// validateServe checks the server side of the configuration.
func (c *Config) validateServe() error {
        if c.Serve == nil {
                return ErrMissingServeSection{Mode: string(c.Mode)}
        }

        if len(c.Serve.ListenNetworks) == 0 {
                return ErrNoListenNetworks{}
        }

        for _, network := range c.Serve.ListenNetworks {
                trimmed := strings.TrimSpace(network)
                if _, unrestricted := unrestrictedNetworks[trimmed]; unrestricted {
                        return ErrUnrestrictedListenNetwork{Network: network}
                }
                prefix, err := netip.ParsePrefix(trimmed)
                if err != nil {
                        return ErrInvalidListenNetwork{Network: network, Cause: err}
                }
                if prefix.Bits() == 0 {
                        return ErrUnrestrictedListenNetwork{Network: network}
                }
        }

        if c.Serve.LocalStratum != 0 && (c.Serve.LocalStratum < 1 || c.Serve.LocalStratum > 15) {
                return ErrInvalidLocalStratum{Stratum: c.Serve.LocalStratum}
        }

        if nts := c.Serve.NTS; nts != nil && nts.Enabled {
                if strings.TrimSpace(nts.CertURI) == "" {
                        return ErrMissingNTSMaterial{Field: "serve.nts.cert-uri"}
                }
                if strings.TrimSpace(nts.KeyURI) == "" {
                        return ErrMissingNTSMaterial{Field: "serve.nts.key-uri"}
                }
        }

        return nil
}

// CommandRunner executes a system command and returns its combined output.
// Every host interaction in this package goes through it, which is what keeps
// unit tests from touching the real clock, RTC, or chronyd.
type CommandRunner interface {
        // Run executes name with args and returns the combined output.
        Run(ctx context.Context, name string, args ...string) ([]byte, error)
}

// execRunner is the production CommandRunner backed by os/exec.
type execRunner struct{}

// Run executes the command and returns its combined output.
func (execRunner) Run(ctx context.Context, name string, args ...string) ([]byte, error) {
        return exec.CommandContext(ctx, name, args...).CombinedOutput()
}

// Tracking is chronyd's raw tracking report, one line of `chronyc -c
// tracking`.
type Tracking struct {
        // ReferenceID is the hex reference identifier.
        ReferenceID string

        // ReferenceName is the reference's name or address.
        ReferenceName string

        // Stratum is the node's distance from a reference clock.
        Stratum uint32

        // RefTime is the timestamp of the last measurement from the
        // reference.
        RefTime time.Time

        // SystemTimeOffset is the current offset in seconds; positive means
        // the system clock is ahead of the reference.
        SystemTimeOffset float64

        // LastOffset is the offset at the last clock update, in seconds.
        LastOffset float64

        // RootDelay is the accumulated round-trip delay, in seconds.
        RootDelay float64

        // RootDispersion is the accumulated dispersion, in seconds.
        RootDispersion float64

        // LeapStatus is chronyd's leap indicator.
        LeapStatus string
}

// Synced reports whether chronyd currently disciplines the clock.
func (t *Tracking) Synced() bool {
        if t == nil {
                return false
        }
        return t.LeapStatus != LeapNotSynchronised &&
                t.Stratum > UnsyncedStratum && t.Stratum < MaxStratum
}

// Controller reads state from and issues commands to a running chronyd over
// its control socket.
type Controller interface {
        // Tracking returns chronyd's current tracking report.
        Tracking(ctx context.Context) (*Tracking, error)

        // Sources returns the configured sources and their measurement state.
        Sources(ctx context.Context) ([]Source, error)

        // MakeStep asks chronyd to correct the clock with an immediate step.
        MakeStep(ctx context.Context) error
}

// chronycBinary is the control client shipped with the chrony package.
const chronycBinary = "chronyc"

// trackingFieldCount is the number of CSV columns `chronyc -c tracking`
// emits.
const trackingFieldCount = 14

// sourcesFieldCount is the number of CSV columns `chronyc -c sources` emits.
const sourcesFieldCount = 10

// chronyc is the production Controller. It shells out to chronyc, which
// speaks chrony's native command protocol over the local Unix control socket
// — the platform never reimplements that protocol.
type chronyc struct {
        runner CommandRunner
        socket string
}

// NewController returns a Controller that reaches chronyd through the given
// control socket. A nil runner uses os/exec.
func NewController(socket string, runner CommandRunner) Controller {
        if runner == nil {
                runner = execRunner{}
        }
        return &chronyc{runner: runner, socket: socket}
}

// args builds the chronyc argument vector for a command. `-c` selects CSV
// output and `-n` suppresses reverse DNS so a broken resolver never stalls a
// status poll.
func (c *chronyc) args(command string) []string {
        args := make([]string, 0, 6)
        args = append(args, "-c", "-n")
        if c.socket != "" {
                args = append(args, "-h", c.socket)
        }
        return append(args, command)
}

// run executes a chronyc command and returns its output.
func (c *chronyc) run(ctx context.Context, command string) (string, error) {
        out, err := c.runner.Run(ctx, chronycBinary, c.args(command)...)
        if err != nil {
                return "", ErrControlUnavailable{Command: command, Output: string(out), Cause: err}
        }
        return string(out), nil
}

// Tracking returns chronyd's current tracking report.
func (c *chronyc) Tracking(ctx context.Context) (*Tracking, error) {
        out, err := c.run(ctx, "tracking")
        if err != nil {
                return nil, err
        }
        return parseTracking(out)
}

// Sources returns the configured sources and their measurement state.
func (c *chronyc) Sources(ctx context.Context) ([]Source, error) {
        out, err := c.run(ctx, "sources")
        if err != nil {
                return nil, err
        }
        return parseSources(out)
}

// MakeStep asks chronyd to correct the clock with an immediate step.
func (c *chronyc) MakeStep(ctx context.Context) error {
        _, err := c.run(ctx, "makestep")
        return err
}

// newCSVReader returns a CSV reader configured for chronyc output: variable
// field counts are validated per record so a malformed line produces a typed
// error instead of a silent short read.
func newCSVReader(payload string) *csv.Reader {
        reader := csv.NewReader(strings.NewReader(payload))
        reader.FieldsPerRecord = -1
        reader.TrimLeadingSpace = true
        return reader
}

// parseTracking converts `chronyc -c tracking` CSV into a Tracking report.
func parseTracking(payload string) (*Tracking, error) {
        records, err := newCSVReader(payload).ReadAll()
        if err != nil {
                return nil, ErrMalformedControlOutput{
                        Command: "tracking",
                        Line:    strings.TrimSpace(payload),
                        Want:    trackingFieldCount,
                        Got:     0,
                }
        }
        if len(records) == 0 {
                return nil, ErrMalformedControlOutput{
                        Command: "tracking",
                        Line:    strings.TrimSpace(payload),
                        Want:    trackingFieldCount,
                        Got:     0,
                }
        }

        fields := records[0]
        if len(fields) < trackingFieldCount {
                return nil, ErrMalformedControlOutput{
                        Command: "tracking",
                        Line:    strings.Join(fields, ","),
                        Want:    trackingFieldCount,
                        Got:     len(fields),
                }
        }

        return &Tracking{
                ReferenceID:      fields[0],
                ReferenceName:    fields[1],
                Stratum:          parseUint32(fields[2]),
                RefTime:          parseUnixFloat(fields[3]),
                SystemTimeOffset: parseFloat(fields[4]),
                LastOffset:       parseFloat(fields[5]),
                RootDelay:        parseFloat(fields[10]),
                RootDispersion:   parseFloat(fields[11]),
                LeapStatus:       strings.TrimSpace(fields[13]),
        }, nil
}

// parseSources converts `chronyc -c sources` CSV into Source records.
func parseSources(payload string) ([]Source, error) {
        records, err := newCSVReader(payload).ReadAll()
        if err != nil {
                return nil, ErrMalformedControlOutput{
                        Command: "sources",
                        Line:    strings.TrimSpace(payload),
                        Want:    sourcesFieldCount,
                        Got:     0,
                }
        }

        sources := make([]Source, 0, len(records))
        for _, fields := range records {
                if len(fields) == 0 || (len(fields) == 1 && strings.TrimSpace(fields[0]) == "") {
                        continue
                }
                if len(fields) < sourcesFieldCount {
                        return nil, ErrMalformedControlOutput{
                                Command: "sources",
                                Line:    strings.Join(fields, ","),
                                Want:    sourcesFieldCount,
                                Got:     len(fields),
                        }
                }

                reach := parseUint32(fields[5])
                sources = append(sources, Source{
                        Mode:          parseSourceMode(strings.TrimSpace(fields[0])),
                        State:         parseSourceState(strings.TrimSpace(fields[1])),
                        Address:       strings.TrimSpace(fields[2]),
                        Stratum:       parseUint32(fields[3]),
                        PollSeconds:   pollIntervalSeconds(fields[4]),
                        Reach:         reach,
                        LastRxSeconds: parseInt64(fields[6]),
                        OffsetSeconds: parseFloat(fields[7]),
                        Reachable:     reach != 0,
                })
        }

        return sources, nil
}

// pollIntervalSeconds converts chrony's log2 poll interval to seconds.
func pollIntervalSeconds(raw string) uint32 {
        exponent, err := strconv.Atoi(strings.TrimSpace(raw))
        if err != nil || exponent < 0 || exponent > 31 {
                return 0
        }
        return uint32(1) << uint(exponent)
}

// parseUint32 parses a decimal field, yielding 0 on malformed input. chronyd
// emits well-formed numerics; a bad field means the column is absent for this
// source, which is semantically zero.
func parseUint32(raw string) uint32 {
        value, err := strconv.ParseUint(strings.TrimSpace(raw), 10, 32)
        if err != nil {
                return 0
        }
        return uint32(value)
}

// parseInt64 parses a signed decimal field, yielding 0 on malformed input.
func parseInt64(raw string) int64 {
        value, err := strconv.ParseInt(strings.TrimSpace(raw), 10, 64)
        if err != nil {
                return 0
        }
        return value
}

// parseFloat parses a float field, yielding 0 on malformed input.
func parseFloat(raw string) float64 {
        value, err := strconv.ParseFloat(strings.TrimSpace(raw), 64)
        if err != nil {
                return 0
        }
        return value
}

// parseUnixFloat parses chrony's fractional UNIX timestamp. A zero or
// malformed value yields the zero time, which callers render as "never".
func parseUnixFloat(raw string) time.Time {
        seconds, err := strconv.ParseFloat(strings.TrimSpace(raw), 64)
        if err != nil || seconds <= 0 {
                return time.Time{}
        }
        whole, fraction := int64(seconds), seconds-float64(int64(seconds))
        return time.Unix(whole, int64(fraction*float64(time.Second))).UTC()
}

const hwclockBinary = "hwclock"

// defaultRunlevel is the OpenRC runlevel chronyd is enabled in.
const defaultRunlevel = "default"

// FileSystem is the subset of filesystem operations the service performs.
// Injecting it keeps unit tests entirely in memory.
type FileSystem interface {
        // ReadFile returns the contents of path.
        ReadFile(path string) ([]byte, error)

        // WriteFile writes data to path with the given permissions.
        WriteFile(path string, data []byte, perm os.FileMode) error

        // MkdirAll creates path and any missing parents.
        MkdirAll(path string, perm os.FileMode) error
}

// osFileSystem is the production FileSystem backed by the os package.
type osFileSystem struct{}

// ReadFile returns the contents of path.
func (osFileSystem) ReadFile(path string) ([]byte, error) { return os.ReadFile(path) }

// WriteFile writes data to path with the given permissions.
func (osFileSystem) WriteFile(path string, data []byte, perm os.FileMode) error {
        return os.WriteFile(path, data, perm)
}

// MkdirAll creates path and any missing parents.
func (osFileSystem) MkdirAll(path string, perm os.FileMode) error {
        return os.MkdirAll(path, perm)
}

// anomalyState records a detected clock anomaly. It is swapped atomically so
// readers never take a lock on the status path.
type anomalyState struct {
        // Reason explains the anomaly in operator-readable form.
        Reason string
}

// Service renders chrony's configuration, manages its OpenRC lifecycle, and
// reports the resulting synchronization state. It owns no NTP protocol logic.
type Service struct {
        config      *Config
        paths       Paths
        controller  Controller
        runner      CommandRunner
        fs          FileSystem
        logger      *slog.Logger
        serviceName string
        poll        time.Duration

        firstSyncDone atomic.Bool
        anomaly       atomic.Pointer[anomalyState]
        lastStatus    atomic.Pointer[Status]
}

// Option customizes a Service. Options exist so tests can substitute the
// command runner, filesystem, and control client; production callers use the
// defaults.
type Option func(*Service)

// WithController overrides the chrony control client.
func WithController(controller Controller) Option {
        return func(s *Service) { s.controller = controller }
}

// WithRunner overrides the system command runner.
func WithRunner(runner CommandRunner) Option {
        return func(s *Service) { s.runner = runner }
}

// WithFileSystem overrides the filesystem.
func WithFileSystem(fs FileSystem) Option {
        return func(s *Service) { s.fs = fs }
}

// WithPaths overrides the chrony filesystem layout.
func WithPaths(paths Paths) Option {
        return func(s *Service) { s.paths = paths }
}

// WithServiceName overrides the OpenRC service name.
func WithServiceName(name string) Option {
        return func(s *Service) { s.serviceName = name }
}

// WithLogger sets the service logger.
func WithLogger(logger *slog.Logger) Option {
        return func(s *Service) { s.logger = logger }
}

// WithPollInterval overrides the startup-gate poll cadence.
func WithPollInterval(interval time.Duration) Option {
        return func(s *Service) { s.poll = interval }
}

// NewService builds a time-synchronization service for config. It validates
// the configuration up front so a malformed `ntp:` section is rejected before
// anything touches the host.
func NewService(config *Config, opts ...Option) (*Service, error) {
        if config == nil {
                return nil, ErrNilConfig{}
        }
        if err := config.Validate(); err != nil {
                return nil, err
        }

        s := &Service{
                config:      config,
                paths:       DefaultPaths(),
                fs:          osFileSystem{},
                runner:      execRunner{},
                logger:      slog.Default(),
                serviceName: DefaultServiceName,
                poll:        DefaultPollInterval,
        }
        for _, opt := range opts {
                opt(s)
        }
        if s.controller == nil {
                s.controller = NewController(s.paths.CommandSocket, s.runner)
        }

        return s, nil
}

// Config returns the service's configuration.
func (s *Service) Config() *Config { return s.config }

// Mode returns the configured serve/consume/both role.
func (s *Service) Mode() Mode { return s.config.Mode }

// Apply renders chrony.conf (plus any NTS material), enables the OpenRC
// service, and restarts chronyd when the rendered configuration changed. It
// is idempotent: an unchanged configuration on a running daemon is a no-op,
// so a platformd restart never disturbs a converged clock.
func (s *Service) Apply(ctx context.Context) error {
        if !s.config.Active() {
                return ErrDisabled{}
        }

        if err := s.writeNTSMaterial(); err != nil {
                return err
        }

        rendered, err := Render(s.config, s.paths)
        if err != nil {
                return err
        }

        changed, err := s.writeConfigFile([]byte(rendered))
        if err != nil {
                return err
        }

        if err := s.enableService(ctx); err != nil {
                return err
        }

        if !changed && s.serviceRunning(ctx) {
                s.logger.Debug("ntp: chrony configuration unchanged and daemon running",
                        slog.String("config", s.paths.ConfigFile))
                return nil
        }

        return s.restartService(ctx)
}

// writeNTSMaterial materializes the resolved NTS certificate and key. The
// platform secrets engine replaces the kms:// URIs with PEM material before
// the service starts; a value that still looks like a URI means the engine
// did not run or could not resolve it, which is a hard error rather than a
// silent downgrade to unauthenticated time.
func (s *Service) writeNTSMaterial() error {
        if !s.config.NTSServing() {
                return nil
        }

        nts := s.config.Serve.NTS
        if err := assertResolved("serve.nts.cert-uri", nts.CertURI); err != nil {
                return err
        }
        if err := assertResolved("serve.nts.key-uri", nts.KeyURI); err != nil {
                return err
        }

        if err := s.fs.MkdirAll(s.paths.NTSDumpDir, dirMode); err != nil {
                return ErrWriteConfigFailed{Path: s.paths.NTSDumpDir, Cause: err}
        }
        if err := s.fs.MkdirAll(filepath.Dir(s.paths.NTSCertFile), dirMode); err != nil {
                return ErrWriteConfigFailed{Path: filepath.Dir(s.paths.NTSCertFile), Cause: err}
        }
        if err := s.fs.WriteFile(s.paths.NTSCertFile, []byte(nts.CertURI), certFileMode); err != nil {
                return ErrWriteConfigFailed{Path: s.paths.NTSCertFile, Cause: err}
        }
        if err := s.fs.WriteFile(s.paths.NTSKeyFile, []byte(nts.KeyURI), keyFileMode); err != nil {
                return ErrWriteConfigFailed{Path: s.paths.NTSKeyFile, Cause: err}
        }

        return nil
}

// unresolvedSchemes are the secret-reference schemes that must have been
// substituted by the platform secrets engine before the service runs.
var unresolvedSchemes = []string{"kms://", "secret://", "file://"}

// assertResolved rejects a value that is still a secret reference.
func assertResolved(field, value string) error {
        trimmed := strings.TrimSpace(value)
        if trimmed == "" {
                return ErrMissingNTSMaterial{Field: field}
        }
        for _, scheme := range unresolvedSchemes {
                if strings.HasPrefix(trimmed, scheme) {
                        return ErrUnresolvedNTSMaterial{Field: field, Value: trimmed}
                }
        }
        return nil
}

// writeConfigFile writes chrony.conf and reports whether the content changed.
func (s *Service) writeConfigFile(rendered []byte) (bool, error) {
        if err := s.fs.MkdirAll(filepath.Dir(s.paths.ConfigFile), dirMode); err != nil {
                return false, ErrWriteConfigFailed{Path: filepath.Dir(s.paths.ConfigFile), Cause: err}
        }

        existing, readErr := s.fs.ReadFile(s.paths.ConfigFile)
        if readErr == nil && bytes.Equal(existing, rendered) {
                return false, nil
        }

        if err := s.fs.WriteFile(s.paths.ConfigFile, rendered, configFileMode); err != nil {
                return false, ErrWriteConfigFailed{Path: s.paths.ConfigFile, Cause: err}
        }
        return true, nil
}

// enableService adds chronyd to the default runlevel. rc-update is
// idempotent, so this is safe on every boot.
func (s *Service) enableService(ctx context.Context) error {
        out, err := s.runner.Run(ctx, openRCUpdateBinary, "add", s.serviceName, defaultRunlevel)
        if err != nil {
                return ErrServiceCommandFailed{
                        Service: s.serviceName,
                        Action:  "rc-update add",
                        Output:  string(out),
                        Cause:   err,
                }
        }
        return nil
}

// restartService restarts chronyd so a freshly rendered configuration takes
// effect.
func (s *Service) restartService(ctx context.Context) error {
        out, err := s.runner.Run(ctx, openRCBinary, s.serviceName, "restart")
        if err != nil {
                return ErrServiceCommandFailed{
                        Service: s.serviceName,
                        Action:  "restart",
                        Output:  string(out),
                        Cause:   err,
                }
        }
        s.logger.Info("ntp: chrony restarted with platform-rendered configuration",
                slog.String("service", s.serviceName),
                slog.String("config", s.paths.ConfigFile),
                slog.String("mode", string(s.config.Mode)))
        return nil
}

// serviceRunning reports whether OpenRC considers chronyd started.
func (s *Service) serviceRunning(ctx context.Context) bool {
        _, err := s.runner.Run(ctx, openRCBinary, s.serviceName, "status")
        return err == nil
}

// Status returns the current synchronization state, combining chronyd's
// tracking report with source reachability counts.
func (s *Service) Status(ctx context.Context) (*Status, error) {
        if !s.config.Active() {
                return nil, ErrDisabled{}
        }

        tracking, err := s.controller.Tracking(ctx)
        if err != nil {
                return nil, err
        }

        status := &Status{
                Synced:         tracking.Synced(),
                Stratum:        tracking.Stratum,
                LeapStatus:     tracking.LeapStatus,
                OffsetSeconds:  tracking.SystemTimeOffset,
                RootDispersion: tracking.RootDispersion,
                RootDelay:      tracking.RootDelay,
                LastSync:       tracking.RefTime,
                Mode:           string(s.config.Mode),
                ReferenceID:    tracking.ReferenceID,
                ReferenceName:  tracking.ReferenceName,
        }

        // Source counts are informational: a control socket that answers
        // tracking but not sources still yields a usable status.
        if sources, srcErr := s.controller.Sources(ctx); srcErr == nil {
                status.SourcesTotal = uint32(len(sources))
                for i := range sources {
                        if sources[i].Reachable {
                                status.SourcesReachable++
                        }
                }
        } else {
                s.logger.Debug("ntp: source enumeration failed", slog.String("error", srcErr.Error()))
        }

        s.detectAnomaly(tracking)
        if state := s.anomaly.Load(); state != nil {
                status.Anomaly = true
                status.AnomalyReason = state.Reason
        }

        s.lastStatus.Store(status)
        return status, nil
}

// detectAnomaly flags a large correction observed after the first-boot
// correction. Before the first sync, large steps are the expected repair of a
// stale RTC; afterwards they mean a failing oscillator or a hostile source.
func (s *Service) detectAnomaly(tracking *Tracking) {
        if !tracking.Synced() {
                return
        }
        if !s.firstSyncDone.Load() {
                s.firstSyncDone.Store(true)
                return
        }
        if s.anomaly.Load() != nil {
                return
        }

        offset := tracking.LastOffset
        if offset < 0 {
                offset = -offset
        }
        if time.Duration(offset*float64(time.Second)) <= LargeStepThreshold {
                return
        }

        reason := "clock corrected by " + formatSeconds(tracking.LastOffset) +
                "s after the first-boot correction"
        s.anomaly.Store(&anomalyState{Reason: reason})
        s.logger.Warn("ntp: large clock correction after initial sync",
                slog.Float64("offset_seconds", tracking.LastOffset),
                slog.String("reference", tracking.ReferenceName))
}

// Sources returns the configured time sources and their measurement state.
func (s *Service) Sources(ctx context.Context) ([]Source, error) {
        if !s.config.Active() {
                return nil, ErrDisabled{}
        }
        return s.controller.Sources(ctx)
}

// Step forces an immediate clock correction. It refuses when chronyd has no
// reachable source, because a step with nothing to step towards would move
// the clock on stale data.
func (s *Service) Step(ctx context.Context) (*Status, error) {
        if !s.config.Active() {
                return nil, ErrDisabled{}
        }

        sources, err := s.controller.Sources(ctx)
        if err != nil {
                return nil, err
        }

        reachable := 0
        for i := range sources {
                if sources[i].Reachable {
                        reachable++
                }
        }
        if reachable == 0 {
                return nil, ErrNoReachableSource{Total: len(sources)}
        }

        if err := s.controller.MakeStep(ctx); err != nil {
                return nil, err
        }

        return s.Status(ctx)
}

// WaitForSync blocks until chronyd reports a synchronized clock or the
// startup gate expires. When the gate is disabled it returns immediately.
// On success it optionally writes the corrected time back to the RTC so the
// next cold boot starts sane even before the network is up.
func (s *Service) WaitForSync(ctx context.Context) error {
        if !s.config.GateEnabled() {
                return nil
        }

        timeout := s.config.GateTimeout()
        gateCtx, cancel := context.WithTimeout(ctx, timeout)
        defer cancel()

        ticker := time.NewTicker(s.poll)
        defer ticker.Stop()

        for {
                status, err := s.Status(gateCtx)
                switch {
                case err != nil:
                        s.logger.Debug("ntp: waiting for chrony control socket",
                                slog.String("error", err.Error()))
                case status.Synced:
                        s.logger.Info("ntp: clock synchronized",
                                slog.Uint64("stratum", uint64(status.Stratum)),
                                slog.Float64("offset_seconds", status.OffsetSeconds),
                                slog.String("reference", status.ReferenceName))
                        s.writeRTC(ctx)
                        return nil
                }

                select {
                case <-gateCtx.Done():
                        return ErrSyncTimeout{Seconds: timeout.Seconds()}
                case <-ticker.C:
                }
        }
}

// writeRTC persists the synchronized system time to the hardware clock. A
// failure is logged, not fatal: the system clock is already correct and the
// health surface reports the degraded state.
func (s *Service) writeRTC(ctx context.Context) {
        if !s.config.RTCWriteBack() {
                return
        }
        out, err := s.runner.Run(ctx, hwclockBinary, "--systohc", "--utc")
        if err != nil {
                s.logger.Warn("ntp: hardware clock write-back failed",
                        slog.String("error", ErrRTCWriteFailed{Output: string(out), Cause: err}.Error()))
                return
        }
        s.logger.Info("ntp: system time written back to the hardware clock")
}

// LastStatus returns the most recently observed status without contacting
// chronyd. It returns nil before the first successful poll.
func (s *Service) LastStatus() *Status { return s.lastStatus.Load() }

// formatSeconds renders a signed offset with millisecond resolution.
func formatSeconds(value float64) string {
        return strconv.FormatFloat(value, 'f', 3, 64)
}

package ntp

import "time"

// LeapNotSynchronised is the leap status chronyd reports while the clock is
// not disciplined. It is the primary "am I synchronized?" signal.
const LeapNotSynchronised = "Not synchronised"

// UnsyncedStratum is the stratum chronyd reports when it has no reference.
const UnsyncedStratum = 0

// MaxStratum is the first stratum value that means "unusable" in NTP.
const MaxStratum = 16

// LargeStepThreshold is the correction magnitude that, observed AFTER the
// first-boot correction, is treated as an anomaly worth flagging in health
// status rather than silently accepted. A healthy disciplined clock stays
// orders of magnitude below this.
const LargeStepThreshold = time.Second

// Status is the platform's view of chronyd's tracking state.
type Status struct {
        // Synced reports whether chronyd currently disciplines the clock.
        Synced bool `json:"synced"`

        // Stratum is the node's distance from a reference clock. 0 means
        // unsynchronized; 16 means unusable.
        Stratum uint32 `json:"stratum"`

        // LeapStatus is chronyd's leap indicator ("Normal", "Insert second",
        // "Delete second", "Not synchronised").
        LeapStatus string `json:"leap_status"`

        // OffsetSeconds is the current system time offset. Positive means the
        // system clock is ahead of the reference.
        OffsetSeconds float64 `json:"offset_seconds"`

        // RootDispersion is the accumulated dispersion to the reference clock,
        // in seconds. It bounds the maximum error.
        RootDispersion float64 `json:"root_dispersion"`

        // RootDelay is the accumulated round-trip delay to the reference
        // clock, in seconds.
        RootDelay float64 `json:"root_delay"`

        // LastSync is the reference timestamp of the last successful update.
        LastSync time.Time `json:"last_sync"`

        // Mode echoes the configured serve/consume/both role.
        Mode string `json:"mode"`

        // ReferenceID is chronyd's hex reference identifier.
        ReferenceID string `json:"reference_id"`

        // ReferenceName is the human-readable name or address of the current
        // reference.
        ReferenceName string `json:"reference_name"`

        // SourcesReachable is the number of sources chronyd can currently
        // reach.
        SourcesReachable uint32 `json:"sources_reachable"`

        // SourcesTotal is the number of configured sources.
        SourcesTotal uint32 `json:"sources_total"`

        // Anomaly is set when a large correction was observed after the
        // first-boot correction, which on a trusted platform indicates either
        // a failing oscillator or a hostile time source.
        Anomaly bool `json:"anomaly"`

        // AnomalyReason explains Anomaly in operator-readable form.
        AnomalyReason string `json:"anomaly_reason,omitempty"`
}

// Healthy reports whether the node's clock can be trusted for attestation,
// certificate, and token validity decisions.
func (s *Status) Healthy() bool {
        if s == nil {
                return false
        }
        return s.Synced && !s.Anomaly && s.Stratum > UnsyncedStratum && s.Stratum < MaxStratum
}

// SourceState is the single-character selection state chronyd assigns to a
// source, expanded into a readable word.
type SourceState string

const (
        // SourceStateSelected is chronyd's current synchronization source.
        SourceStateSelected SourceState = "selected"

        // SourceStateCombined participates in the combined estimate.
        SourceStateCombined SourceState = "combined"

        // SourceStateExcluded was excluded by the combining algorithm.
        SourceStateExcluded SourceState = "excluded"

        // SourceStateUnreachable has not answered recently.
        SourceStateUnreachable SourceState = "unreachable"

        // SourceStateFalseTicker disagrees with the majority.
        SourceStateFalseTicker SourceState = "falseticker"

        // SourceStateJittery has too much variability to be usable.
        SourceStateJittery SourceState = "jittery"

        // SourceStateUnknown covers any state chronyd adds in the future.
        SourceStateUnknown SourceState = "unknown"
)

// sourceStates maps chronyd's single-character state column to the readable
// SourceState vocabulary (map-based for O(1) lookup).
var sourceStates = map[string]SourceState{
        "*": SourceStateSelected,
        "+": SourceStateCombined,
        "-": SourceStateExcluded,
        "?": SourceStateUnreachable,
        "x": SourceStateFalseTicker,
        "~": SourceStateJittery,
}

// parseSourceState converts chronyd's state column to a SourceState.
func parseSourceState(raw string) SourceState {
        if state, ok := sourceStates[raw]; ok {
                return state
        }
        return SourceStateUnknown
}

// SourceMode is chronyd's source-kind column, expanded into a readable word.
type SourceMode string

const (
        // SourceModeServer is a unicast NTP server.
        SourceModeServer SourceMode = "server"

        // SourceModePeer is a symmetric peer.
        SourceModePeer SourceMode = "peer"

        // SourceModeRefClock is a local reference clock (PPS, GPS).
        SourceModeRefClock SourceMode = "refclock"

        // SourceModeUnknown covers any mode chronyd adds in the future.
        SourceModeUnknown SourceMode = "unknown"
)

// sourceModes maps chronyd's mode column to the readable SourceMode
// vocabulary (map-based for O(1) lookup).
var sourceModes = map[string]SourceMode{
        "^": SourceModeServer,
        "=": SourceModePeer,
        "#": SourceModeRefClock,
}

// parseSourceMode converts chronyd's mode column to a SourceMode.
func parseSourceMode(raw string) SourceMode {
        if mode, ok := sourceModes[raw]; ok {
                return mode
        }
        return SourceModeUnknown
}

// Source is one configured time source and its live measurement state.
type Source struct {
        // Address is the source's host name or IP.
        Address string `json:"address"`

        // Mode is server, peer, or refclock.
        Mode SourceMode `json:"mode"`

        // State is chronyd's selection verdict for this source.
        State SourceState `json:"state"`

        // Stratum is the source's own distance from a reference clock.
        Stratum uint32 `json:"stratum"`

        // PollSeconds is the current polling interval in seconds.
        PollSeconds uint32 `json:"poll_seconds"`

        // Reach is the 8-bit reachability shift register. 255 means the last
        // eight polls all succeeded.
        Reach uint32 `json:"reach"`

        // LastRxSeconds is how long ago the last sample arrived.
        LastRxSeconds int64 `json:"last_rx_seconds"`

        // OffsetSeconds is the last measured offset for this source.
        OffsetSeconds float64 `json:"offset_seconds"`

        // Reachable reports whether chronyd has heard from the source
        // recently (any bit set in the reachability register).
        Reachable bool `json:"reachable"`
}

// thin binding over the same in-process NTPServiceServer that backs gRPC,

// basePath is the REST prefix for time-synchronization endpoints.
const basePath = "/ntp"

// NTPRouter wires REST endpoints onto an NTPServiceServer.
type NTPRouter struct {
        logger     *logging.Logger
        server     platformv1.NTPServiceServer
        respWriter response.HttpWriter
        router.WebServiceRouter
}

// NewNTPRouter creates a new time-synchronization REST router.
func NewNTPRouter(
        logger *logging.Logger,
        server platformv1.NTPServiceServer,
        respWriter response.HttpWriter,
) *NTPRouter {
        return &NTPRouter{logger: logger, server: server, respWriter: respWriter}
}

// RegisterRoutes mounts the time-synchronization routes.
//
//        GET  /api/v1/ntp/status   - GetStatus
//        GET  /api/v1/ntp/sources  - ListSources
//        POST /api/v1/ntp/step     - Step (operator recovery, ntp:write)
func (r *NTPRouter) RegisterRoutes(parent *mux.Router) {
        sr := parent.PathPrefix(basePath).Subrouter()
        sr.HandleFunc("/status", r.getStatus).Methods(http.MethodGet)
        sr.HandleFunc("/sources", r.listSources).Methods(http.MethodGet)
        sr.HandleFunc("/step", r.step).Methods(http.MethodPost)
}

// ensure reports whether the backing service is configured, writing a 503
// when the node has no `ntp:` section.
func (r *NTPRouter) ensure(w http.ResponseWriter, req *http.Request) bool {
        if r.server == nil {
                protorest.WriteError(w, req, r.respWriter, &protorest.ErrServiceUnavailable{Service: "ntp"})
                return false
        }
        return true
}

// getStatus handles GET /api/v1/ntp/status.
func (r *NTPRouter) getStatus(w http.ResponseWriter, req *http.Request) {
        if !r.ensure(w, req) {
                return
        }
        resp, err := r.server.GetStatus(req.Context(), &platformv1.GetNTPStatusRequest{})
        if err != nil {
                protorest.WriteError(w, req, r.respWriter, err)
                return
        }
        protorest.WriteSuccess(w, req, r.respWriter, resp)
}

// listSources handles GET /api/v1/ntp/sources.
func (r *NTPRouter) listSources(w http.ResponseWriter, req *http.Request) {
        if !r.ensure(w, req) {
                return
        }
        resp, err := r.server.ListSources(req.Context(), &platformv1.ListNTPSourcesRequest{})
        if err != nil {
                protorest.WriteError(w, req, r.respWriter, err)
                return
        }
        protorest.WriteSuccess(w, req, r.respWriter, resp)
}

// step handles POST /api/v1/ntp/step. The handler returns 409 Conflict when
// no source is reachable: protorest maps the service's FailedPrecondition
// status onto that code, so a step request that cannot be satisfied is
// rejected rather than silently moving the clock on stale data.
func (r *NTPRouter) step(w http.ResponseWriter, req *http.Request) {
        if !r.ensure(w, req) {
                return
        }
        resp, err := r.server.Step(req.Context(), &platformv1.StepNTPClockRequest{})
        if err != nil {
                protorest.WriteError(w, req, r.respWriter, err)
                return
        }
        protorest.WriteSuccess(w, req, r.respWriter, resp)
}
