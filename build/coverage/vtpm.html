// Archiver handles bundle export/import operations.
type Archiver struct {
        config *Config
        logger *logging.Logger
}

// NewArchiver creates a new bundle archiver.
func NewArchiver(config *Config, logger *logging.Logger) *Archiver {
        if logger == nil {
                logger = logging.DefaultLogger()
        }
        return &Archiver{
                config: config,
                logger: logger,
        }
}

// Export exports a bundle to a tar.gz archive.
func (a *Archiver) Export(ctx context.Context, bundleID string, w io.Writer) error {
        bundlePath := a.config.BundlePath(bundleID)

        // Verify bundle exists
        manifestPath := filepath.Join(bundlePath, "manifest.json")
        if _, err := os.Stat(manifestPath); err != nil {
                return ErrBundleNotFound{ID: bundleID}
        }

        a.logger.Info("Exporting bundle", "id", bundleID)

        // Create gzip writer
        gzw := gzip.NewWriter(w)
        defer gzw.Close()

        // Create tar writer
        tw := tar.NewWriter(gzw)
        defer tw.Close()

        // Walk the bundle directory and add files
        archivePrefix := fmt.Sprintf("vtpm-bundle-%s", bundleID)

        err := filepath.Walk(bundlePath, func(path string, info os.FileInfo, err error) error {
                if err != nil {
                        return err
                }

                // Get relative path from bundle root
                relPath, err := filepath.Rel(bundlePath, path)
                if err != nil {
                        return err
                }

                // Create archive path with prefix
                archivePath := filepath.Join(archivePrefix, relPath)

                // Create tar header
                header, err := tar.FileInfoHeader(info, "")
                if err != nil {
                        return err
                }

                header.Name = archivePath

                // Handle directories
                if info.IsDir() {
                        header.Name += "/"
                        if err := tw.WriteHeader(header); err != nil {
                                return err
                        }
                        return nil
                }

                // Write file header
                if err := tw.WriteHeader(header); err != nil {
                        return err
                }

                // Write file content
                f, err := os.Open(path)
                if err != nil {
                        return err
                }
                defer f.Close()

                if _, err := io.Copy(tw, f); err != nil {
                        return err
                }

                return nil
        })

        if err != nil {
                return ErrArchiveOperationFailed{Operation: "export", Cause: err}
        }

        a.logger.Info("Bundle exported successfully", "id", bundleID)
        return nil
}

// Import imports a bundle from a tar.gz archive.
func (a *Archiver) Import(ctx context.Context, r io.Reader) (*Bundle, error) {
        a.logger.Info("Importing bundle from archive")

        // Create gzip reader
        gzr, err := gzip.NewReader(r)
        if err != nil {
                return nil, ErrArchiveOperationFailed{Operation: "import", Cause: fmt.Errorf("failed to create gzip reader: %w", err)}
        }
        defer gzr.Close()

        // Create tar reader
        tr := tar.NewReader(gzr)

        // First pass: read manifest to get bundle ID
        var manifest *Manifest
        var archivePrefix string

        for {
                header, err := tr.Next()
                if err == io.EOF {
                        break
                }
                if err != nil {
                        return nil, ErrArchiveOperationFailed{Operation: "import", Cause: fmt.Errorf("failed to read tar header: %w", err)}
                }

                // Look for manifest.json
                if strings.HasSuffix(header.Name, "manifest.json") {
                        data, err := io.ReadAll(tr)
                        if err != nil {
                                return nil, ErrArchiveOperationFailed{Operation: "import", Cause: fmt.Errorf("failed to read manifest: %w", err)}
                        }

                        manifest = &Manifest{}
                        if err := json.Unmarshal(data, manifest); err != nil {
                                return nil, ErrInvalidManifest{Reason: fmt.Sprintf("failed to parse manifest: %v", err)}
                        }

                        // Extract archive prefix from manifest path
                        archivePrefix = filepath.Dir(header.Name)
                        break
                }
        }

        if manifest == nil {
                return nil, ErrInvalidManifest{Reason: "manifest.json not found in archive"}
        }

        // Note: archivePrefix is used for context but extraction requires re-reading
        _ = archivePrefix

        // Check if bundle already exists
        bundlePath := a.config.BundlePath(manifest.ID)
        if _, err := os.Stat(bundlePath); err == nil {
                return nil, ErrBundleAlreadyExists{ID: manifest.ID}
        }

        // Create bundle directory
        if err := os.MkdirAll(bundlePath, 0700); err != nil {
                return nil, ErrArchiveOperationFailed{Operation: "import", Cause: fmt.Errorf("failed to create bundle directory: %w", err)}
        }

        // We need to re-read the archive from the beginning
        // Since we can't seek on the reader, we need to close and reopen
        // For now, we'll extract what we can and note this limitation

        // Write manifest first (we already have it)
        manifestPath := filepath.Join(bundlePath, "manifest.json")
        manifestData, _ := json.MarshalIndent(manifest, "", "  ")
        if err := os.WriteFile(manifestPath, manifestData, 0600); err != nil {
                os.RemoveAll(bundlePath)
                return nil, ErrArchiveOperationFailed{Operation: "import", Cause: fmt.Errorf("failed to write manifest: %w", err)}
        }

        a.logger.Warn("Import limitation: archive must be re-read for full extraction - manifest imported successfully")

        // Return partial bundle (manifest only for now)
        bundle := &Bundle{
                ID:        manifest.ID,
                Mode:      manifest.Mode,
                Manifest:  manifest,
                CreatedAt: manifest.CreatedAt,
                TPMState: &TPMState{
                        StateDir:    filepath.Join(bundlePath, "tpm"),
                        PermallFile: filepath.Join(bundlePath, "tpm", "tpm2-00.permall"),
                },
                OVMFState: &OVMFState{
                        NVRAMFile: filepath.Join(bundlePath, "ovmf", "OVMF_VARS.fd"),
                },
        }

        return bundle, nil
}

// ImportFromFile imports a bundle from a tar.gz file (supports full extraction).
func (a *Archiver) ImportFromFile(ctx context.Context, archivePath string) (*Bundle, error) {
        a.logger.Info("Importing bundle from file", "path", archivePath)

        f, err := os.Open(archivePath)
        if err != nil {
                return nil, ErrArchiveOperationFailed{Operation: "import", Cause: fmt.Errorf("failed to open archive: %w", err)}
        }
        defer f.Close()

        // Create gzip reader
        gzr, err := gzip.NewReader(f)
        if err != nil {
                return nil, ErrArchiveOperationFailed{Operation: "import", Cause: fmt.Errorf("failed to create gzip reader: %w", err)}
        }
        defer gzr.Close()

        // Create tar reader
        tr := tar.NewReader(gzr)

        // First pass: find manifest
        var manifest *Manifest
        var archivePrefix string

        for {
                header, err := tr.Next()
                if err == io.EOF {
                        break
                }
                if err != nil {
                        return nil, ErrArchiveOperationFailed{Operation: "import", Cause: err}
                }

                if strings.HasSuffix(header.Name, "manifest.json") && !header.FileInfo().IsDir() {
                        data, err := io.ReadAll(tr)
                        if err != nil {
                                return nil, ErrArchiveOperationFailed{Operation: "import", Cause: err}
                        }

                        manifest = &Manifest{}
                        if err := json.Unmarshal(data, manifest); err != nil {
                                return nil, ErrInvalidManifest{Reason: err.Error()}
                        }

                        archivePrefix = filepath.Dir(header.Name)
                        break
                }
        }

        if manifest == nil {
                return nil, ErrInvalidManifest{Reason: "manifest.json not found"}
        }

        // Check if bundle exists
        bundlePath := a.config.BundlePath(manifest.ID)
        if _, err := os.Stat(bundlePath); err == nil {
                return nil, ErrBundleAlreadyExists{ID: manifest.ID}
        }

        // Reopen and extract
        f.Seek(0, 0)
        gzr2, _ := gzip.NewReader(f)
        defer gzr2.Close()
        tr2 := tar.NewReader(gzr2)

        for {
                header, err := tr2.Next()
                if err == io.EOF {
                        break
                }
                if err != nil {
                        os.RemoveAll(bundlePath)
                        return nil, ErrArchiveOperationFailed{Operation: "import", Cause: err}
                }

                // Calculate destination path
                relPath := strings.TrimPrefix(header.Name, archivePrefix+"/")
                if relPath == "" || relPath == header.Name {
                        continue
                }

                destPath := filepath.Join(bundlePath, relPath)

                // Security check: prevent path traversal
                if !strings.HasPrefix(destPath, bundlePath) {
                        continue
                }

                switch header.Typeflag {
                case tar.TypeDir:
                        if err := os.MkdirAll(destPath, 0700); err != nil {
                                os.RemoveAll(bundlePath)
                                return nil, ErrArchiveOperationFailed{Operation: "import", Cause: err}
                        }

                case tar.TypeReg:
                        // Create parent directory
                        if err := os.MkdirAll(filepath.Dir(destPath), 0700); err != nil {
                                os.RemoveAll(bundlePath)
                                return nil, ErrArchiveOperationFailed{Operation: "import", Cause: err}
                        }

                        // Create file
                        outFile, err := os.OpenFile(destPath, os.O_CREATE|os.O_WRONLY|os.O_TRUNC, 0600)
                        if err != nil {
                                os.RemoveAll(bundlePath)
                                return nil, ErrArchiveOperationFailed{Operation: "import", Cause: err}
                        }

                        if _, err := io.Copy(outFile, tr2); err != nil {
                                outFile.Close()
                                os.RemoveAll(bundlePath)
                                return nil, ErrArchiveOperationFailed{Operation: "import", Cause: err}
                        }
                        outFile.Close()
                }
        }

        a.logger.Info("Bundle imported successfully", "id", manifest.ID)

        // Build bundle struct
        bundle := &Bundle{
                ID:        manifest.ID,
                Mode:      manifest.Mode,
                Manifest:  manifest,
                CreatedAt: manifest.CreatedAt,
                TPMState: &TPMState{
                        StateDir:    filepath.Join(bundlePath, "tpm"),
                        PermallFile: filepath.Join(bundlePath, "tpm", "tpm2-00.permall"),
                },
                OVMFState: &OVMFState{
                        NVRAMFile: filepath.Join(bundlePath, "ovmf", "OVMF_VARS.fd"),
                },
        }

        if manifest.SecureBootEnabled {
                sbDir := filepath.Join(bundlePath, "secure-boot-keys")
                bundle.SecureBootKeys = &SecureBootKeyPaths{
                        Dir:     sbDir,
                        PKKey:   filepath.Join(sbDir, "PK.key"),
                        PKCert:  filepath.Join(sbDir, "PK.crt"),
                        KEKKey:  filepath.Join(sbDir, "KEK.key"),
                        KEKCert: filepath.Join(sbDir, "KEK.crt"),
                        DBKey:   filepath.Join(sbDir, "DB.key"),
                        DBCert:  filepath.Join(sbDir, "DB.crt"),
                }
        }

        return bundle, nil
}

// ValidateArchive checks archive integrity without extracting.
func (a *Archiver) ValidateArchive(ctx context.Context, r io.Reader) (*Manifest, error) {
        gzr, err := gzip.NewReader(r)
        if err != nil {
                return nil, ErrArchiveOperationFailed{Operation: "validate", Cause: err}
        }
        defer gzr.Close()

        tr := tar.NewReader(gzr)

        var manifest *Manifest

        for {
                header, err := tr.Next()
                if err == io.EOF {
                        break
                }
                if err != nil {
                        return nil, ErrArchiveOperationFailed{Operation: "validate", Cause: err}
                }

                if strings.HasSuffix(header.Name, "manifest.json") && !header.FileInfo().IsDir() {
                        data, err := io.ReadAll(tr)
                        if err != nil {
                                return nil, ErrArchiveOperationFailed{Operation: "validate", Cause: err}
                        }

                        manifest = &Manifest{}
                        if err := json.Unmarshal(data, manifest); err != nil {
                                return nil, ErrInvalidManifest{Reason: err.Error()}
                        }
                        break
                }
        }

        if manifest == nil {
                return nil, ErrInvalidManifest{Reason: "manifest.json not found"}
        }

        a.logger.Debug("Archive validation passed", "bundleID", manifest.ID)
        return manifest, nil
}

// Config holds the configuration for the vTPM service.
type Config struct {
        // BaseDir is the base directory for vTPM bundles and templates.
        // Bundles are stored in BaseDir/bundles/{id}/
        // Templates are stored in BaseDir/templates/{id}/
        BaseDir string `yaml:"base-dir" json:"base_dir"`

        // OVMFCode is the path to OVMF firmware code.
        OVMFCode string `yaml:"ovmf-code" json:"ovmf_code"`

        // OVMFCodeSecure is the path to OVMF firmware code with Secure Boot.
        OVMFCodeSecure string `yaml:"ovmf-code-secure" json:"ovmf_code_secure"`

        // OVMFVars is the path to OVMF NVRAM template.
        OVMFVars string `yaml:"ovmf-vars" json:"ovmf_vars"`

        // PCRBanks specifies the PCR banks for swtpm initialization.
        // Default: ["sha256"]
        PCRBanks []string `yaml:"pcr-banks" json:"pcr_banks"`

        // SecureBootKeysDir is the optional path to existing Secure Boot keys.
        // If not set, keys will be generated for each bundle/template.
        SecureBootKeysDir string `yaml:"secure-boot-keys-dir" json:"secure_boot_keys_dir"`

        // SecureBootKeySize is the RSA key size for Secure Boot keys.
        // Default: 2048
        SecureBootKeySize int `yaml:"secure-boot-key-size" json:"secure_boot_key_size"`

        // SecureBootValidYears is the certificate validity in years.
        // Default: 10
        SecureBootValidYears int `yaml:"secure-boot-valid-years" json:"secure_boot_valid_years"`

        // SecureBootCommonName is the CN prefix for Secure Boot certificates.
        // Default: "Trusted Platform"
        SecureBootCommonName string `yaml:"secure-boot-common-name" json:"secure_boot_common_name"`

        // SecureBootOrganization is the organization for Secure Boot certificates.
        // Default: "Trusted Platform"
        SecureBootOrganization string `yaml:"secure-boot-organization" json:"secure_boot_organization"`

        // EKAlgorithm is the default EK key algorithm.
        // Default: "RSA"
        EKAlgorithm EKAlgorithm `yaml:"ek-algorithm" json:"ek_algorithm"`

        // Debug enables debug output.
        Debug bool `yaml:"debug" json:"debug"`
}

// DefaultConfig returns a Config with sensible defaults.
func DefaultConfig() *Config {
        return &Config{
                BaseDir:                "/var/lib/trusted-platform/vtpm",
                OVMFCode:               "/usr/share/OVMF/OVMF_CODE_4M.fd",
                OVMFCodeSecure:         "/usr/share/OVMF/OVMF_CODE_4M.secboot.fd",
                OVMFVars:               "/usr/share/OVMF/OVMF_VARS_4M.fd",
                PCRBanks:               []string{"sha256"},
                SecureBootKeySize:      2048,
                SecureBootValidYears:   10,
                SecureBootCommonName:   "Trusted Platform",
                SecureBootOrganization: "Trusted Platform",
                EKAlgorithm:            EKAlgorithmRSA,
                Debug:                  false,
        }
}

// Validate validates the configuration.
func (c *Config) Validate() error {
        if c.BaseDir == "" {
                return ErrInvalidConfig{Field: "BaseDir", Reason: "cannot be empty"}
        }

        if c.OVMFCode == "" {
                return ErrInvalidConfig{Field: "OVMFCode", Reason: "cannot be empty"}
        }

        if c.OVMFVars == "" {
                return ErrInvalidConfig{Field: "OVMFVars", Reason: "cannot be empty"}
        }

        // Check OVMF files exist
        if _, err := os.Stat(c.OVMFCode); err != nil {
                return ErrOVMFNotFound{Path: c.OVMFCode}
        }

        if _, err := os.Stat(c.OVMFVars); err != nil {
                return ErrOVMFNotFound{Path: c.OVMFVars}
        }

        // OVMFCodeSecure is optional, only validate if specified
        if c.OVMFCodeSecure != "" {
                if _, err := os.Stat(c.OVMFCodeSecure); err != nil {
                        return ErrOVMFNotFound{Path: c.OVMFCodeSecure}
                }
        }

        // Validate PCR banks
        if len(c.PCRBanks) == 0 {
                return ErrInvalidConfig{Field: "PCRBanks", Reason: "at least one PCR bank is required"}
        }

        validBanks := map[string]bool{"sha1": true, "sha256": true, "sha384": true, "sha512": true}
        for _, bank := range c.PCRBanks {
                if !validBanks[bank] {
                        return ErrInvalidConfig{Field: "PCRBanks", Reason: fmt.Sprintf("invalid PCR bank: %s", bank)}
                }
        }

        // Validate Secure Boot key size
        if c.SecureBootKeySize < 2048 {
                return ErrInvalidConfig{Field: "SecureBootKeySize", Reason: "must be at least 2048"}
        }

        // Validate EK algorithm
        if c.EKAlgorithm != "" && !c.EKAlgorithm.IsValid() {
                return ErrInvalidConfig{Field: "EKAlgorithm", Reason: fmt.Sprintf("invalid algorithm: %s", c.EKAlgorithm)}
        }

        // Validate Secure Boot keys directory if specified
        if c.SecureBootKeysDir != "" {
                if _, err := os.Stat(c.SecureBootKeysDir); err != nil {
                        return ErrInvalidConfig{Field: "SecureBootKeysDir", Reason: fmt.Sprintf("directory not found: %s", c.SecureBootKeysDir)}
                }
        }

        return nil
}

// BundlesDir returns the path to the bundles directory.
func (c *Config) BundlesDir() string {
        return filepath.Join(c.BaseDir, "bundles")
}

// TemplatesDir returns the path to the templates directory.
func (c *Config) TemplatesDir() string {

                return c.OVMFCodeSecure
        }
        return c.OVMFCode
}

// ApplyDefaults applies default values to empty fields.
func (c *Config) ApplyDefaults() {
        defaults := DefaultConfig()

        if c.BaseDir == "" {
                c.BaseDir = defaults.BaseDir
        }
        if c.OVMFCode == "" {
                c.OVMFCode = defaults.OVMFCode
        }
        if c.OVMFCodeSecure == "" {
                c.OVMFCodeSecure = defaults.OVMFCodeSecure
        }
        if c.OVMFVars == "" {
                c.OVMFVars = defaults.OVMFVars
        }
        if len(c.PCRBanks) == 0 {
                c.PCRBanks = defaults.PCRBanks
        }
        if c.SecureBootKeySize == 0 {
                c.SecureBootKeySize = defaults.SecureBootKeySize
        }
        if c.SecureBootValidYears == 0 {
                c.SecureBootValidYears = defaults.SecureBootValidYears
        }
        if c.SecureBootCommonName == "" {
                c.SecureBootCommonName = defaults.SecureBootCommonName
        }
        if c.SecureBootOrganization == "" {
                c.SecureBootOrganization = defaults.SecureBootOrganization
        }
        if c.EKAlgorithm == "" {
                c.EKAlgorithm = defaults.EKAlgorithm
        }
}

// NV RAM indices for EK certificates (TCG PC Client Platform spec)
const (
        // NVIndexRSAEKCert is the NV index for RSA EK certificates.
        NVIndexRSAEKCert uint32 = 0x01C00002

        // NVIndexECCEKCert is the NV index for ECC EK certificates.
        NVIndexECCEKCert uint32 = 0x01C0000A

        // DefaultNVBufferMax is the default maximum NV buffer size for writes.
        // Most TPMs support at least 1024 bytes per write.
        DefaultNVBufferMax uint16 = 1024
)

// EKCertInjector handles EK certificate injection into swtpm NV RAM.
type EKCertInjector struct {
        socketPath  string
        logger      *logging.Logger
        nvBufferMax uint16
}

// NewEKCertInjector creates a new EK certificate injector.
func NewEKCertInjector(socketPath string, logger *logging.Logger) *EKCertInjector {
        if logger == nil {
                logger = logging.DefaultLogger()
        }
        return &EKCertInjector{
                socketPath:  socketPath,
                logger:      logger,
                nvBufferMax: DefaultNVBufferMax,
        }
}

// SetNVBufferMax sets the maximum NV buffer size for writes.
// This should match TPM_PT_NV_BUFFER_MAX from the TPM capabilities.
func (e *EKCertInjector) SetNVBufferMax(size uint16) {
        e.nvBufferMax = size
}

// InjectCertificate writes an EK certificate to the appropriate NV index.
func (e *EKCertInjector) InjectCertificate(ctx context.Context, cert *x509.Certificate, algorithm EKAlgorithm) error {
        if cert == nil {
                return ErrEKCertInjectionFailed{Cause: fmt.Errorf("certificate cannot be nil")}
        }

        nvIndex := algorithm.NVIndex()
        certDER := cert.Raw

        e.logger.Info("Injecting EK certificate to NV RAM",
                "algorithm", algorithm,
                "nvIndex", fmt.Sprintf("0x%08X", nvIndex),
                "certSize", len(certDER))

        // Open connection to swtpm socket
        t, err := linuxudstpm.Open(e.socketPath)
        if err != nil {
                return ErrEKCertInjectionFailed{Cause: fmt.Errorf("failed to open swtpm socket: %w", err)}
        }
        defer t.Close()

        // Define the NV space
        if err := e.defineNVSpace(t, nvIndex, uint16(len(certDER))); err != nil {
                return err
        }

        // Write the certificate data
        if err := e.writeNVData(t, nvIndex, certDER); err != nil {
                return err
        }

        e.logger.Info("EK certificate injected successfully",
                "nvIndex", fmt.Sprintf("0x%08X", nvIndex))

        return nil
}

// defineNVSpace defines the NV space for the EK certificate.
func (e *EKCertInjector) defineNVSpace(t transport.TPM, nvIndex uint32, size uint16) error {
        e.logger.Debug("Defining NV space",
                "nvIndex", fmt.Sprintf("0x%08X", nvIndex),
                "size", size)

        // Create NV handle
        nvHandle := tpm2.TPMHandle(nvIndex)

        // Define NV attributes for EK certificate storage
        // TPMA_NV_PPWRITE: Platform hierarchy can write
        // TPMA_NV_PPREAD: Platform hierarchy can read
        // TPMA_NV_OWNERREAD: Owner hierarchy can read
        // TPMA_NV_AUTHREAD: Reads require authorization
        // TPMA_NV_NO_DA: Dictionary attack protection not applicable
        // TPMA_NV_PLATFORMCREATE: Created by platform hierarchy
        nvAttrs := tpm2.TPMANV{
                PPWrite:        true,
                PPRead:         true,
                OwnerRead:      true,
                AuthRead:       true,
                NoDA:           true,
                PlatformCreate: true,
        }

        // Create NV public area
        nvPublic := tpm2.New2B(tpm2.TPMSNVPublic{
                NVIndex:    nvHandle,
                NameAlg:    tpm2.TPMAlgSHA256,
                Attributes: nvAttrs,
                DataSize:   size,
        })

        // Define the NV space using platform hierarchy
        _, err := tpm2.NVDefineSpace{
                AuthHandle: tpm2.TPMRHPlatform,
                PublicInfo: nvPublic,
        }.Execute(t)
        if err != nil {
                // Check if NV index already exists
                if isNVDefinedError(err) {
                        e.logger.Debug("NV index already defined, undefining and redefining",
                                "nvIndex", fmt.Sprintf("0x%08X", nvIndex))

                        // Undefine existing NV space
                        _, undefErr := tpm2.NVUndefineSpace{
                                AuthHandle: tpm2.TPMRHPlatform,
                                NVIndex:    nvHandle,
                        }.Execute(t)
                        if undefErr != nil {
                                return ErrNVRAMWriteFailed{NVIndex: nvIndex, Cause: fmt.Errorf("failed to undefine existing NV space: %w", undefErr)}
                        }

                        // Retry define
                        _, err = tpm2.NVDefineSpace{
                                AuthHandle: tpm2.TPMRHPlatform,
                                PublicInfo: nvPublic,
                        }.Execute(t)
                        if err != nil {
                                return ErrNVRAMWriteFailed{NVIndex: nvIndex, Cause: fmt.Errorf("failed to define NV space after undefine: %w", err)}
                        }
                        return nil
                }
                return ErrNVRAMWriteFailed{NVIndex: nvIndex, Cause: fmt.Errorf("failed to define NV space: %w", err)}
        }

        return nil
}

// isNVDefinedError checks if the error indicates the NV index is already defined.
func isNVDefinedError(err error) bool {
        if tpmErr, ok := err.(tpm2.TPMRC); ok {
                // TPM_RC_NV_DEFINED is 0x14C (base) + format one format
                return uint32(tpmErr)&0xFF == 0x49 // RC_NV_DEFINED
        }
        return false
}

// writeNVData writes data to the NV space in chunks.
func (e *EKCertInjector) writeNVData(t transport.TPM, nvIndex uint32, data []byte) error {
        nvHandle := tpm2.TPMHandle(nvIndex)

        // Write data in chunks respecting nvBufferMax
        offset := uint16(0)
        remaining := len(data)

        for remaining > 0 {
                chunkSize := int(e.nvBufferMax)
                if remaining < chunkSize {
                        chunkSize = remaining
                }

                chunk := data[offset : offset+uint16(chunkSize)]

                e.logger.Debug("Writing NV chunk",
                        "nvIndex", fmt.Sprintf("0x%08X", nvIndex),
                        "offset", offset,
                        "chunkSize", chunkSize,
                        "remaining", remaining)

                _, err := tpm2.NVWrite{
                        AuthHandle: tpm2.TPMRHPlatform,
                        NVIndex:    nvHandle,
                        Data: tpm2.TPM2BMaxNVBuffer{
                                Buffer: chunk,
                        },
                        Offset: offset,
                }.Execute(t)
                if err != nil {
                        return ErrNVRAMWriteFailed{NVIndex: nvIndex, Cause: fmt.Errorf("failed to write NV data at offset %d: %w", offset, err)}
                }

                offset += uint16(chunkSize)
                remaining -= chunkSize
        }

        return nil
}

// ReadCertificate reads an EK certificate from NV RAM.
func (e *EKCertInjector) ReadCertificate(ctx context.Context, algorithm EKAlgorithm) (*x509.Certificate, error) {
        nvIndex := algorithm.NVIndex()

        e.logger.Debug("Reading EK certificate from NV RAM",
                "algorithm", algorithm,
                "nvIndex", fmt.Sprintf("0x%08X", nvIndex))

        // Open connection to swtpm socket
        t, err := linuxudstpm.Open(e.socketPath)
        if err != nil {
                return nil, ErrEKCertInjectionFailed{Cause: fmt.Errorf("failed to open swtpm socket: %w", err)}
        }
        defer t.Close()

        // First, read the NV public to get the size
        nvHandle := tpm2.TPMHandle(nvIndex)

        nvReadPublicResp, err := tpm2.NVReadPublic{
                NVIndex: nvHandle,
        }.Execute(t)
        if err != nil {
                return nil, ErrEKCertInjectionFailed{Cause: fmt.Errorf("failed to read NV public: %w", err)}
        }

        nvPublicContents, err := nvReadPublicResp.NVPublic.Contents()
        if err != nil {
                return nil, ErrEKCertInjectionFailed{Cause: fmt.Errorf("failed to get NV public contents: %w", err)}
        }
        dataSize := nvPublicContents.DataSize

        // Read the data in chunks
        data := make([]byte, 0, dataSize)
        offset := uint16(0)

        for offset < dataSize {
                chunkSize := e.nvBufferMax
                if dataSize-offset < chunkSize {
                        chunkSize = dataSize - offset
                }

                resp, err := tpm2.NVRead{
                        AuthHandle: tpm2.TPMRHOwner,
                        NVIndex:    nvHandle,
                        Size:       chunkSize,
                        Offset:     offset,
                }.Execute(t)
                if err != nil {
                        return nil, ErrEKCertInjectionFailed{Cause: fmt.Errorf("failed to read NV data at offset %d: %w", offset, err)}
                }

                data = append(data, resp.Data.Buffer...)
                offset += chunkSize
        }

        // Parse the certificate
        cert, err := x509.ParseCertificate(data)
        if err != nil {
                return nil, ErrEKCertInjectionFailed{Cause: fmt.Errorf("failed to parse certificate: %w", err)}
        }

        return cert, nil
}

// VerifyCertificate reads back and verifies the stored certificate matches.
func (e *EKCertInjector) VerifyCertificate(ctx context.Context, expected *x509.Certificate, algorithm EKAlgorithm) error {
        actual, err := e.ReadCertificate(ctx, algorithm)
        if err != nil {
                return err
        }

        if !actual.Equal(expected) {
                return ErrEKCertInjectionFailed{Cause: fmt.Errorf("certificate verification failed: stored certificate does not match")}
        }

        e.logger.Info("EK certificate verification successful",
                "nvIndex", fmt.Sprintf("0x%08X", algorithm.NVIndex()))

        return nil
}

// CertificateAuthority interface for CA operations.
// This abstracts the CA dependency to avoid circular imports.
// Implementations should wrap the actual CA and provide EK certificate issuance.
type CertificateAuthority interface {
        // IssueEKCertificate issues an EK certificate for the given public key.
        // The publicKey is the EK public key (RSA or ECC) and hostname is for the certificate CN.
        IssueEKCertificate(ctx context.Context, publicKey crypto.PublicKey, hostname string) (*x509.Certificate, error)
}

// Service implements the VTPMService interface.

}

// NewService creates a new vTPM service.
func NewService(config *Config, logger *logging.Logger, ca CertificateAuthority) (*Service, error) {
        if config == nil {
                config = DefaultConfig()
        }

        config.ApplyDefaults()

        if err := config.Validate(); err != nil {
                return nil, err
        }

        if logger == nil {
                logger = logging.DefaultLogger()
        }

        // Create base directories
        if err := os.MkdirAll(config.BundlesDir(), 0700); err != nil {
                return nil, ErrInvalidConfig{Field: "BaseDir", Reason: fmt.Sprintf("failed to create bundles directory: %v", err)}
        }
        if err := os.MkdirAll(config.TemplatesDir(), 0700); err != nil {
                return nil, ErrInvalidConfig{Field: "BaseDir", Reason: fmt.Sprintf("failed to create templates directory: %v", err)}
        }

        return &Service{
                config: config,
                logger: logger,
                ca:     ca,
        }, nil
}

// CreateBundle creates a new vTPM bundle.
func (s *Service) CreateBundle(ctx context.Context, opts *CreateBundleOptions) (*Bundle, error) {
        s.mu.Lock()
        defer s.mu.Unlock()

        if opts == nil {
                return nil, ErrInvalidConfig{Field: "opts", Reason: "cannot be nil"}
        }

        // Set defaults
        if opts.Mode == "" {
                opts.Mode = BundleModePreAssigned
        }
        if opts.EKAlgorithm == "" {
                opts.EKAlgorithm = s.config.EKAlgorithm
        }

        // Validate options
        if !opts.Mode.IsValid() {
                return nil, ErrInvalidConfig{Field: "Mode", Reason: fmt.Sprintf("invalid mode: %s", opts.Mode)}
        }
        if opts.Mode == BundleModePreAssigned && opts.Hostname == "" {
                return nil, ErrInvalidConfig{Field: "Hostname", Reason: "required for pre-assigned mode"}
        }
        if !opts.EKAlgorithm.IsValid() {
                return nil, ErrInvalidConfig{Field: "EKAlgorithm", Reason: fmt.Sprintf("invalid algorithm: %s", opts.EKAlgorithm)}
        }

        // Generate bundle ID
        bundleID := s.generateBundleID(opts.Hostname)
        bundlePath := s.config.BundlePath(bundleID)

        // Check if bundle already exists
        if _, err := os.Stat(bundlePath); err == nil {
                return nil, ErrBundleAlreadyExists{ID: bundleID}
        }

        s.logger.Info("Creating vTPM bundle",
                "id", bundleID,
                "mode", opts.Mode,
                "hostname", opts.Hostname,
                "secureBoot", opts.SecureBootEnabled)

        // Create bundle directory structure
        if err := s.createBundleDirectories(bundlePath); err != nil {
                return nil, err
        }

        // Build the bundle
        bundle, err := s.buildBundle(ctx, bundleID, bundlePath, opts)
        if err != nil {
                // Cleanup on failure
                os.RemoveAll(bundlePath)
                return nil, err
        }

        return bundle, nil
}

// generateBundleID generates a unique bundle ID.
func (s *Service) generateBundleID(hostname string) string {
        timestamp := time.Now().UnixNano()
        if hostname != "" {
                return fmt.Sprintf("%s-%d", hostname, timestamp)
        }
        return fmt.Sprintf("vtpm-%d", timestamp)
}

// createBundleDirectories creates the bundle directory structure.
func (s *Service) createBundleDirectories(bundlePath string) error {
        dirs := []string{
                bundlePath,
                filepath.Join(bundlePath, "tpm"),
                filepath.Join(bundlePath, "ovmf"),
                filepath.Join(bundlePath, "secure-boot-keys"),
        }

        for _, dir := range dirs {
                if err := os.MkdirAll(dir, 0700); err != nil {
                        return ErrInvalidConfig{Field: "bundlePath", Reason: fmt.Sprintf("failed to create directory %s: %v", dir, err)}
                }
        }

        return nil
}

// buildBundle builds the complete vTPM bundle.
func (s *Service) buildBundle(ctx context.Context, bundleID, bundlePath string, opts *CreateBundleOptions) (*Bundle, error) {
        tpmDir := filepath.Join(bundlePath, "tpm")
        ovmfDir := filepath.Join(bundlePath, "ovmf")
        sbKeysDir := filepath.Join(bundlePath, "secure-boot-keys")

        // Step 1: Initialize swtpm
        s.logger.Debug("Initializing swtpm state")
        swtpmConfig := &swtpm.Config{
                StateDir: tpmDir,
                PCRBanks: s.config.PCRBanks,
                Debug:    s.config.Debug,
        }

        sw, err := swtpm.NewSWTPM(swtpmConfig, s.logger)
        if err != nil {
                return nil, ErrSWTPMOperationFailed{Operation: "create", Cause: err}
        }

        if err := sw.Initialize(ctx); err != nil {
                return nil, ErrSWTPMOperationFailed{Operation: "initialize", Cause: err}
        }

        // Step 2: Start swtpm and send startup
        if err := sw.Start(ctx); err != nil {
                return nil, ErrSWTPMOperationFailed{Operation: "start", Cause: err}
        }
        defer sw.Stop()

        if err := sw.SendStartup(); err != nil {
                return nil, ErrSWTPMOperationFailed{Operation: "startup", Cause: err}
        }

        // Step 3: For pre-assigned mode, inject EK certificate
        var ekCertInfo *EKCertInfo
        if opts.Mode == BundleModePreAssigned && s.ca != nil {
                ekCertInfo, err = s.injectEKCertificate(ctx, sw.SocketPath(), opts)
                if err != nil {
                        return nil, err
                }
        }

        // Step 4: Stop swtpm before copying NVRAM (state is persisted)
        if err := sw.Stop(); err != nil {
                s.logger.Warn("Failed to stop swtpm gracefully", "error", err)
        }

        // Step 5: Copy OVMF NVRAM template
        nvramPath := filepath.Join(ovmfDir, "OVMF_VARS.fd")
        if err := s.copyFile(s.config.OVMFVars, nvramPath); err != nil {
                return nil, ErrInvalidConfig{Field: "OVMFVars", Reason: fmt.Sprintf("failed to copy NVRAM: %v", err)}
        }

        // Step 6: Generate and enroll Secure Boot keys
        var sbKeyInfo *SecureBootKeyInfo
        var sbKeyPaths *SecureBootKeyPaths
        if opts.SecureBootEnabled {
                sbKeyInfo, sbKeyPaths, err = s.setupSecureBoot(ctx, sbKeysDir, nvramPath, bundleID)
                if err != nil {
                        return nil, err
                }
        }

        // Step 7: Generate manifest
        manifest, err := s.generateManifest(bundleID, bundlePath, opts, ekCertInfo, sbKeyInfo)
        if err != nil {
                return nil, err
        }

        // Write manifest
        manifestPath := filepath.Join(bundlePath, "manifest.json")
        manifestData, err := json.MarshalIndent(manifest, "", "  ")
        if err != nil {
                return nil, ErrInvalidManifest{Reason: fmt.Sprintf("failed to marshal manifest: %v", err)}
        }
        if err := os.WriteFile(manifestPath, manifestData, 0600); err != nil {
                return nil, ErrInvalidManifest{Reason: fmt.Sprintf("failed to write manifest: %v", err)}
        }

        // Build bundle struct
        bundle := &Bundle{
                ID:       bundleID,
                Mode:     opts.Mode,
                Manifest: manifest,
                TPMState: &TPMState{
                        StateDir:    tpmDir,
                        PermallFile: filepath.Join(tpmDir, "tpm2-00.permall"),
                },
                OVMFState: &OVMFState{
                        NVRAMFile: nvramPath,
                },
                SecureBootKeys: sbKeyPaths,
                CreatedAt:      manifest.CreatedAt,
        }

        s.logger.Info("vTPM bundle created successfully",
                "id", bundleID,
                "path", bundlePath)

        return bundle, nil
}

// setupSecureBoot generates Secure Boot keys and enrolls them.
func (s *Service) setupSecureBoot(ctx context.Context, keysDir, nvramPath, bundleID string) (*SecureBootKeyInfo, *SecureBootKeyPaths, error) {
        s.logger.Debug("Setting up Secure Boot keys")

        // Generate Secure Boot keys using secureboot.KeychainManager
        sbConfig := &secureboot.KeychainConfig{
                KeysDir:      keysDir,
                Backend:      secureboot.BackendFile,
                CommonName:   s.config.SecureBootCommonName,
                Organization: s.config.SecureBootOrganization,
                KeySize:      s.config.SecureBootKeySize,
                ValidYears:   s.config.SecureBootValidYears,
                ExportKeys:   true,
        }

        km, err := secureboot.NewKeychainManager(sbConfig, s.logger)
        if err != nil {
                return nil, nil, ErrSecureBootKeyGenFailed{Cause: err}
        }
        defer km.Close()

        if err := km.GenerateAll(ctx); err != nil {
                return nil, nil, ErrSecureBootKeyGenFailed{Cause: err}
        }

        // Get key paths and read certificates from files
        keyPaths := km.GetKeyPaths()

        pkCert, err := loadCertificateFromFile(keyPaths.PKCert)
        if err != nil {
                return nil, nil, ErrSecureBootKeyGenFailed{Cause: fmt.Errorf("failed to load PK certificate: %w", err)}
        }
        kekCert, err := loadCertificateFromFile(keyPaths.KEKCert)
        if err != nil {
                return nil, nil, ErrSecureBootKeyGenFailed{Cause: fmt.Errorf("failed to load KEK certificate: %w", err)}
        }
        dbCert, err := loadCertificateFromFile(keyPaths.DBCert)
        if err != nil {
                return nil, nil, ErrSecureBootKeyGenFailed{Cause: fmt.Errorf("failed to load DB certificate: %w", err)}
        }

        // Enroll keys in OVMF NVRAM
        enrollerConfig := &qemu.SecureBootEnrollerConfig{
                KeysDir:      keysDir,
                NVRAMDir:     filepath.Dir(nvramPath),
                OVMFVarsPath: s.config.OVMFVars,
                Debug:        s.config.Debug,
        }

        enroller, err := qemu.NewSecureBootEnroller(enrollerConfig, s.logger)
        if err != nil {
                return nil, nil, ErrSecureBootEnrollmentFailed{Cause: err}
        }

        if err := enroller.EnrollKeys(ctx); err != nil {
                return nil, nil, ErrSecureBootEnrollmentFailed{Cause: err}
        }

        sbKeyInfo := &SecureBootKeyInfo{
                PKFingerprint:  certFingerprint(pkCert),
                KEKFingerprint: certFingerprint(kekCert),
                DBFingerprint:  certFingerprint(dbCert),
                OwnerGUID:      "", // Owner GUID is managed internally by enroller
        }

        sbKeyPaths := &SecureBootKeyPaths{
                Dir:     keysDir,
                PKKey:   filepath.Join(keysDir, "PK.key"),
                PKCert:  filepath.Join(keysDir, "PK.crt"),
                KEKKey:  filepath.Join(keysDir, "KEK.key"),
                KEKCert: filepath.Join(keysDir, "KEK.crt"),
                DBKey:   filepath.Join(keysDir, "DB.key"),
                DBCert:  filepath.Join(keysDir, "DB.crt"),
        }

        s.logger.Info("Secure Boot keys generated and enrolled",
                "keysDir", keysDir)

        return sbKeyInfo, sbKeyPaths, nil
}

// loadCertificateFromFile loads an X.509 certificate from a PEM file.
func loadCertificateFromFile(path string) (*x509.Certificate, error) {
        data, err := os.ReadFile(path)
        if err != nil {
                return nil, err
        }

        block, _ := pem.Decode(data)
        if block == nil {
                return nil, fmt.Errorf("failed to decode PEM block")
        }

        return x509.ParseCertificate(block.Bytes)
}

// generateManifest generates the bundle manifest.
func (s *Service) generateManifest(bundleID, bundlePath string, opts *CreateBundleOptions, ekCertInfo *EKCertInfo, sbKeyInfo *SecureBootKeyInfo) (*Manifest, error) {
        now := time.Now()

        // Calculate checksums
        checksums := make(map[string]string)

        // TPM state file
        tpmPermall := filepath.Join(bundlePath, "tpm", "tpm2-00.permall")
        if hash, err := fileChecksum(tpmPermall); err == nil {
                checksums["tpm/tpm2-00.permall"] = hash
        }

        // OVMF NVRAM
        ovmfVars := filepath.Join(bundlePath, "ovmf", "OVMF_VARS.fd")
        if hash, err := fileChecksum(ovmfVars); err == nil {
                checksums["ovmf/OVMF_VARS.fd"] = hash
        }

        // Secure Boot keys
        if opts.SecureBootEnabled {
                sbFiles := []string{"PK.key", "PK.crt", "KEK.key", "KEK.crt", "DB.key", "DB.crt"}
                for _, f := range sbFiles {
                        path := filepath.Join(bundlePath, "secure-boot-keys", f)
                        if hash, err := fileChecksum(path); err == nil {
                                checksums["secure-boot-keys/"+f] = hash
                        }
                }
        }

        manifest := &Manifest{
                Version:           ManifestVersion,
                ID:                bundleID,
                Mode:              opts.Mode,
                Hostname:          opts.Hostname,
                CreatedAt:         now,
                EKCertificate:     ekCertInfo,
                EKAlgorithm:       opts.EKAlgorithm,
                EKHandle:          opts.EKAlgorithm.NVIndex(),
                SecureBootEnabled: opts.SecureBootEnabled,
                SecureBootKeys:    sbKeyInfo,
                Checksums:         checksums,
                Labels:            opts.Labels,
        }

        return manifest, nil
}

                return nil, ErrInvalidManifest{Reason: fmt.Sprintf("failed to parse manifest: %v", err)}
        }

        bundle := &Bundle{
                ID:       manifest.ID,
                Mode:     manifest.Mode,
                Manifest: &manifest,
                TPMState: &TPMState{
                        StateDir:    filepath.Join(bundlePath, "tpm"),

                },
                OVMFState: &OVMFState{
                        NVRAMFile: filepath.Join(bundlePath, "ovmf", "OVMF_VARS.fd"),
                },
                CreatedAt:  manifest.CreatedAt,
                ClonedFrom: manifest.ClonedFrom,
        }

        if manifest.SecureBootEnabled {
                sbDir := filepath.Join(bundlePath, "secure-boot-keys")
                bundle.SecureBootKeys = &SecureBootKeyPaths{
                        Dir:     sbDir,
                        PKKey:   filepath.Join(sbDir, "PK.key"),
                        PKCert:  filepath.Join(sbDir, "PK.crt"),

                        DBKey:   filepath.Join(sbDir, "DB.key"),
                        DBCert:  filepath.Join(sbDir, "DB.crt"),
                }
        }

        return bundle, nil
}

// ListBundles lists bundles matching the filter options.
func (s *Service) ListBundles(ctx context.Context, opts *ListOptions) ([]*Bundle, error) {
        s.mu.RLock()
        defer s.mu.RUnlock()

        entries, err := os.ReadDir(s.config.BundlesDir())
        if err != nil {
                if os.IsNotExist(err) {
                        return []*Bundle{}, nil
                }
                return nil, fmt.Errorf("failed to list bundles: %w", err)
        }

        var bundles []*Bundle
        for _, entry := range entries {
                if !entry.IsDir() {
                        continue
                }

                // Use locked version since we already hold the read lock
                bundle, err := s.getBundleLocked(entry.Name())
                if err != nil {
                        s.logger.Warn("Failed to load bundle", "id", entry.Name(), "error", err)
                        continue
                }

                // Apply filters
                if opts != nil {
                        if opts.Mode != "" && bundle.Mode != opts.Mode {
                                continue
                        }
                        if len(opts.Labels) > 0 {
                                match := true
                                for k, v := range opts.Labels {
                                        if bundle.Manifest.Labels[k] != v {
                                                match = false
                                                break
                                        }
                                }
                                if !match {
                                        continue
                                }
                        }
                }

                bundles = append(bundles, bundle)
        }

        // Apply pagination
        if opts != nil {
                if opts.Offset > 0 && opts.Offset < len(bundles) {
                        bundles = bundles[opts.Offset:]
                }
                if opts.Limit > 0 && opts.Limit < len(bundles) {
                        bundles = bundles[:opts.Limit]
                }
        }

        return bundles, nil
}

// CreateTemplate creates a new vTPM template.
func (s *Service) CreateTemplate(ctx context.Context, opts *CreateTemplateOptions) (*Template, error) {
        // Templates are bundles without EK certificate
        bundleOpts := &CreateBundleOptions{
                Mode:              BundleModeTemplate,
                EKAlgorithm:       opts.EKAlgorithm,
                SecureBootEnabled: opts.SecureBootEnabled,
        }

        bundle, err := s.CreateBundle(ctx, bundleOpts)
        if err != nil {
                return nil, err
        }

        // Move from bundles to templates directory
        oldPath := s.config.BundlePath(bundle.ID)
        newPath := s.config.TemplatePath(opts.ID)

        if err := os.Rename(oldPath, newPath); err != nil {
                os.RemoveAll(oldPath)
                return nil, fmt.Errorf("failed to move template: %w", err)
        }

        // Update manifest with template ID
        manifestPath := filepath.Join(newPath, "manifest.json")
        bundle.Manifest.ID = opts.ID
        manifestData, _ := json.MarshalIndent(bundle.Manifest, "", "  ")
        os.WriteFile(manifestPath, manifestData, 0600)

        template := &Template{
                ID:                opts.ID,
                Description:       opts.Description,
                TPMStateDir:       filepath.Join(newPath, "tpm"),
                NVRAMPath:         filepath.Join(newPath, "ovmf", "OVMF_VARS.fd"),
                SecureBootKeysDir: filepath.Join(newPath, "secure-boot-keys"),
                EKAlgorithm:       opts.EKAlgorithm,
                SecureBootEnabled: opts.SecureBootEnabled,
                CreatedAt:         bundle.CreatedAt,
        }

        s.logger.Info("Template created", "id", opts.ID)
        return template, nil
}

// CloneTemplate clones a template to create a new bundle with identity.
func (s *Service) CloneTemplate(ctx context.Context, opts *CloneOptions) (*Bundle, error) {
        s.mu.Lock()
        defer s.mu.Unlock()

        // Get template (use locked version since we already hold the write lock)
        template, err := s.getTemplateLocked(opts.TemplateID)
        if err != nil {
                return nil, err
        }

        // Generate new bundle ID
        bundleID := s.generateBundleID(opts.Hostname)
        bundlePath := s.config.BundlePath(bundleID)

        // Create bundle directories
        if err := s.createBundleDirectories(bundlePath); err != nil {
                return nil, err
        }

        // Copy template files
        templatePath := s.config.TemplatePath(opts.TemplateID)
        if err := s.copyDir(filepath.Join(templatePath, "tpm"), filepath.Join(bundlePath, "tpm")); err != nil {
                os.RemoveAll(bundlePath)
                return nil, fmt.Errorf("failed to copy TPM state: %w", err)
        }
        if err := s.copyDir(filepath.Join(templatePath, "ovmf"), filepath.Join(bundlePath, "ovmf")); err != nil {
                os.RemoveAll(bundlePath)
                return nil, fmt.Errorf("failed to copy OVMF state: %w", err)
        }
        if template.SecureBootEnabled {
                if err := s.copyDir(filepath.Join(templatePath, "secure-boot-keys"), filepath.Join(bundlePath, "secure-boot-keys")); err != nil {
                        os.RemoveAll(bundlePath)
                        return nil, fmt.Errorf("failed to copy Secure Boot keys: %w", err)
                }
        }

        // TODO: Start swtpm, inject EK certificate, stop swtpm
        // For now, create bundle without EK cert injection

        // Generate manifest
        manifest := &Manifest{
                Version:           ManifestVersion,
                ID:                bundleID,
                Mode:              BundleModePreAssigned,
                Hostname:          opts.Hostname,
                CreatedAt:         time.Now(),
                EKAlgorithm:       template.EKAlgorithm,
                EKHandle:          template.EKAlgorithm.NVIndex(),
                SecureBootEnabled: template.SecureBootEnabled,
                Labels:            opts.Labels,
                ClonedFrom:        opts.TemplateID,
        }

        manifestData, _ := json.MarshalIndent(manifest, "", "  ")
        os.WriteFile(filepath.Join(bundlePath, "manifest.json"), manifestData, 0600)

        bundle := &Bundle{
                ID:        bundleID,
                Mode:      BundleModePreAssigned,
                Manifest:  manifest,
                CreatedAt: manifest.CreatedAt,
                ClonedFrom: opts.TemplateID,
                TPMState: &TPMState{
                        StateDir:    filepath.Join(bundlePath, "tpm"),
                        PermallFile: filepath.Join(bundlePath, "tpm", "tpm2-00.permall"),
                },
                OVMFState: &OVMFState{
                        NVRAMFile: filepath.Join(bundlePath, "ovmf", "OVMF_VARS.fd"),
                },
        }

        s.logger.Info("Bundle cloned from template",
                "bundleID", bundleID,
                "templateID", opts.TemplateID)

        return bundle, nil
}

// GetTemplate retrieves a template by ID.
func (s *Service) GetTemplate(ctx context.Context, id string) (*Template, error) {
        s.mu.RLock()
        defer s.mu.RUnlock()

        return s.getTemplateLocked(id)
}

// getTemplateLocked retrieves a template by ID without acquiring a lock.
// Caller must hold at least a read lock.
func (s *Service) getTemplateLocked(id string) (*Template, error) {
        templatePath := s.config.TemplatePath(id)
        manifestPath := filepath.Join(templatePath, "manifest.json")

        if _, err := os.Stat(manifestPath); err != nil {
                return nil, ErrTemplateNotFound{ID: id}
        }

        data, err := os.ReadFile(manifestPath)
        if err != nil {
                return nil, ErrInvalidManifest{Reason: fmt.Sprintf("failed to read manifest: %v", err)}
        }

        var manifest Manifest
        if err := json.Unmarshal(data, &manifest); err != nil {
                return nil, ErrInvalidManifest{Reason: fmt.Sprintf("failed to parse manifest: %v", err)}
        }

        template := &Template{
                ID:                manifest.ID,
                TPMStateDir:       filepath.Join(templatePath, "tpm"),
                NVRAMPath:         filepath.Join(templatePath, "ovmf", "OVMF_VARS.fd"),
                SecureBootKeysDir: filepath.Join(templatePath, "secure-boot-keys"),
                EKAlgorithm:       manifest.EKAlgorithm,
                SecureBootEnabled: manifest.SecureBootEnabled,
                CreatedAt:         manifest.CreatedAt,
        }

        return template, nil
}

// DeleteTemplate deletes a template.
func (s *Service) DeleteTemplate(ctx context.Context, id string) error {
        s.mu.Lock()
        defer s.mu.Unlock()

        templatePath := s.config.TemplatePath(id)

        if _, err := os.Stat(templatePath); err != nil {
                return ErrTemplateNotFound{ID: id}
        }

        if err := os.RemoveAll(templatePath); err != nil {
                return fmt.Errorf("failed to delete template: %w", err)
        }

        s.logger.Info("Template deleted", "id", id)
        return nil
}

// ListTemplates lists all templates.
func (s *Service) ListTemplates(ctx context.Context) ([]*Template, error) {
        s.mu.RLock()
        defer s.mu.RUnlock()

        entries, err := os.ReadDir(s.config.TemplatesDir())
        if err != nil {
                if os.IsNotExist(err) {
                        return []*Template{}, nil
                }
                return nil, fmt.Errorf("failed to list templates: %w", err)
        }

        var templates []*Template
        for _, entry := range entries {
                if !entry.IsDir() {
                        continue
                }

                // Use locked version since we already hold the read lock
                template, err := s.getTemplateLocked(entry.Name())
                if err != nil {
                        s.logger.Warn("Failed to load template", "id", entry.Name(), "error", err)
                        continue
                }

                templates = append(templates, template)
        }

        return templates, nil
}

// ValidateBundle validates bundle integrity.
func (s *Service) ValidateBundle(ctx context.Context, id string) error {
        bundle, err := s.GetBundle(ctx, id)
        if err != nil {
                return err
        }

        bundlePath := s.config.BundlePath(id)

        // Verify checksums
        for relPath, expectedHash := range bundle.Manifest.Checksums {
                fullPath := filepath.Join(bundlePath, relPath)
                actualHash, err := fileChecksum(fullPath)
                if err != nil {
                        return ErrChecksumMismatch{File: relPath, Expected: expectedHash, Actual: "file not found"}
                }
                if actualHash != expectedHash {
                        return ErrChecksumMismatch{File: relPath, Expected: expectedHash, Actual: actualHash}
                }
        }

        s.logger.Debug("Bundle validation passed", "id", id)
        return nil
}

// Helper functions

func (s *Service) copyFile(src, dst string) error {
        in, err := os.Open(src)
        if err != nil {
                return err
        }
        defer in.Close()

        out, err := os.Create(dst)
        if err != nil {
                return err
        }
        defer out.Close()

        if _, err := io.Copy(out, in); err != nil {
                return err
        }

        return out.Chmod(0600)
}

func (s *Service) copyDir(src, dst string) error {
        entries, err := os.ReadDir(src)
        if err != nil {
                return err
        }

        for _, entry := range entries {
                srcPath := filepath.Join(src, entry.Name())
                dstPath := filepath.Join(dst, entry.Name())

                if entry.IsDir() {
                        if err := os.MkdirAll(dstPath, 0700); err != nil {
                                return err
                        }
                        if err := s.copyDir(srcPath, dstPath); err != nil {
                                return err
                        }
                } else {
                        if err := s.copyFile(srcPath, dstPath); err != nil {
                                return err
                        }
                }
        }

        return nil
}

func fileChecksum(path string) (string, error) {
        data, err := os.ReadFile(path)
        if err != nil {
                return "", err
        }
        hash := sha256.Sum256(data)
        return "sha256:" + hex.EncodeToString(hash[:]), nil
}

func certFingerprint(cert *x509.Certificate) string {
        hash := sha256.Sum256(cert.Raw)
        return "sha256:" + hex.EncodeToString(hash[:])
}

// BundleMode represents the bundle creation mode.
type BundleMode string

const (
        // BundleModeTemplate creates a template without identity (EK cert issued at clone time).
        BundleModeTemplate BundleMode = "template"

        // BundleModePreAssigned creates a complete bundle with identity (EK cert issued at creation time).
        BundleModePreAssigned BundleMode = "pre-assigned"
)

// String returns the string representation of BundleMode.
func (m BundleMode) String() string {
        return string(m)
}

// IsValid checks if the bundle mode is valid.
func (m BundleMode) IsValid() bool {
        switch m {
        case BundleModeTemplate, BundleModePreAssigned:
                return true
        }
        return false
}

// EKAlgorithm represents the EK key algorithm.
type EKAlgorithm string

const (
        // EKAlgorithmRSA uses RSA 2048 for the EK.
        EKAlgorithmRSA EKAlgorithm = "RSA"

        // EKAlgorithmECC uses ECC P-256 for the EK.
        EKAlgorithmECC EKAlgorithm = "ECC"
)

// String returns the string representation of EKAlgorithm.
func (a EKAlgorithm) String() string {
        return string(a)
}

// IsValid checks if the algorithm is valid.
func (a EKAlgorithm) IsValid() bool {
        switch a {
        case EKAlgorithmRSA, EKAlgorithmECC:
                return true
        }
        return false
}

// NVIndex returns the TCG-defined NV index for EK certificates.
func (a EKAlgorithm) NVIndex() uint32 {
        switch a {
        case EKAlgorithmRSA:
                return 0x01C00002 // TCG PC Client Platform spec
        case EKAlgorithmECC:
                return 0x01C0000A // TCG PC Client Platform spec
        default:
                return 0x01C00002
        }
}
