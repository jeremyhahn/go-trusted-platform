// Cookie prefixes for flow identification
const (
        // CookieBaseFlow identifies base flows
        CookieBaseFlow uint64 = 0x1000000000000000

        // CookieIsolationFlow identifies isolation flows
        CookieIsolationFlow uint64 = 0x2000000000000000

        // CookiePortFlow identifies port-specific flows
        CookiePortFlow uint64 = 0x3000000000000000

        // CookieSuspendFlow identifies suspend drop flows
        CookieSuspendFlow uint64 = 0x4000000000000000
)

// Priority levels for flows
const (
        // PriorityDrop is the lowest priority for drop rules
        PriorityDrop uint16 = 0

        // PriorityDefault is the default priority
        PriorityDefault uint16 = 100

        // PriorityBase is the priority for base flows
        PriorityBase uint16 = 1000

        // PriorityVLAN is the priority for VLAN matching flows
        PriorityVLAN uint16 = 2000

        // PriorityIsolation is the priority for isolation flows
        PriorityIsolation uint16 = 3000

        // PriorityPort is the priority for port-specific flows
        PriorityPort uint16 = 4000

        // PriorityHigh is a high priority for important flows
        PriorityHigh uint16 = 50000

        // PriorityMax is the maximum priority
        PriorityMax uint16 = 65535
)

// FlowManager handles OpenFlow rules for tenant isolation
type FlowManager struct {
        ofClient OpenFlowClient
        logger   *slog.Logger
}

// NewFlowManager creates a new flow manager
func NewFlowManager(ofClient OpenFlowClient, logger *slog.Logger) *FlowManager {
        if logger == nil {
                logger = slog.Default()
        }

        return &FlowManager{
                ofClient: ofClient,
                logger:   logger,
        }
}

// InstallBaseFlows installs basic tenant bridge flows
// Table 0: Ingress classification
// Table 1: VLAN validation
// Table 2: Forwarding
// Table 3: Egress processing
func (fm *FlowManager) InstallBaseFlows(ctx context.Context, bridge string, vlanRange VLANRange) error {
        fm.logger.Info("installing base flows",
                "bridge", bridge,
                "vlan_range", fmt.Sprintf("%d-%d", vlanRange.Start, vlanRange.End),
        )

        // Table 0: Ingress classification - forward to VLAN validation
        ingressFlow := &Flow{
                Table:    TableIngress,
                Priority: PriorityBase,
                Match:    FlowMatch{},
                Actions: []Action{
                        {Type: ActionGotoTable, Table: TableVLANValidation},
                },
                Cookie: CookieBaseFlow | uint64(TableIngress),
        }

        if err := fm.ofClient.AddFlow(ctx, bridge, ingressFlow); err != nil {
                return ErrFlowInstallFailed{Bridge: bridge, Err: fmt.Errorf("ingress flow: %w", err)}
        }

        // Table 1: VLAN validation - default drop for invalid VLANs
        vlanDropFlow := &Flow{
                Table:    TableVLANValidation,
                Priority: PriorityDrop,
                Match:    FlowMatch{},
                Actions: []Action{
                        {Type: ActionDrop},
                },
                Cookie: CookieBaseFlow | uint64(TableVLANValidation),
        }

        if err := fm.ofClient.AddFlow(ctx, bridge, vlanDropFlow); err != nil {
                return ErrFlowInstallFailed{Bridge: bridge, Err: fmt.Errorf("vlan drop flow: %w", err)}
        }

        // Table 1: Allow untagged traffic (VLAN 0 or no VLAN) to forwarding
        vlanZero := uint16(0)
        untaggedFlow := &Flow{
                Table:    TableVLANValidation,
                Priority: PriorityVLAN,
                Match: FlowMatch{
                        VLAN: &vlanZero,
                },
                Actions: []Action{
                        {Type: ActionGotoTable, Table: TableForwarding},
                },
                Cookie: CookieBaseFlow | uint64(TableVLANValidation) | 0x0001,
        }

        if err := fm.ofClient.AddFlow(ctx, bridge, untaggedFlow); err != nil {
                return ErrFlowInstallFailed{Bridge: bridge, Err: fmt.Errorf("untagged flow: %w", err)}
        }

        // Table 1: Allow valid VLAN range to forwarding
        for vlan := vlanRange.Start; vlan <= vlanRange.End; vlan++ {
                vlanID := vlan
                vlanFlow := &Flow{
                        Table:    TableVLANValidation,
                        Priority: PriorityVLAN,
                        Match: FlowMatch{
                                VLAN: &vlanID,
                        },
                        Actions: []Action{
                                {Type: ActionGotoTable, Table: TableForwarding},
                        },
                        Cookie: CookieBaseFlow | uint64(TableVLANValidation) | uint64(vlan),
                }

                if err := fm.ofClient.AddFlow(ctx, bridge, vlanFlow); err != nil {
                        return ErrFlowInstallFailed{Bridge: bridge, Err: fmt.Errorf("vlan %d flow: %w", vlan, err)}
                }
        }

        // Table 2: Forwarding - normal L2 forwarding
        forwardFlow := &Flow{
                Table:    TableForwarding,
                Priority: PriorityBase,
                Match:    FlowMatch{},
                Actions: []Action{
                        {Type: ActionNormal},
                },
                Cookie: CookieBaseFlow | uint64(TableForwarding),
        }

        if err := fm.ofClient.AddFlow(ctx, bridge, forwardFlow); err != nil {
                return ErrFlowInstallFailed{Bridge: bridge, Err: fmt.Errorf("forward flow: %w", err)}
        }

        // Table 3: Egress processing - pass through
        egressFlow := &Flow{
                Table:    TableEgress,
                Priority: PriorityBase,
                Match:    FlowMatch{},
                Actions: []Action{
                        {Type: ActionNormal},
                },
                Cookie: CookieBaseFlow | uint64(TableEgress),
        }

        if err := fm.ofClient.AddFlow(ctx, bridge, egressFlow); err != nil {
                return ErrFlowInstallFailed{Bridge: bridge, Err: fmt.Errorf("egress flow: %w", err)}
        }

        fm.logger.Info("base flows installed successfully",
                "bridge", bridge,
                "flow_count", 4+vlanRange.Size(),
        )

        return nil
}

// InstallIsolationFlows installs tenant isolation flows
// - Drop packets with out-of-range VLANs
// - Drop broadcast from other tenants
func (fm *FlowManager) InstallIsolationFlows(ctx context.Context, bridge string, vlanRange VLANRange) error {
        fm.logger.Info("installing isolation flows",
                "bridge", bridge,
                "vlan_range", fmt.Sprintf("%d-%d", vlanRange.Start, vlanRange.End),
        )

        // Drop flows for VLANs outside tenant range (below start)
        if vlanRange.Start > 1 {
                for vlan := uint16(1); vlan < vlanRange.Start; vlan++ {
                        vlanID := vlan
                        dropFlow := &Flow{
                                Table:    TableIngress,
                                Priority: PriorityIsolation,
                                Match: FlowMatch{
                                        VLAN: &vlanID,
                                },
                                Actions: []Action{
                                        {Type: ActionDrop},
                                },
                                Cookie: CookieIsolationFlow | uint64(vlan),
                        }

                        if err := fm.ofClient.AddFlow(ctx, bridge, dropFlow); err != nil {
                                return ErrFlowInstallFailed{Bridge: bridge, Err: fmt.Errorf("isolation drop flow vlan %d: %w", vlan, err)}
                        }
                }
        }

        // Drop flows for VLANs outside tenant range (above end)
        if vlanRange.End < 4094 {
                // Only install a few representative flows to avoid excessive rules
                // In production, this would be handled by the default drop rule
                for vlan := vlanRange.End + 1; vlan <= vlanRange.End+10 && vlan <= 4094; vlan++ {
                        vlanID := vlan
                        dropFlow := &Flow{
                                Table:    TableIngress,
                                Priority: PriorityIsolation,
                                Match: FlowMatch{
                                        VLAN: &vlanID,
                                },
                                Actions: []Action{
                                        {Type: ActionDrop},
                                },
                                Cookie: CookieIsolationFlow | uint64(vlan),
                        }

                        if err := fm.ofClient.AddFlow(ctx, bridge, dropFlow); err != nil {
                                return ErrFlowInstallFailed{Bridge: bridge, Err: fmt.Errorf("isolation drop flow vlan %d: %w", vlan, err)}
                        }
                }
        }

        // Drop broadcast/multicast from outside tenant VLANs
        broadcastMAC := "ff:ff:ff:ff:ff:ff"
        broadcastDropFlow := &Flow{
                Table:    TableIngress,
                Priority: PriorityIsolation + 1,
                Match: FlowMatch{
                        EthDst: &broadcastMAC,
                },
                Actions: []Action{
                        {Type: ActionGotoTable, Table: TableVLANValidation},
                },
                Cookie: CookieIsolationFlow | 0xFFFF,
        }

        if err := fm.ofClient.AddFlow(ctx, bridge, broadcastDropFlow); err != nil {
                return ErrFlowInstallFailed{Bridge: bridge, Err: fmt.Errorf("broadcast flow: %w", err)}
        }

        fm.logger.Info("isolation flows installed successfully",
                "bridge", bridge,
        )

        return nil
}

// InstallPortFlows installs flows for a specific port
func (fm *FlowManager) InstallPortFlows(ctx context.Context, bridge string, port uint32, vlan uint16) error {
        fm.logger.Info("installing port flows",
                "bridge", bridge,
                "port", port,
                "vlan", vlan,
        )

        // Flow to tag untagged packets from this port with the assigned VLAN
        portNum := port
        tagFlow := &Flow{
                Table:    TableIngress,
                Priority: PriorityPort,
                Match: FlowMatch{
                        InPort: &portNum,
                },
                Actions: []Action{
                        {Type: ActionPushVLAN, VLAN: 0x8100},
                        {Type: ActionSetVLAN, VLAN: vlan},
                        {Type: ActionGotoTable, Table: TableVLANValidation},
                },
                Cookie: CookiePortFlow | uint64(port),
        }

        if err := fm.ofClient.AddFlow(ctx, bridge, tagFlow); err != nil {
                return ErrFlowInstallFailed{Bridge: bridge, Err: fmt.Errorf("port %d tag flow: %w", port, err)}
        }

        // Flow to strip VLAN tag when sending to this port
        vlanID := vlan
        stripFlow := &Flow{
                Table:    TableEgress,
                Priority: PriorityPort,
                Match: FlowMatch{
                        VLAN: &vlanID,
                },
                Actions: []Action{
                        {Type: ActionPopVLAN},
                        {Type: ActionOutput, Port: port},
                },
                Cookie: CookiePortFlow | uint64(port) | 0x10000,
        }

        if err := fm.ofClient.AddFlow(ctx, bridge, stripFlow); err != nil {
                return ErrFlowInstallFailed{Bridge: bridge, Err: fmt.Errorf("port %d strip flow: %w", port, err)}
        }

        fm.logger.Info("port flows installed successfully",
                "bridge", bridge,
                "port", port,
                "vlan", vlan,
        )

        return nil
}

// InstallARPFlows installs flows to handle ARP within tenant VLAN range
func (fm *FlowManager) InstallARPFlows(ctx context.Context, bridge string, vlanRange VLANRange) error {
        fm.logger.Info("installing ARP flows",
                "bridge", bridge,
                "vlan_range", fmt.Sprintf("%d-%d", vlanRange.Start, vlanRange.End),
        )

        ethTypeARP := EthTypeARP

        // Allow ARP traffic within tenant VLAN range
        for vlan := vlanRange.Start; vlan <= vlanRange.End; vlan++ {
                vlanID := vlan
                arpFlow := &Flow{
                        Table:    TableIngress,
                        Priority: PriorityHigh,
                        Match: FlowMatch{
                                VLAN:    &vlanID,
                                EthType: &ethTypeARP,
                        },
                        Actions: []Action{
                                {Type: ActionNormal},
                        },
                        Cookie: CookieBaseFlow | uint64(vlan) | 0xA000,
                }

                if err := fm.ofClient.AddFlow(ctx, bridge, arpFlow); err != nil {
                        return ErrFlowInstallFailed{Bridge: bridge, Err: fmt.Errorf("arp flow vlan %d: %w", vlan, err)}
                }
        }

        fm.logger.Info("ARP flows installed successfully",
                "bridge", bridge,
        )

        return nil
}

// InstallDHCPFlows installs flows to handle DHCP traffic
func (fm *FlowManager) InstallDHCPFlows(ctx context.Context, bridge string, vlanRange VLANRange) error {
        fm.logger.Info("installing DHCP flows",
                "bridge", bridge,
                "vlan_range", fmt.Sprintf("%d-%d", vlanRange.Start, vlanRange.End),
        )

        ethTypeIPv4 := EthTypeIPv4
        ipProtoUDP := IPProtoUDP
        dhcpClientPort := uint16(68)
        dhcpServerPort := uint16(67)

        // Allow DHCP client traffic (port 68)
        for vlan := vlanRange.Start; vlan <= vlanRange.End; vlan++ {
                vlanID := vlan

                // DHCP client (bootpc) traffic
                dhcpClientFlow := &Flow{
                        Table:    TableIngress,
                        Priority: PriorityHigh,
                        Match: FlowMatch{
                                VLAN:    &vlanID,
                                EthType: &ethTypeIPv4,
                                IPProto: &ipProtoUDP,
                                UDPSrc:  &dhcpClientPort,
                        },
                        Actions: []Action{
                                {Type: ActionNormal},
                        },
                        Cookie: CookieBaseFlow | uint64(vlan) | 0xD0C0,
                }

                if err := fm.ofClient.AddFlow(ctx, bridge, dhcpClientFlow); err != nil {
                        return ErrFlowInstallFailed{Bridge: bridge, Err: fmt.Errorf("dhcp client flow vlan %d: %w", vlan, err)}
                }

                // DHCP server (bootps) traffic
                dhcpServerFlow := &Flow{
                        Table:    TableIngress,
                        Priority: PriorityHigh,
                        Match: FlowMatch{
                                VLAN:    &vlanID,
                                EthType: &ethTypeIPv4,
                                IPProto: &ipProtoUDP,
                                UDPDst:  &dhcpServerPort,
                        },
                        Actions: []Action{
                                {Type: ActionNormal},
                        },
                        Cookie: CookieBaseFlow | uint64(vlan) | 0xD050,
                }

                if err := fm.ofClient.AddFlow(ctx, bridge, dhcpServerFlow); err != nil {
                        return ErrFlowInstallFailed{Bridge: bridge, Err: fmt.Errorf("dhcp server flow vlan %d: %w", vlan, err)}
                }
        }

        fm.logger.Info("DHCP flows installed successfully",
                "bridge", bridge,
        )

        return nil
}

// LifecycleHooks provides callbacks for tenant lifecycle events
type LifecycleHooks struct {
        manager     *Manager
        flowManager *FlowManager
        logger      *slog.Logger
}

// NewLifecycleHooks creates new lifecycle hooks
func NewLifecycleHooks(manager *Manager, flowManager *FlowManager, logger *slog.Logger) *LifecycleHooks {
        if logger == nil {
                logger = slog.Default()
        }

        return &LifecycleHooks{
                manager:     manager,
                flowManager: flowManager,
                logger:      logger,
        }
}

// OnTenantCreate is called when a tenant is created
// Creates the OVS bridge and installs base flows
func (h *LifecycleHooks) OnTenantCreate(ctx context.Context, tenantID, tenantName string, vlanRange VLANRange) error {
        h.logger.Info("handling tenant create event",
                "tenant_id", tenantID,
                "tenant_name", tenantName,
                "vlan_range_start", vlanRange.Start,
                "vlan_range_end", vlanRange.End,
        )

        // Create the bridge
        bridge, err := h.manager.CreateBridge(ctx, tenantID, tenantName, vlanRange)
        if err != nil {
                h.logger.Error("failed to create bridge on tenant create",
                        "tenant_id", tenantID,
                        "error", err,
                )
                return err
        }

        // Install base flows
        if h.flowManager != nil {
                if err := h.flowManager.InstallBaseFlows(ctx, bridge.Name, vlanRange); err != nil {
                        h.logger.Error("failed to install base flows, rolling back bridge",
                                "tenant_id", tenantID,
                                "bridge_name", bridge.Name,
                                "error", err,
                        )

                        // Rollback: delete the bridge
                        if delErr := h.manager.DeleteBridge(ctx, tenantID); delErr != nil {
                                h.logger.Error("failed to rollback bridge creation",
                                        "tenant_id", tenantID,
                                        "error", delErr,
                                )
                        }

                        return err
                }

                // Install isolation flows
                if err := h.flowManager.InstallIsolationFlows(ctx, bridge.Name, vlanRange); err != nil {
                        h.logger.Error("failed to install isolation flows",
                                "tenant_id", tenantID,
                                "bridge_name", bridge.Name,
                                "error", err,
                        )

                        // Rollback: delete the bridge
                        if delErr := h.manager.DeleteBridge(ctx, tenantID); delErr != nil {
                                h.logger.Error("failed to rollback bridge creation",
                                        "tenant_id", tenantID,
                                        "error", delErr,
                                )
                        }

                        return err
                }

                // Install ARP flows
                if err := h.flowManager.InstallARPFlows(ctx, bridge.Name, vlanRange); err != nil {
                        h.logger.Warn("failed to install ARP flows, continuing",
                                "tenant_id", tenantID,
                                "bridge_name", bridge.Name,
                                "error", err,
                        )
                        // ARP flows are not critical, don't rollback
                }

                // Install DHCP flows
                if err := h.flowManager.InstallDHCPFlows(ctx, bridge.Name, vlanRange); err != nil {
                        h.logger.Warn("failed to install DHCP flows, continuing",
                                "tenant_id", tenantID,
                                "bridge_name", bridge.Name,
                                "error", err,
                        )
                        // DHCP flows are not critical, don't rollback
                }
        }

        h.logger.Info("tenant create event handled successfully",
                "tenant_id", tenantID,
                "bridge_name", bridge.Name,
        )

        return nil
}

// OnTenantActivate is called when a tenant becomes active
// Ensures flows are properly installed and bridge is operational
func (h *LifecycleHooks) OnTenantActivate(ctx context.Context, tenantID string) error {
        h.logger.Info("handling tenant activate event",
                "tenant_id", tenantID,
        )

        // Get the bridge
        bridge, err := h.manager.GetBridge(ctx, tenantID)
        if err != nil {
                h.logger.Error("failed to get bridge on tenant activate",
                        "tenant_id", tenantID,
                        "error", err,
                )
                return err
        }

        // Check current status
        if bridge.Status == StatusError {
                h.logger.Warn("bridge is in error state, attempting recovery",
                        "tenant_id", tenantID,
                        "bridge_name", bridge.Name,
                        "error_message", bridge.ErrorMessage,
                )

                // Try to refresh bridge state
                if err := h.manager.Refresh(ctx, tenantID); err != nil {
                        return err
                }
        }

        // Set status to active
        if err := h.manager.SetStatus(ctx, tenantID, StatusActive); err != nil {
                h.logger.Error("failed to set bridge status to active",
                        "tenant_id", tenantID,
                        "error", err,
                )
                return err
        }

        h.logger.Info("tenant activate event handled successfully",
                "tenant_id", tenantID,
                "bridge_name", bridge.Name,
        )

        return nil
}

// OnTenantSuspend is called when a tenant is suspended
// Installs drop flows to block all tenant traffic
func (h *LifecycleHooks) OnTenantSuspend(ctx context.Context, tenantID string) error {
        h.logger.Info("handling tenant suspend event",
                "tenant_id", tenantID,
        )

        // Get the bridge
        bridge, err := h.manager.GetBridge(ctx, tenantID)
        if err != nil {
                h.logger.Error("failed to get bridge on tenant suspend",
                        "tenant_id", tenantID,
                        "error", err,
                )
                return err
        }

        // Install drop-all flow at highest priority
        if h.flowManager != nil {
                dropFlow := &Flow{
                        Table:    TableIngress,
                        Priority: PriorityMax,
                        Match:    FlowMatch{},
                        Actions: []Action{
                                {Type: ActionDrop},
                        },
                        Cookie: CookieSuspendFlow,
                }

                if err := h.flowManager.ofClient.AddFlow(ctx, bridge.Name, dropFlow); err != nil {
                        h.logger.Error("failed to install suspend drop flow",
                                "tenant_id", tenantID,
                                "bridge_name", bridge.Name,
                                "error", err,
                        )
                        return ErrFlowInstallFailed{Bridge: bridge.Name, Err: err}
                }
        }

        // Set status to suspended
        if err := h.manager.SetStatus(ctx, tenantID, StatusSuspended); err != nil {
                h.logger.Error("failed to set bridge status to suspended",
                        "tenant_id", tenantID,
                        "error", err,
                )
                return err
        }

        h.logger.Info("tenant suspend event handled successfully",
                "tenant_id", tenantID,
                "bridge_name", bridge.Name,
        )

        return nil
}

// OnTenantResume is called when a tenant is resumed
// Removes the drop flow installed during suspend
func (h *LifecycleHooks) OnTenantResume(ctx context.Context, tenantID string) error {
        h.logger.Info("handling tenant resume event",
                "tenant_id", tenantID,
        )

        // Get the bridge
        bridge, err := h.manager.GetBridge(ctx, tenantID)
        if err != nil {
                h.logger.Error("failed to get bridge on tenant resume",
                        "tenant_id", tenantID,
                        "error", err,
                )
                return err
        }

        // Remove the suspend drop flow by reinstalling base flows
        // This is safer than trying to delete specific flows
        if h.flowManager != nil {
                // Delete all flows
                if err := h.flowManager.DeleteAllFlows(ctx, bridge.Name); err != nil {
                        h.logger.Error("failed to delete flows on resume",
                                "tenant_id", tenantID,
                                "bridge_name", bridge.Name,
                                "error", err,
                        )
                        return err
                }

                // Reinstall base flows
                if err := h.flowManager.InstallBaseFlows(ctx, bridge.Name, bridge.VLANRange); err != nil {
                        h.logger.Error("failed to reinstall base flows on resume",
                                "tenant_id", tenantID,
                                "bridge_name", bridge.Name,
                                "error", err,
                        )
                        return err
                }

                // Reinstall isolation flows
                if err := h.flowManager.InstallIsolationFlows(ctx, bridge.Name, bridge.VLANRange); err != nil {
                        h.logger.Error("failed to reinstall isolation flows on resume",
                                "tenant_id", tenantID,
                                "bridge_name", bridge.Name,
                                "error", err,
                        )
                        return err
                }

                // Reinstall ARP flows
                if err := h.flowManager.InstallARPFlows(ctx, bridge.Name, bridge.VLANRange); err != nil {
                        h.logger.Warn("failed to reinstall ARP flows on resume",
                                "tenant_id", tenantID,
                                "bridge_name", bridge.Name,
                                "error", err,
                        )
                }

                // Reinstall DHCP flows
                if err := h.flowManager.InstallDHCPFlows(ctx, bridge.Name, bridge.VLANRange); err != nil {
                        h.logger.Warn("failed to reinstall DHCP flows on resume",
                                "tenant_id", tenantID,
                                "bridge_name", bridge.Name,
                                "error", err,
                        )
                }
        }

        // Set status to active
        if err := h.manager.SetStatus(ctx, tenantID, StatusActive); err != nil {
                h.logger.Error("failed to set bridge status to active",
                        "tenant_id", tenantID,
                        "error", err,
                )
                return err
        }

        h.logger.Info("tenant resume event handled successfully",
                "tenant_id", tenantID,
                "bridge_name", bridge.Name,
        )

        return nil
}

// OnTenantDelete is called when a tenant is deleted
// Deletes the OVS bridge and all associated flows
func (h *LifecycleHooks) OnTenantDelete(ctx context.Context, tenantID string) error {
        h.logger.Info("handling tenant delete event",
                "tenant_id", tenantID,
        )

        // Get bridge name before deletion for logging
        bridge, err := h.manager.GetBridge(ctx, tenantID)
        bridgeName := ""
        if err == nil && bridge != nil {
                bridgeName = bridge.Name
        }

        // Delete the bridge (this also deletes flows)
        if err := h.manager.DeleteBridge(ctx, tenantID); err != nil {
                h.logger.Error("failed to delete bridge on tenant delete",
                        "tenant_id", tenantID,
                        "error", err,
                )
                return err
        }

        h.logger.Info("tenant delete event handled successfully",
                "tenant_id", tenantID,
                "bridge_name", bridgeName,
        )

        return nil
}

// OnPortAdd is called when a port is added to a tenant bridge
func (h *LifecycleHooks) OnPortAdd(ctx context.Context, tenantID string, port uint32, vlan uint16) error {
        h.logger.Info("handling port add event",
                "tenant_id", tenantID,
                "port", port,
                "vlan", vlan,
        )

        // Get the bridge
        bridge, err := h.manager.GetBridge(ctx, tenantID)
        if err != nil {
                h.logger.Error("failed to get bridge on port add",
                        "tenant_id", tenantID,
                        "error", err,
                )
                return err
        }

        // Verify VLAN is within tenant range
        if !bridge.VLANRange.Contains(vlan) {
                h.logger.Error("VLAN not in tenant range",
                        "tenant_id", tenantID,
                        "vlan", vlan,
                        "range_start", bridge.VLANRange.Start,
                        "range_end", bridge.VLANRange.End,
                )
                return ErrInvalidVLANRange{
                        Start:  vlan,
                        End:    vlan,
                        Reason: "VLAN not in tenant allocated range",
                }
        }

        // Install port flows
        if h.flowManager != nil {
                if err := h.flowManager.InstallPortFlows(ctx, bridge.Name, port, vlan); err != nil {
                        h.logger.Error("failed to install port flows",
                                "tenant_id", tenantID,
                                "bridge_name", bridge.Name,
                                "port", port,
                                "error", err,
                        )
                        return err
                }
        }

        // Refresh bridge info to update port count
        if err := h.manager.Refresh(ctx, tenantID); err != nil {
                h.logger.Warn("failed to refresh bridge info after port add",
                        "tenant_id", tenantID,
                        "error", err,
                )
        }

        h.logger.Info("port add event handled successfully",
                "tenant_id", tenantID,
                "bridge_name", bridge.Name,
                "port", port,
                "vlan", vlan,
        )

        return nil
}

const (
        // DefaultBridgePrefix is the default prefix for tenant bridges
        DefaultBridgePrefix = "br-tenant-"

        // DefaultDatapath is the default OVS datapath type
        DefaultDatapath = "system"

        // DefaultFailMode is the default bridge fail mode
        DefaultFailMode = "secure"

        // MaxBridgeNameLength is the maximum length for a bridge name
        MaxBridgeNameLength = 15

        // MinTenantNameLength is the minimum length for a tenant name
        MinTenantNameLength = 1

        // MaxTenantNameLength is the maximum length for a tenant name
        MaxTenantNameLength = 64
)

var (
        // validTenantNameRegex matches valid tenant names (alphanumeric, hyphens, underscores)
        validTenantNameRegex = regexp.MustCompile(`^[a-zA-Z][a-zA-Z0-9_-]*$`)
)

// Config holds configuration for the bridge manager
type Config struct {
        // BridgePrefix is the prefix for tenant bridge names
        BridgePrefix string

        // Datapath is the OVS datapath type (system, netdev)
        Datapath string

        // FailMode is the default fail mode for bridges
        FailMode string

        // Protocols are the OpenFlow protocols to enable
        Protocols []string

        // OVSSocket is the path to the OVSDB socket
        OVSSocket string
}

// DefaultConfig returns the default configuration
func DefaultConfig() *Config {
        return &Config{
                BridgePrefix: DefaultBridgePrefix,
                Datapath:     DefaultDatapath,
                FailMode:     DefaultFailMode,
                Protocols:    []string{"OpenFlow13", "OpenFlow15"},
                OVSSocket:    "/var/run/openvswitch/db.sock",
        }
}

// Manager handles per-tenant OVS bridge lifecycle
type Manager struct {
        ovsClient OVSClient
        ofClient  OpenFlowClient
        logger    *slog.Logger
        config    *Config

        // Runtime tracking
        mu      sync.RWMutex
        bridges map[string]*TenantBridge // tenantID -> bridge
}

// NewManager creates a new bridge manager
func NewManager(ovsClient OVSClient, ofClient OpenFlowClient, cfg *Config, logger *slog.Logger) *Manager {
        if cfg == nil {
                cfg = DefaultConfig()
        }

        if cfg.BridgePrefix == "" {
                cfg.BridgePrefix = DefaultBridgePrefix
        }

        if cfg.Datapath == "" {
                cfg.Datapath = DefaultDatapath
        }

        if logger == nil {
                logger = slog.Default()
        }

        return &Manager{
                ovsClient: ovsClient,
                ofClient:  ofClient,
                logger:    logger,
                config:    cfg,
                bridges:   make(map[string]*TenantBridge),
        }
}

// CreateBridge creates a new OVS bridge for a tenant
func (m *Manager) CreateBridge(ctx context.Context, tenantID, tenantName string, vlanRange VLANRange) (*TenantBridge, error) {
        // Validate inputs
        if err := m.validateTenantName(tenantName); err != nil {
                return nil, err
        }

        if !vlanRange.IsValid() {
                return nil, ErrInvalidVLANRange{
                        Start:  vlanRange.Start,
                        End:    vlanRange.End,
                        Reason: "VLAN range must be 1-4094 and start <= end",
                }
        }

        bridgeName := m.bridgeName(tenantName)

        m.logger.Info("creating tenant bridge",
                "tenant_id", tenantID,
                "tenant_name", tenantName,
                "bridge_name", bridgeName,
                "vlan_range", vlanRange,
        )

        // Check for existing bridge
        m.mu.Lock()
        if existing, ok := m.bridges[tenantID]; ok {
                m.mu.Unlock()
                if existing.Status == StatusPending || existing.Status == StatusDeleting {
                        return nil, ErrOperationInProgress{
                                TenantID:  tenantID,
                                Operation: string(existing.Status),
                        }
                }
                return nil, ErrBridgeExists{Name: bridgeName}
        }

        // Create placeholder with pending status
        bridge := &TenantBridge{
                Name:       bridgeName,
                TenantID:   tenantID,
                TenantName: tenantName,
                Datapath:   m.config.Datapath,
                VLANRange:  vlanRange,
                CreatedAt:  time.Now(),
                UpdatedAt:  time.Now(),
                Status:     StatusPending,
        }
        m.bridges[tenantID] = bridge
        m.mu.Unlock()

        // Create the OVS bridge
        if err := m.ovsClient.CreateBridge(ctx, bridgeName); err != nil {
                m.mu.Lock()
                bridge.Status = StatusError
                bridge.ErrorMessage = err.Error()
                bridge.UpdatedAt = time.Now()
                m.mu.Unlock()

                m.logger.Error("failed to create OVS bridge",
                        "bridge_name", bridgeName,
                        "error", err,
                )

                return nil, ErrBridgeCreateFailed{Name: bridgeName, Err: err}
        }

        // Configure bridge properties
        if err := m.configureBridge(ctx, bridgeName); err != nil {
                // Rollback: delete the bridge
                m.logger.Warn("rolling back bridge creation due to configuration failure",
                        "bridge_name", bridgeName,
                        "error", err,
                )

                if delErr := m.ovsClient.DeleteBridge(ctx, bridgeName); delErr != nil {
                        m.logger.Error("failed to rollback bridge creation",
                                "bridge_name", bridgeName,
                                "error", delErr,
                        )
                }

                m.mu.Lock()
                bridge.Status = StatusError
                bridge.ErrorMessage = err.Error()
                bridge.UpdatedAt = time.Now()
                m.mu.Unlock()

                return nil, ErrBridgeCreateFailed{Name: bridgeName, Err: err}
        }

        // Get bridge info from OVS
        info, err := m.ovsClient.GetBridge(ctx, bridgeName)
        if err != nil {
                m.logger.Warn("failed to get bridge info after creation",
                        "bridge_name", bridgeName,
                        "error", err,
                )
        } else if info != nil {
                bridge.UUID = info.UUID
                bridge.PortCount = len(info.Ports)
        }

        // Update status to active
        m.mu.Lock()
        bridge.Status = StatusActive
        bridge.UpdatedAt = time.Now()
        m.mu.Unlock()

        m.logger.Info("tenant bridge created successfully",
                "tenant_id", tenantID,
                "bridge_name", bridgeName,
                "uuid", bridge.UUID,
        )

        return bridge, nil
}

// DeleteBridge removes a tenant's OVS bridge
func (m *Manager) DeleteBridge(ctx context.Context, tenantID string) error {
        m.mu.Lock()
        bridge, ok := m.bridges[tenantID]
        if !ok {
                m.mu.Unlock()
                return ErrTenantNotFound{TenantID: tenantID}
        }

        if bridge.Status == StatusDeleting {
                m.mu.Unlock()
                return ErrOperationInProgress{
                        TenantID:  tenantID,
                        Operation: "delete",
                }
        }

        // Mark as deleting
        bridge.Status = StatusDeleting
        bridge.UpdatedAt = time.Now()
        m.mu.Unlock()

        bridgeName := bridge.Name

        m.logger.Info("deleting tenant bridge",
                "tenant_id", tenantID,
                "bridge_name", bridgeName,
        )

        // Delete all flows first
        if m.ofClient != nil {
                if err := m.ofClient.DeleteAllFlows(ctx, bridgeName); err != nil {
                        m.logger.Warn("failed to delete flows before bridge deletion",
                                "bridge_name", bridgeName,
                                "error", err,
                        )
                }
        }

        // Delete the OVS bridge
        if err := m.ovsClient.DeleteBridge(ctx, bridgeName); err != nil {
                m.mu.Lock()
                bridge.Status = StatusError
                bridge.ErrorMessage = err.Error()
                bridge.UpdatedAt = time.Now()
                m.mu.Unlock()

                m.logger.Error("failed to delete OVS bridge",
                        "bridge_name", bridgeName,
                        "error", err,
                )

                return ErrBridgeDeleteFailed{Name: bridgeName, Err: err}
        }

        // Remove from tracking
        m.mu.Lock()
        delete(m.bridges, tenantID)
        m.mu.Unlock()

        m.logger.Info("tenant bridge deleted successfully",
                "tenant_id", tenantID,
                "bridge_name", bridgeName,
        )

        return nil
}

// GetBridge returns a tenant's bridge info
func (m *Manager) GetBridge(ctx context.Context, tenantID string) (*TenantBridge, error) {
        m.mu.RLock()
        bridge, ok := m.bridges[tenantID]
        m.mu.RUnlock()

        if !ok {
                return nil, ErrTenantNotFound{TenantID: tenantID}
        }

        // Return a copy to prevent external modification
        return m.copyBridge(bridge), nil
}

// GetBridgeByName returns bridge info by bridge name
func (m *Manager) GetBridgeByName(ctx context.Context, name string) (*TenantBridge, error) {
        m.mu.RLock()
        defer m.mu.RUnlock()

        for _, bridge := range m.bridges {
                if bridge.Name == name {
                        return m.copyBridge(bridge), nil
                }
        }

        return nil, ErrBridgeNotFound{Name: name}
}

// ListBridges returns all tenant bridges
func (m *Manager) ListBridges(ctx context.Context) ([]*TenantBridge, error) {
        m.mu.RLock()
        defer m.mu.RUnlock()

        bridges := make([]*TenantBridge, 0, len(m.bridges))
        for _, bridge := range m.bridges {
                bridges = append(bridges, m.copyBridge(bridge))
        }

        return bridges, nil
}

// GetStatus returns bridge operational status
func (m *Manager) GetStatus(ctx context.Context, tenantID string) (BridgeStatus, error) {
        m.mu.RLock()
        bridge, ok := m.bridges[tenantID]
        m.mu.RUnlock()

        if !ok {
                return "", ErrTenantNotFound{TenantID: tenantID}
        }

        return bridge.Status, nil
}

// Refresh updates bridge info from OVS
func (m *Manager) Refresh(ctx context.Context, tenantID string) error {
        m.mu.RLock()
        bridge, ok := m.bridges[tenantID]
        m.mu.RUnlock()

        if !ok {
                return ErrTenantNotFound{TenantID: tenantID}
        }

        m.logger.Debug("refreshing bridge info",
                "tenant_id", tenantID,
                "bridge_name", bridge.Name,
        )

        // Get bridge info from OVS
        info, err := m.ovsClient.GetBridge(ctx, bridge.Name)
        if err != nil {
                m.mu.Lock()
                bridge.Status = StatusError
                bridge.ErrorMessage = err.Error()
                bridge.UpdatedAt = time.Now()
                m.mu.Unlock()
                return err
        }

        // Get flow count
        var flowCount int
        if m.ofClient != nil {
                flows, err := m.ofClient.GetFlows(ctx, bridge.Name)
                if err == nil {
                        flowCount = len(flows)
                }
        }

        // Update bridge info
        m.mu.Lock()
        bridge.UUID = info.UUID
        bridge.PortCount = len(info.Ports)
        bridge.FlowCount = flowCount
        bridge.UpdatedAt = time.Now()

        // Update status based on OVS state
        if bridge.Status == StatusError {
                bridge.Status = StatusActive
                bridge.ErrorMessage = ""
        }
        m.mu.Unlock()

        return nil
}

// SetStatus sets the bridge status
func (m *Manager) SetStatus(ctx context.Context, tenantID string, status BridgeStatus) error {
        m.mu.Lock()
        defer m.mu.Unlock()

        bridge, ok := m.bridges[tenantID]
        if !ok {
                return ErrTenantNotFound{TenantID: tenantID}
        }

        bridge.Status = status
        bridge.UpdatedAt = time.Now()

        m.logger.Info("bridge status updated",
                "tenant_id", tenantID,
                "bridge_name", bridge.Name,
                "status", status,
        )

        return nil
}

// bridgeName generates the bridge name for a tenant
func (m *Manager) bridgeName(tenantName string) string {
        // Normalize tenant name: lowercase, replace spaces with hyphens
        name := strings.ToLower(tenantName)
        name = strings.ReplaceAll(name, " ", "-")
        name = strings.ReplaceAll(name, "_", "-")

        // Construct bridge name
        bridgeName := m.config.BridgePrefix + name

        // Truncate if too long (OVS has a 15 char limit on interface names)
        if len(bridgeName) > MaxBridgeNameLength {
                bridgeName = bridgeName[:MaxBridgeNameLength]
        }

        return bridgeName
}

// validateTenantName validates a tenant name
func (m *Manager) validateTenantName(name string) error {
        if len(name) < MinTenantNameLength {
                return ErrInvalidBridgeName{
                        Name:   name,
                        Reason: "tenant name is too short",
                }
        }

        if len(name) > MaxTenantNameLength {
                return ErrInvalidBridgeName{
                        Name:   name,
                        Reason: "tenant name is too long",
                }
        }

        if !validTenantNameRegex.MatchString(name) {
                return ErrInvalidBridgeName{
                        Name:   name,
                        Reason: "tenant name must start with a letter and contain only alphanumeric characters, hyphens, and underscores",
                }
        }

        return nil
}

// configureBridge configures bridge properties after creation
func (m *Manager) configureBridge(ctx context.Context, bridgeName string) error {
        // Set datapath type if not default
        if m.config.Datapath != "" && m.config.Datapath != "system" {
                if err := m.ovsClient.SetBridgeDatapath(ctx, bridgeName, m.config.Datapath); err != nil {
                        return err
                }
        }

        // Set fail mode
        if m.config.FailMode != "" {
                if err := m.ovsClient.SetBridgeFailMode(ctx, bridgeName, m.config.FailMode); err != nil {
                        return err
                }
        }

        // Set protocols
        if len(m.config.Protocols) > 0 {
                if err := m.ovsClient.SetBridgeProtocols(ctx, bridgeName, m.config.Protocols); err != nil {
                        return err
                }
        }

        return nil
}

// copyBridge creates a copy of a TenantBridge
func (m *Manager) copyBridge(bridge *TenantBridge) *TenantBridge {
        return &TenantBridge{
                Name:         bridge.Name,
                TenantID:     bridge.TenantID,
                TenantName:   bridge.TenantName,
                Datapath:     bridge.Datapath,
                VLANRange:    bridge.VLANRange,
                CreatedAt:    bridge.CreatedAt,
                UpdatedAt:    bridge.UpdatedAt,
                Status:       bridge.Status,
                PortCount:    bridge.PortCount,
                FlowCount:    bridge.FlowCount,
                UUID:         bridge.UUID,
                ErrorMessage: bridge.ErrorMessage,
        }
}

// Sync synchronizes the manager's state with OVS
func (m *Manager) Sync(ctx context.Context) error {
        m.logger.Info("synchronizing bridge state with OVS")

        // Get all bridges from OVS
        bridgeNames, err := m.ovsClient.ListBridges(ctx)
        if err != nil {
                return ErrOVSConnectionFailed{Err: err}
        }

        m.mu.Lock()
        defer m.mu.Unlock()

        // Create a set of bridge names we manage
        managedBridges := make(map[string]bool)
        for _, bridge := range m.bridges {
                managedBridges[bridge.Name] = true
        }

        // Check for bridges that exist in OVS but not in our state
        for _, name := range bridgeNames {
                if strings.HasPrefix(name, m.config.BridgePrefix) {
                        if !managedBridges[name] {
                                m.logger.Warn("found orphaned tenant bridge in OVS",
                                        "bridge_name", name,
                                )
                        }
                }
        }

        // Check for bridges in our state that don't exist in OVS
        ovsBridgeSet := make(map[string]bool)
        for _, name := range bridgeNames {
                ovsBridgeSet[name] = true
        }

        for tenantID, bridge := range m.bridges {
                if !ovsBridgeSet[bridge.Name] {
                        m.logger.Warn("managed bridge not found in OVS, marking as error",
                                "tenant_id", tenantID,
                                "bridge_name", bridge.Name,
                        )
                        bridge.Status = StatusError
                        bridge.ErrorMessage = "bridge not found in OVS"
                        bridge.UpdatedAt = time.Now()
                }
        }

        return nil
}

// BridgeStatus represents the operational status of a tenant bridge
type BridgeStatus string

const (
        // StatusPending indicates the bridge is being created
        StatusPending BridgeStatus = "pending"

        // StatusActive indicates the bridge is operational
        StatusActive BridgeStatus = "active"

        // StatusError indicates the bridge is in an error state
        StatusError BridgeStatus = "error"

        // StatusDeleting indicates the bridge is being deleted
        StatusDeleting BridgeStatus = "deleting"

        // StatusSuspended indicates the bridge is suspended
        StatusSuspended BridgeStatus = "suspended"
)

// VLANRange represents a range of VLANs allocated to a tenant
type VLANRange struct {
        Start uint16
        End   uint16
}

// IsValid checks if the VLAN range is valid
func (v VLANRange) IsValid() bool {
        return v.Start > 0 && v.End > 0 && v.Start <= v.End && v.End <= 4094
}

// Contains checks if a VLAN ID is within the range
func (v VLANRange) Contains(vlan uint16) bool {
        return vlan >= v.Start && vlan <= v.End
}

// Size returns the number of VLANs in the range
func (v VLANRange) Size() int {
        if !v.IsValid() {
                return 0
        }
        return int(v.End - v.Start + 1)
}

// TenantBridge represents a tenant's OVS bridge
type TenantBridge struct {
        // Name is the OVS bridge name (br-tenant-{tenantName})
        Name string

        // TenantID is the unique identifier of the tenant
        TenantID string

        // TenantName is the human-readable tenant name
        TenantName string

        // Datapath specifies the OVS datapath type (system, netdev, etc.)
        Datapath string

        // VLANRange is the VLAN range allocated to this tenant
        VLANRange VLANRange

        // CreatedAt is the timestamp when the bridge was created
        CreatedAt time.Time

        // UpdatedAt is the timestamp when the bridge was last updated

}

// Validate validates the configuration.
func (c *Config) Validate() error {
        if c.DefaultVLANRangeStart < 1 || c.DefaultVLANRangeStart > 4094 {
                return ErrConfigValidation{
                        Field:  "DefaultVLANRangeStart",
                        Reason: "must be between 1 and 4094",
                }
        }

        if c.DefaultVLANRangeSize < 1 {
                return ErrConfigValidation{
                        Field:  "DefaultVLANRangeSize",
                        Reason: "must be at least 1",
                }
        }

        if c.MaxVLANRangeSize < c.DefaultVLANRangeSize {
                return ErrConfigValidation{
                        Field:  "MaxVLANRangeSize",
                        Reason: "must be greater than or equal to DefaultVLANRangeSize",
                }
        }

        // Check if default range would exceed VLAN limits
        endVLAN := int(c.DefaultVLANRangeStart) + int(c.DefaultVLANRangeSize) - 1
        if endVLAN > 4094 {
                return ErrConfigValidation{
                        Field:  "DefaultVLANRangeStart",
                        Reason: "default VLAN range exceeds maximum VLAN ID 4094",
                }
        }

        if c.BridgePrefix == "" {
                return ErrConfigValidation{
                        Field:  "BridgePrefix",
                        Reason: "must not be empty",
                }
        }

        // Validate quota defaults
        if err := c.validateComputeQuota(); err != nil {
                return err
        }

        if err := c.validateNetworkQuota(); err != nil {
                return err
        }

        if err := c.validateStorageQuota(); err != nil {
                return err
        }

        // Validate DEK configuration when enabled
        if c.EnableDEKIsolation {
                if err := c.validateDEKConfig(); err != nil {
                        return err
                }
        }

        return nil
}

// validateDEKConfig validates the DEK isolation configuration.
func (c *Config) validateDEKConfig() error {
        // Validate algorithm
        validAlgorithms := map[string]bool{
                DEKAlgorithmAES128GCM:         true,
                DEKAlgorithmAES192GCM:         true,
                DEKAlgorithmAES256GCM:         true,
                DEKAlgorithmChaCha20Poly1305:  true,
                DEKAlgorithmXChaCha20Poly1305: true,
        }

        if c.DEKAlgorithm != "" && !validAlgorithms[c.DEKAlgorithm] {
                return ErrConfigValidation{
                        Field:  "DEKAlgorithm",
                        Reason: "must be one of: aes-128-gcm, aes-192-gcm, aes-256-gcm, chacha20-poly1305, xchacha20-poly1305",
                }
        }

        // Validate warning percent
        if c.DEKWarningPercent < 0 || c.DEKWarningPercent > 100 {
                return ErrConfigValidation{
                        Field:  "DEKWarningPercent",
                        Reason: "must be between 0 and 100",
                }
        }

        return nil
}

// VLANRange represents a range of VLAN IDs allocated to a tenant.
type VLANRange struct {
        Start uint16
        End   uint16
}

// Contains checks if a VLAN ID is within the range.
func (v VLANRange) Contains(vlan uint16) bool {
        return vlan >= v.Start && vlan <= v.End
}

// Size returns the number of VLANs in the range (inclusive).
func (v VLANRange) Size() int {
        if v.End < v.Start {
                return 0
        }
        return int(v.End-v.Start) + 1
}

// Validate checks if the VLAN range is valid.
func (v VLANRange) Validate() error {
        // VLAN IDs must be between 1 and 4094
        if v.Start < 1 || v.Start > 4094 {
                return ErrInvalidVLANRange{
                        Start:  v.Start,
                        End:    v.End,
                        Reason: "start VLAN must be between 1 and 4094",
                }
        }

        if v.End < 1 || v.End > 4094 {
                return ErrInvalidVLANRange{
                        Start:  v.Start,
                        End:    v.End,
                        Reason: "end VLAN must be between 1 and 4094",
                }
        }

        if v.End < v.Start {
                return ErrInvalidVLANRange{
                        Start:  v.Start,
                        End:    v.End,
                        Reason: "end VLAN must be greater than or equal to start VLAN",
                }
        }

        return nil
}

// IsEmpty checks if the VLAN range is empty (uninitialized).
func (v VLANRange) IsEmpty() bool {
        return v.Start == 0 && v.End == 0
}

// Overlaps checks if this VLAN range overlaps with another range.
func (v VLANRange) Overlaps(other VLANRange) bool {
        return v.Start <= other.End && other.Start <= v.End
}

// FromProto converts an IAM proto Tenant to a TenantEntity.
// Platform-specific fields are initialized with defaults.
func FromProto(t *iampb.Tenant) *TenantEntity {
        if t == nil {
                return nil
        }

        entity := &TenantEntity{
                ID:          t.Id,
                Name:        t.Name,
                DisplayName: t.Name, // IAM doesn't have DisplayName, use Name
                State:       StateCreated,
                Labels:      make(map[string]string),
                Annotations: make(map[string]string),
        }

        // Convert timestamps
        if t.CreatedAt != nil {
                entity.CreatedAt = t.CreatedAt.AsTime()
        }

        if t.UpdatedAt != nil {
                entity.UpdatedAt = t.UpdatedAt.AsTime()
        }

        return entity
}

// ToProto converts a TenantEntity to an IAM proto Tenant.
// Note: Platform-specific fields are not included in the proto.
func (e *TenantEntity) ToProto() *iampb.Tenant {
        if e == nil {
                return nil
        }

        tenant := &iampb.Tenant{
                Id:        e.ID,
                Name:      e.Name,
                CreatedAt: timestamppb.New(e.CreatedAt),
                UpdatedAt: timestamppb.New(e.UpdatedAt),
        }

        return tenant
}

// Clone creates a deep copy of the TenantEntity.
func (e *TenantEntity) Clone() *TenantEntity {
        if e == nil {
                return nil
        }

        clone := &TenantEntity{
                ID:           e.ID,
                Name:         e.Name,
                DisplayName:  e.DisplayName,
                State:        e.State,
                OVSBridge:    e.OVSBridge,
                VLANRange:    e.VLANRange,
                DEKPath:      e.DEKPath,
                ComputeQuota: e.ComputeQuota,
                NetworkQuota: e.NetworkQuota,
                StorageQuota: e.StorageQuota,
                CreatedAt:    e.CreatedAt,
                UpdatedAt:    e.UpdatedAt,
        }

        // Deep copy Labels
        if e.Labels != nil {
                clone.Labels = make(map[string]string, len(e.Labels))
                for k, v := range e.Labels {
                        clone.Labels[k] = v
                }
        }

        // Deep copy Annotations
        if e.Annotations != nil {
                clone.Annotations = make(map[string]string, len(e.Annotations))
                for k, v := range e.Annotations {
                        clone.Annotations[k] = v
                }
        }

        // Deep copy DeletedAt
        if e.DeletedAt != nil {
                deletedAt := *e.DeletedAt
                clone.DeletedAt = &deletedAt
        }

        return clone
}

// Validate validates the TenantEntity fields.
func (e *TenantEntity) Validate() error {
        if e.ID == "" {
                return ErrInvalidRequest{Reason: "tenant ID is required"}
        }

        if e.Name == "" {
                return ErrInvalidRequest{Reason: "tenant name is required"}
        }

        if !e.State.IsValid() {
                return ErrInvalidState{Current: e.State}
        }

        if !e.VLANRange.IsEmpty() {
                if err := e.VLANRange.Validate(); err != nil {
                        return err
                }
        }

        return nil
}

// SetState transitions the tenant to a new state.
// Returns an error if the transition is not allowed.

        if !e.State.CanTransitionTo(newState) {
                return ErrInvalidTransition{
                        From: e.State,
                        To:   newState,
                }
        }

        e.State = newState
        e.UpdatedAt = time.Now()

        if newState == StateDeleted {
                now := time.Now()
                e.DeletedAt = &now
        }

        return nil
}

// DEKHandle extracts a numeric handle from the DEK path for backward compatibility.
// Returns 0 if the path is empty or cannot be parsed.
// The handle is extracted from the path suffix if it contains a numeric component,
// otherwise it returns a hash-based identifier derived from the path.
func (e *TenantEntity) DEKHandle() uint32 {
        if e.DEKPath == "" {
                return 0
        }

        // Try to extract numeric suffix from path (e.g., "/tenant/123/default" -> 123)
        parts := strings.Split(e.DEKPath, "/")
        for i := len(parts) - 1; i >= 0; i-- {
                if parts[i] == "" {
                        continue
                }
                if n, err := strconv.ParseUint(parts[i], 10, 32); err == nil {
                        return uint32(n)
                }
                // If we find a non-numeric part that's not empty, try to find tenant ID
                // Format: /tenant/{tenant-id}/default
                if i > 0 && parts[i-1] == "tenant" {
                        // Return a hash of the tenant ID for backward compat
                        return hashStringToUint32(parts[i])
                }
        }

        // Fallback: hash the entire path
        return hashStringToUint32(e.DEKPath)
}

// hashStringToUint32 creates a simple hash of a string for backward compatibility.
func hashStringToUint32(s string) uint32 {
        var h uint32 = 2166136261
        for i := 0; i < len(s); i++ {
                h ^= uint32(s[i])
                h *= 16777619
        }
        return h
}

// HasDEK returns true if the tenant has a DEK configured.
func (e *TenantEntity) HasDEK() bool {
        return e.DEKPath != ""
}

// BuildDEKPath constructs the standard DEK path for a tenant.
// The path follows the convention: /tenant/{tenant-id}/default
func BuildDEKPath(tenantID string) string {
        return "/tenant/" + tenantID + "/default"
}

                return nil, ErrInvalidRequest{Reason: "DEK client is required when EnableDEKIsolation is true"}

                lifecycleHooks: cfg.LifecycleHooks,
                config:         cfg.Config,
                logger:         cfg.Logger,
        }

        return m, nil

// CreateTenant creates a new tenant with full resource provisioning.
// Steps:
// 1. Create in DragonDB IAM
// 2. Allocate VLAN range
// 3. Create OVS bridge
// 4. Initialize quotas
// 5. Create DEK (if enabled)
func (m *Manager) CreateTenant(ctx context.Context, req *CreateTenantRequest) (*TenantEntity, error) {
        if req == nil {
                return nil, ErrInvalidRequest{Reason: "create request is required"}
        }

        if req.Name == "" {
                return nil, ErrInvalidRequest{Reason: "tenant name is required"}
        }

        m.logger.Info("creating tenant",
                slog.String("name", req.Name),
                slog.String("display_name", req.DisplayName),
        )

        // Check if tenant already exists by name
        existingTenant, err := m.GetTenantByName(ctx, req.Name)
        if err == nil && existingTenant != nil {
                return nil, ErrTenantExists{Name: req.Name}
        }

        // Generate tenant ID
        tenantID := uuid.New().String()

        // Track resources for rollback
        var rollbackActions []func()
        defer func() {
                // Execute rollback actions in reverse order if any remain
                for i := len(rollbackActions) - 1; i >= 0; i-- {
                        rollbackActions[i]()
                }
        }()

        // Step 1: Create tenant in DragonDB IAM
        iamTenant, err := m.createInIAM(ctx, tenantID, req.Name)
        if err != nil {
                m.logger.Error("failed to create tenant in IAM",
                        slog.String("name", req.Name),
                        slog.Any("error", err),
                )
                return nil, ErrStorageFailed{Operation: "CreateTenant", Err: err}
        }

        rollbackActions = append(rollbackActions, func() {
                m.logger.Warn("rolling back IAM tenant creation",
                        slog.String("tenant_id", tenantID),
                )
                if delErr := m.deleteFromIAM(ctx, tenantID); delErr != nil {
                        m.logger.Error("failed to rollback IAM tenant",
                                slog.String("tenant_id", tenantID),
                                slog.Any("error", delErr),
                        )
                }
        })

        // Create entity from IAM response
        entity := FromProto(iamTenant)
        entity.ID = tenantID
        entity.DisplayName = req.DisplayName
        if entity.DisplayName == "" {
                entity.DisplayName = req.Name
        }
        entity.State = StateCreated
        entity.Labels = req.Labels
        entity.Annotations = req.Annotations

        // Step 2: Allocate VLAN range
        if m.vlanAllocator != nil {
                var allocatedRange vlan.Range
                if req.VLANRange != nil {
                        // Use requested range
                        requestedRange := vlan.Range{
                                Start: req.VLANRange.Start,
                                End:   req.VLANRange.End,
                        }
                        allocation, err := m.vlanAllocator.AllocateSpecific(ctx, tenantID, requestedRange)
                        if err != nil {
                                m.logger.Error("failed to allocate specific VLAN range",
                                        slog.String("tenant_id", tenantID),
                                        slog.Any("error", err),
                                )
                                return nil, ErrVLANAllocationFailed{TenantID: tenantID, Err: err}
                        }
                        allocatedRange = allocation.Range
                } else {
                        // Auto-allocate
                        allocation, err := m.vlanAllocator.Allocate(ctx, tenantID, m.config.DefaultVLANRangeSize)
                        if err != nil {
                                m.logger.Error("failed to allocate VLAN range",
                                        slog.String("tenant_id", tenantID),
                                        slog.Any("error", err),
                                )
                                return nil, ErrVLANAllocationFailed{TenantID: tenantID, Err: err}
                        }
                        allocatedRange = allocation.Range
                }

                entity.VLANRange = VLANRange{
                        Start: allocatedRange.Start,
                        End:   allocatedRange.End,
                }

                rollbackActions = append(rollbackActions, func() {
                        m.logger.Warn("rolling back VLAN allocation",
                                slog.String("tenant_id", tenantID),
                        )
                        if relErr := m.vlanAllocator.Release(ctx, tenantID); relErr != nil {
                                m.logger.Error("failed to rollback VLAN allocation",
                                        slog.String("tenant_id", tenantID),
                                        slog.Any("error", relErr),
                                )
                        }
                })
        }

        // Step 3: Create OVS bridge
        if m.lifecycleHooks != nil {
                bridgeVLANRange := bridge.VLANRange{
                        Start: entity.VLANRange.Start,
                        End:   entity.VLANRange.End,
                }

                if err := m.lifecycleHooks.OnTenantCreate(ctx, tenantID, req.Name, bridgeVLANRange); err != nil {
                        m.logger.Error("failed to create OVS bridge",
                                slog.String("tenant_id", tenantID),
                                slog.Any("error", err),
                        )
                        return nil, ErrBridgeOperationFailed{
                                Bridge:    m.config.BridgePrefix + req.Name,
                                Operation: "create",
                                Err:       err,
                        }
                }

                entity.OVSBridge = m.config.BridgePrefix + req.Name

                rollbackActions = append(rollbackActions, func() {
                        m.logger.Warn("rolling back OVS bridge creation",
                                slog.String("tenant_id", tenantID),
                        )
                        if delErr := m.lifecycleHooks.OnTenantDelete(ctx, tenantID); delErr != nil {
                                m.logger.Error("failed to rollback OVS bridge",
                                        slog.String("tenant_id", tenantID),
                                        slog.Any("error", delErr),
                                )
                        }
                })
        }

        // Step 4: Initialize quotas
        entity.ComputeQuota = m.config.DefaultComputeQuota
        entity.NetworkQuota = m.config.DefaultNetworkQuota
        entity.StorageQuota = m.config.DefaultStorageQuota

        if req.ComputeQuota != nil {
                entity.ComputeQuota = *req.ComputeQuota
        }
        if req.NetworkQuota != nil {
                entity.NetworkQuota = *req.NetworkQuota
        }
        if req.StorageQuota != nil {
                entity.StorageQuota = *req.StorageQuota
        }

        // Step 5: Create DEK (if enabled)
        if m.config.EnableDEKIsolation {
                dekPath, err := m.createDEK(ctx, tenantID)
                if err != nil {
                        m.logger.Error("failed to create DEK",
                                slog.String("tenant_id", tenantID),
                                slog.Any("error", err),
                        )
                        return nil, ErrDEKOperationFailed{
                                TenantID:  tenantID,
                                Operation: "create",
                                Err:       err,
                        }
                }
                entity.DEKPath = dekPath

                m.logger.Info("DEK created for tenant",
                        slog.String("tenant_id", tenantID),
                        slog.String("dek_path", dekPath),
                )

                rollbackActions = append(rollbackActions, func() {
                        m.logger.Warn("rolling back DEK creation",
                                slog.String("tenant_id", tenantID),
                        )
                        if delErr := m.deleteDEK(ctx, tenantID, dekPath); delErr != nil {
                                m.logger.Error("failed to rollback DEK",
                                        slog.String("tenant_id", tenantID),
                                        slog.Any("error", delErr),
                                )
                        }
                })
        }

        // Transition to active state
        if err := entity.SetState(StateActive); err != nil {
                return nil, err
        }

        // Activate the bridge
        if m.lifecycleHooks != nil {
                if err := m.lifecycleHooks.OnTenantActivate(ctx, tenantID); err != nil {
                        m.logger.Warn("failed to activate tenant bridge",
                                slog.String("tenant_id", tenantID),
                                slog.Any("error", err),
                        )
                        // Non-fatal, continue
                }
        }

        // Store in cache
        m.cache.Store(tenantID, entity)

        // Clear rollback actions since we succeeded
        rollbackActions = nil

        m.logger.Info("tenant created successfully",
                slog.String("tenant_id", tenantID),
                slog.String("name", req.Name),
                slog.String("bridge", entity.OVSBridge),
                slog.String("vlan_range", fmt.Sprintf("%d-%d", entity.VLANRange.Start, entity.VLANRange.End)),
                slog.String("dek_path", entity.DEKPath),
        )

        return entity.Clone(), nil
}

// GetTenant retrieves a tenant by ID.
func (m *Manager) GetTenant(ctx context.Context, id string) (*TenantEntity, error) {
        if id == "" {
                return nil, ErrInvalidRequest{Reason: "tenant ID is required"}
        }

        // Check cache first
        if cached, ok := m.cache.Load(id); ok {
                entity := cached.(*TenantEntity)
                return entity.Clone(), nil
        }

        // Fetch from IAM
        iamTenant, err := m.getFromIAM(ctx, id)
        if err != nil {
                return nil, ErrTenantNotFound{ID: id}
        }

        entity := FromProto(iamTenant)

        // Enrich with VLAN allocation
        if m.vlanAllocator != nil {
                allocation, err := m.vlanAllocator.Get(ctx, id)
                if err == nil && allocation != nil {
                        entity.VLANRange = VLANRange{
                                Start: allocation.Range.Start,
                                End:   allocation.Range.End,
                        }
                }
        }

        // Enrich with bridge info
        if m.bridgeManager != nil {
                bridgeInfo, err := m.bridgeManager.GetBridge(ctx, id)
                if err == nil && bridgeInfo != nil {
                        entity.OVSBridge = bridgeInfo.Name
                }
        }

        // Enrich with DEK path
        if m.config.EnableDEKIsolation {
                entity.DEKPath = BuildDEKPath(id)
        }

        // Cache and return
        m.cache.Store(id, entity)

        return entity.Clone(), nil
}

// GetTenantByName retrieves a tenant by name.
func (m *Manager) GetTenantByName(ctx context.Context, name string) (*TenantEntity, error) {
        if name == "" {
                return nil, ErrInvalidRequest{Reason: "tenant name is required"}
        }

        // Search cache first
        var found *TenantEntity
        m.cache.Range(func(key, value any) bool {
                entity := value.(*TenantEntity)
                if entity.Name == name {
                        found = entity
                        return false
                }
                return true
        })

        if found != nil {
                return found.Clone(), nil
        }

        // Fetch from IAM
        iamTenant, err := m.getFromIAMByName(ctx, name)
        if err != nil {
                return nil, ErrTenantNotFound{Name: name}
        }

        entity := FromProto(iamTenant)

        // Enrich with platform-specific data
        if m.vlanAllocator != nil {
                allocation, err := m.vlanAllocator.Get(ctx, entity.ID)
                if err == nil && allocation != nil {
                        entity.VLANRange = VLANRange{
                                Start: allocation.Range.Start,
                                End:   allocation.Range.End,
                        }
                }
        }

        if m.bridgeManager != nil {
                bridgeInfo, err := m.bridgeManager.GetBridge(ctx, entity.ID)
                if err == nil && bridgeInfo != nil {
                        entity.OVSBridge = bridgeInfo.Name
                }
        }

        // Enrich with DEK path
        if m.config.EnableDEKIsolation {
                entity.DEKPath = BuildDEKPath(entity.ID)
        }

        // Cache and return
        m.cache.Store(entity.ID, entity)

        return entity.Clone(), nil
}

                                entity.VLANRange = VLANRange{
                                        Start: allocation.Range.Start,
                                        End:   allocation.Range.End,
                                }
                        }
                }

                if m.bridgeManager != nil {
                        bridgeInfo, err := m.bridgeManager.GetBridge(ctx, entity.ID)
                        if err == nil && bridgeInfo != nil {
                                entity.OVSBridge = bridgeInfo.Name
                        }
                }

                // Enrich with DEK path
                if m.config.EnableDEKIsolation {
                        entity.DEKPath = BuildDEKPath(entity.ID)
                }

                // Update cache
                m.cache.Store(entity.ID, entity)
                entities = append(entities, entity.Clone())
        }

        return entities, total, nil
}

// DeleteTenant deletes a tenant and all its resources.
// Steps:
// 1. Set state to deleting
// 2. Delete OVS bridge
// 3. Release VLAN allocation
// 4. Delete DEK
// 5. Delete from DragonDB IAM
func (m *Manager) DeleteTenant(ctx context.Context, id string) error {
        if id == "" {
                return ErrInvalidRequest{Reason: "tenant ID is required"}
        }

        entity, err := m.GetTenant(ctx, id)
        if err != nil {
                return err
        }

        m.logger.Info("deleting tenant",
                slog.String("tenant_id", id),
                slog.String("name", entity.Name),
        )

        // Step 1: Set state to deleting
        if err := entity.SetState(StateDeleting); err != nil {
                return err
        }
        m.cache.Store(id, entity)

        // Step 2: Delete OVS bridge
        if m.lifecycleHooks != nil {
                if err := m.lifecycleHooks.OnTenantDelete(ctx, id); err != nil {
                        m.logger.Error("failed to delete OVS bridge",
                                slog.String("tenant_id", id),
                                slog.Any("error", err),
                        )
                        // Continue with deletion, log the error
                }
        }

        // Step 3: Release VLAN allocation
        if m.vlanAllocator != nil {
                if err := m.vlanAllocator.Release(ctx, id); err != nil {
                        m.logger.Error("failed to release VLAN allocation",
                                slog.String("tenant_id", id),
                                slog.Any("error", err),
                        )
                        // Continue with deletion, log the error
                }
        }

        // Step 4: Delete DEK
        if m.config.EnableDEKIsolation && entity.DEKPath != "" {
                if err := m.deleteDEK(ctx, id, entity.DEKPath); err != nil {
                        m.logger.Error("failed to delete DEK",
                                slog.String("tenant_id", id),
                                slog.String("dek_path", entity.DEKPath),
                                slog.Any("error", err),
                        )
                        // Continue with deletion, log the error
                } else {
                        m.logger.Info("DEK deleted for tenant",
                                slog.String("tenant_id", id),
                                slog.String("dek_path", entity.DEKPath),
                        )
                }
        }

        // Step 5: Delete from DragonDB IAM
        if err := m.deleteFromIAM(ctx, id); err != nil {
                m.logger.Error("failed to delete tenant from IAM",
                        slog.String("tenant_id", id),
                        slog.Any("error", err),
                )
                return ErrStorageFailed{Operation: "DeleteTenant", Err: err}
        }

        // Remove from cache
        m.cache.Delete(id)

        // Invalidate usage tracker cache
        if m.usageTracker != nil {
                m.usageTracker.Invalidate(id)
        }

        m.logger.Info("tenant deleted successfully",
                slog.String("tenant_id", id),
                slog.String("name", entity.Name),
        )

        return nil
}

// SuspendTenant suspends a tenant (blocks traffic).
func (m *Manager) SuspendTenant(ctx context.Context, id string) error {
        if id == "" {
                return ErrInvalidRequest{Reason: "tenant ID is required"}
        }

        entity, err := m.GetTenant(ctx, id)
        if err != nil {
                return err
        }

        m.logger.Info("suspending tenant",
                slog.String("tenant_id", id),
                slog.String("name", entity.Name),
        )

        // Transition state
        if err := entity.SetState(StateSuspended); err != nil {
                return err
        }

        // Suspend bridge traffic
        if m.lifecycleHooks != nil {
                if err := m.lifecycleHooks.OnTenantSuspend(ctx, id); err != nil {
                        m.logger.Error("failed to suspend tenant bridge",
                                slog.String("tenant_id", id),
                                slog.Any("error", err),
                        )
                        return ErrBridgeOperationFailed{
                                Bridge:    entity.OVSBridge,
                                Operation: "suspend",
                                Err:       err,
                        }
                }
        }

        // Update cache
        entity.UpdatedAt = time.Now()
        m.cache.Store(id, entity)

        m.logger.Info("tenant suspended successfully",
                slog.String("tenant_id", id),
        )

        return nil
}

// ResumeTenant resumes a suspended tenant.
func (m *Manager) ResumeTenant(ctx context.Context, id string) error {
        if id == "" {
                return ErrInvalidRequest{Reason: "tenant ID is required"}
        }

        entity, err := m.GetTenant(ctx, id)
        if err != nil {
                return err
        }

        if entity.State != StateSuspended {
                return ErrInvalidTransition{
                        From: entity.State,
                        To:   StateActive,
                }
        }

        m.logger.Info("resuming tenant",
                slog.String("tenant_id", id),
                slog.String("name", entity.Name),
        )

        // Resume bridge traffic
        if m.lifecycleHooks != nil {
                if err := m.lifecycleHooks.OnTenantResume(ctx, id); err != nil {
                        m.logger.Error("failed to resume tenant bridge",
                                slog.String("tenant_id", id),
                                slog.Any("error", err),
                        )
                        return ErrBridgeOperationFailed{
                                Bridge:    entity.OVSBridge,
                                Operation: "resume",
                                Err:       err,
                        }
                }
        }

        // Transition state
        if err := entity.SetState(StateActive); err != nil {
                return err
        }

        // Update cache
        entity.UpdatedAt = time.Now()
        m.cache.Store(id, entity)

        m.logger.Info("tenant resumed successfully",
                slog.String("tenant_id", id),
        )

        return nil
}

// RotateTenantDEKWithOptions rotates the DEK for a tenant with additional options.
// If background is true, the rotation will be performed asynchronously.
// Returns a DEKRotationResult containing the old and new version numbers.
func (m *Manager) RotateTenantDEKWithOptions(ctx context.Context, tenantID string, background bool) (*DEKRotationResult, error) {
        if tenantID == "" {
                return nil, ErrInvalidRequest{Reason: "tenant ID is required"}
        }

        if !m.config.EnableDEKIsolation {
                return nil, ErrInvalidRequest{Reason: "DEK isolation is not enabled"}
        }

        if m.dekClient == nil {
                return nil, ErrInvalidRequest{Reason: "DEK client is not configured"}
        }

        entity, err := m.GetTenant(ctx, tenantID)
        if err != nil {
                return nil, err
        }

        if entity.DEKPath == "" {
                return nil, ErrInvalidRequest{Reason: "tenant has no DEK configured"}
        }

        m.logger.Info("rotating tenant DEK",
                slog.String("tenant_id", tenantID),
                slog.String("dek_path", entity.DEKPath),
                slog.Bool("background", background),
        )

        req := &pb.RotateDEKRequest{
                Path:       entity.DEKPath,
                Background: background,
        }

        resp, err := m.dekClient.RotateDEK(ctx, req)
        if err != nil {
                m.logger.Error("failed to rotate DEK",
                        slog.String("tenant_id", tenantID),
                        slog.String("dek_path", entity.DEKPath),
                        slog.Any("error", err),
                )
                return nil, ErrDEKOperationFailed{
                        TenantID:  tenantID,
                        Operation: "rotate",
                        Err:       err,
                }
        }

        m.logger.Info("DEK rotated successfully",
                slog.String("tenant_id", tenantID),
                slog.String("dek_path", entity.DEKPath),
                slog.Uint64("old_version", resp.OldVersion),
                slog.Uint64("new_version", resp.NewVersion),
        )

        return &DEKRotationResult{
                TenantID:   tenantID,
                OldVersion: uint32(resp.OldVersion),
                NewVersion: uint32(resp.NewVersion),
                Timestamp:  time.Now(),
        }, nil
}

// GetTenantDEKStatus returns the status of a tenant's DEK.
func (m *Manager) GetTenantDEKStatus(ctx context.Context, tenantID string) (*DEKStatus, error) {
        if tenantID == "" {
                return nil, ErrInvalidRequest{Reason: "tenant ID is required"}
        }

        if !m.config.EnableDEKIsolation {
                return nil, ErrInvalidRequest{Reason: "DEK isolation is not enabled"}
        }

        if m.dekClient == nil {
                return nil, ErrInvalidRequest{Reason: "DEK client is not configured"}
        }

        entity, err := m.GetTenant(ctx, tenantID)
        if err != nil {
                return nil, err
        }

        if entity.DEKPath == "" {
                return nil, ErrInvalidRequest{Reason: "tenant has no DEK configured"}
        }

        req := &pb.GetDEKStatusRequest{
                Path: entity.DEKPath,
        }

        resp, err := m.dekClient.GetDEKStatus(ctx, req)
        if err != nil {
                m.logger.Error("failed to get DEK status",
                        slog.String("tenant_id", tenantID),
                        slog.String("dek_path", entity.DEKPath),
                        slog.Any("error", err),
                )
                return nil, ErrDEKOperationFailed{
                        TenantID:  tenantID,
                        Operation: "status",
                        Err:       err,
                }
        }

        status := &DEKStatus{
                Path:           resp.Dek.Path,
                Algorithm:      protoKeyAlgorithmToString(resp.Dek.Algorithm),
                Version:        resp.Dek.Version,
                Status:         resp.Dek.Status,
                IsDefault:      resp.Dek.IsDefault,
                CreatedAt:      time.Unix(resp.Dek.CreatedAt, 0),
                RotatedAt:      time.Unix(resp.Dek.RotatedAt, 0),
                UsageCount:     resp.Dek.UsageCount,
                BytesUsed:      resp.Dek.BytesUsed,
                MaxUsageCount:  resp.Dek.MaxUsageCount,
                MaxBytesUsed:   resp.Dek.MaxBytesUsed,
                WarningPercent: resp.Dek.WarningPercent,
        }

        // Calculate usage percent
        if status.MaxUsageCount > 0 {
                status.UsagePercent = float64(status.UsageCount) / float64(status.MaxUsageCount) * 100
        }

        // Determine if rotation is needed
        if status.MaxUsageCount > 0 && status.UsageCount >= status.MaxUsageCount {
                status.NeedsRotation = true
        }
        if status.MaxBytesUsed > 0 && status.BytesUsed >= status.MaxBytesUsed {
                status.NeedsRotation = true
        }

        // Include re-encryption progress if available
        if resp.ReencryptionProgress != nil {
                status.ReencryptProgress = &ReencryptionProgress{
                        InProgress:          resp.ReencryptionProgress.InProgress,
                        TotalKeys:           resp.ReencryptionProgress.TotalKeys,
                        ProcessedKeys:       resp.ReencryptionProgress.ProcessedKeys,
                        FailedKeys:          resp.ReencryptionProgress.FailedKeys,
                        StartedAt:           time.Unix(resp.ReencryptionProgress.StartedAt, 0),
                        EstimatedCompletion: time.Unix(resp.ReencryptionProgress.EstimatedCompletion, 0),
                }
        }

        return status, nil
}

// Sync synchronizes manager state with external systems.
func (m *Manager) Sync(ctx context.Context) error {
        m.logger.Info("synchronizing manager state")

        // Sync VLAN allocator
        if m.vlanAllocator != nil {
                if err := m.vlanAllocator.Reload(ctx); err != nil {
                        m.logger.Error("failed to sync VLAN allocator",
                                slog.Any("error", err),
                        )
                }
        }

        // Sync bridge manager
        if m.bridgeManager != nil {
                if err := m.bridgeManager.Sync(ctx); err != nil {
                        m.logger.Error("failed to sync bridge manager",
                                slog.Any("error", err),
                        )
                }
        }

        // Sync usage tracker
        if m.usageTracker != nil {
                if err := m.usageTracker.SyncAll(ctx); err != nil {
                        m.logger.Error("failed to sync usage tracker",
                                slog.Any("error", err),
                        )
                }
        }

        // Clean up expired quota reservations
        if m.quotaEnforcer != nil {
                cleaned := m.quotaEnforcer.CleanupExpiredReservations()
                if cleaned > 0 {
                        m.logger.Info("cleaned up expired reservations",
                                slog.Int("count", cleaned),
                        )
                }
        }

        m.logger.Info("manager state synchronized")

        return nil
}

// algorithmStringToProto converts a string algorithm to proto enum.
func (m *Manager) algorithmStringToProto(algorithm string) pb.KeyAlgorithm {
        algorithmMap := map[string]pb.KeyAlgorithm{
                DEKAlgorithmAES128GCM:         pb.KeyAlgorithm_KEY_ALG_AES_128_GCM,
                DEKAlgorithmAES192GCM:         pb.KeyAlgorithm_KEY_ALG_AES_192_GCM,
                DEKAlgorithmAES256GCM:         pb.KeyAlgorithm_KEY_ALG_AES_256_GCM,
                DEKAlgorithmChaCha20Poly1305:  pb.KeyAlgorithm_KEY_ALG_CHACHA20_POLY1305,
                DEKAlgorithmXChaCha20Poly1305: pb.KeyAlgorithm_KEY_ALG_XCHACHA20_POLY1305,
        }

        if alg, ok := algorithmMap[algorithm]; ok {
                return alg
        }
        return pb.KeyAlgorithm_KEY_ALG_AES_256_GCM // Default
}

// protoKeyAlgorithmToString converts a proto enum to string.

                pb.KeyAlgorithm_KEY_ALG_AES_192_GCM:         DEKAlgorithmAES192GCM,

        }

        if str, ok := algorithmMap[algorithm]; ok {
                return str
        }
        return "unknown"
}

// Enforcer validates and reserves quota before resource creation.
type Enforcer struct {
        store  QuotaStore
        logger *slog.Logger

        // pending holds reservations that have not yet been committed.
        // Key is reservation ID, value is *Reservation.
        pending sync.Map

        // reservationTTL is the time-to-live for reservations.
        reservationTTL time.Duration
}

// EnforcerOption is a functional option for configuring the Enforcer.
type EnforcerOption func(*Enforcer)

// WithReservationTTL sets the reservation time-to-live.
func WithReservationTTL(ttl time.Duration) EnforcerOption {
        return func(e *Enforcer) {
                e.reservationTTL = ttl
        }
}

// NewEnforcer creates a new quota enforcer.
func NewEnforcer(store QuotaStore, logger *slog.Logger, opts ...EnforcerOption) (*Enforcer, error) {
        if store == nil {
                return nil, ErrNilStore{}
        }
        if logger == nil {
                return nil, ErrNilLogger{}
        }

        e := &Enforcer{
                store:          store,
                logger:         logger,
                reservationTTL: DefaultReservationTTL,
        }

        for _, opt := range opts {
                opt(e)
        }

        return e, nil
}

// getReservedAmount returns the current reserved amount for a tenant resource.
func (e *Enforcer) getReservedAmount(tenantID string, resource ResourceType) int64 {
        var total int64

        e.pending.Range(func(_, value any) bool {
                res := value.(*Reservation)
                if res.TenantID == tenantID && res.Resource == resource && !res.Committed && !res.IsExpired() {
                        total += res.Amount
                }
                return true
        })

        return total
}

// CheckQuota verifies if the request can be fulfilled.
func (e *Enforcer) CheckQuota(ctx context.Context, tenantID string, resource ResourceType, amount int64) error {
        if !resource.IsValid() {
                return ErrInvalidResource{Resource: resource}
        }
        if amount < 0 {
                return ErrInvalidAmount{Resource: resource, Amount: amount}
        }

        quotas, err := e.store.GetQuota(ctx, tenantID)
        if err != nil {
                return ErrStoreFailed{Operation: "GetQuota", Err: err}
        }
        if quotas == nil {
                return ErrTenantNotFound{TenantID: tenantID}
        }

        usage, err := e.store.GetUsage(ctx, tenantID)
        if err != nil {
                return ErrStoreFailed{Operation: "GetUsage", Err: err}
        }

        limit := quotas.GetLimit(resource)
        currentUsage := int64(0)
        if usage != nil {
                currentUsage = usage.GetUsage(resource)
        }
        reserved := e.getReservedAmount(tenantID, resource)

        available := limit - currentUsage - reserved
        if amount > available {
                return ErrQuotaExceeded{
                        TenantID:  tenantID,
                        Resource:  resource,
                        Requested: amount,
                        Available: available,
                }
        }

        return nil
}

// Reserve creates a temporary quota reservation.
func (e *Enforcer) Reserve(ctx context.Context, tenantID string, resource ResourceType, amount int64) (*Reservation, error) {
        if !resource.IsValid() {
                return nil, ErrInvalidResource{Resource: resource}
        }
        if amount <= 0 {
                return nil, ErrInvalidAmount{Resource: resource, Amount: amount}
        }

        // Check if quota is available before reserving.
        if err := e.CheckQuota(ctx, tenantID, resource, amount); err != nil {
                return nil, err
        }

        now := time.Now()
        reservation := &Reservation{
                ID:        uuid.New().String(),
                TenantID:  tenantID,
                Resource:  resource,
                Amount:    amount,
                CreatedAt: now,
                ExpiresAt: now.Add(e.reservationTTL),
                Committed: false,
        }

        e.pending.Store(reservation.ID, reservation)

        e.logger.Debug("quota reservation created",
                slog.String("reservation_id", reservation.ID),
                slog.String("tenant_id", tenantID),
                slog.String("resource", string(resource)),
                slog.Int64("amount", amount),
        )

        return reservation, nil
}

// Commit finalizes a reservation, updating actual usage.
func (e *Enforcer) Commit(ctx context.Context, reservationID string) error {
        value, ok := e.pending.Load(reservationID)
        if !ok {
                return ErrReservationNotFound{ID: reservationID}
        }

        reservation := value.(*Reservation)

        if reservation.Committed {
                return ErrReservationAlreadyCommitted{ID: reservationID}
        }

        if reservation.IsExpired() {
                e.pending.Delete(reservationID)
                return ErrReservationExpired{ID: reservationID}
        }

        // Get current usage.
        usage, err := e.store.GetUsage(ctx, reservation.TenantID)
        if err != nil {
                return ErrStoreFailed{Operation: "GetUsage", Err: err}
        }

        if usage == nil {
                usage = &TenantUsage{
                        TenantID:  reservation.TenantID,
                        Timestamp: time.Now(),
                }
        }

        // Update usage.
        currentUsage := usage.GetUsage(reservation.Resource)
        usage.SetUsage(reservation.Resource, currentUsage+reservation.Amount)
        usage.Timestamp = time.Now()

        if err := e.store.UpdateUsage(ctx, reservation.TenantID, usage); err != nil {
                return ErrStoreFailed{Operation: "UpdateUsage", Err: err}
        }

        // Mark reservation as committed and remove from pending.
        reservation.Committed = true
        e.pending.Delete(reservationID)

        e.logger.Debug("quota reservation committed",
                slog.String("reservation_id", reservationID),
                slog.String("tenant_id", reservation.TenantID),
                slog.String("resource", string(reservation.Resource)),
                slog.Int64("amount", reservation.Amount),
        )

        return nil
}

// Release cancels a reservation without updating usage.
func (e *Enforcer) Release(ctx context.Context, reservationID string) error {
        value, ok := e.pending.Load(reservationID)
        if !ok {
                return ErrReservationNotFound{ID: reservationID}
        }

        reservation := value.(*Reservation)

        if reservation.Committed {
                return ErrReservationAlreadyCommitted{ID: reservationID}
        }

        e.pending.Delete(reservationID)

        e.logger.Debug("quota reservation released",
                slog.String("reservation_id", reservationID),
                slog.String("tenant_id", reservation.TenantID),
                slog.String("resource", string(reservation.Resource)),
                slog.Int64("amount", reservation.Amount),
        )

        return nil
}

// ReleaseUsage directly releases used quota (for resource deletion).
func (e *Enforcer) ReleaseUsage(ctx context.Context, tenantID string, resource ResourceType, amount int64) error {
        if !resource.IsValid() {
                return ErrInvalidResource{Resource: resource}
        }
        if amount <= 0 {
                return ErrInvalidAmount{Resource: resource, Amount: amount}
        }

        usage, err := e.store.GetUsage(ctx, tenantID)
        if err != nil {
                return ErrStoreFailed{Operation: "GetUsage", Err: err}
        }

        if usage == nil {
                return ErrTenantNotFound{TenantID: tenantID}
        }

        currentUsage := usage.GetUsage(resource)
        newUsage := currentUsage - amount
        if newUsage < 0 {
                newUsage = 0
        }

        usage.Timestamp = time.Now()

        if err := e.store.UpdateUsage(ctx, tenantID, usage); err != nil {
                return ErrStoreFailed{Operation: "UpdateUsage", Err: err}
        }

        e.logger.Debug("quota usage released",
                slog.String("tenant_id", tenantID),
                slog.String("resource", string(resource)),
                slog.Int64("amount", amount),
                slog.Int64("new_usage", newUsage),
        )

        return nil
}

// GetUtilization returns quota utilization percentage (0.0 to 1.0).
func (e *Enforcer) GetUtilization(ctx context.Context, tenantID string, resource ResourceType) (float64, error) {
        if !resource.IsValid() {
                return 0, ErrInvalidResource{Resource: resource}
        }

        quotas, err := e.store.GetQuota(ctx, tenantID)
        if err != nil {
                return 0, ErrStoreFailed{Operation: "GetQuota", Err: err}
        }
        if quotas == nil {
                return 0, ErrTenantNotFound{TenantID: tenantID}
        }

        usage, err := e.store.GetUsage(ctx, tenantID)
        if err != nil {
                return 0, ErrStoreFailed{Operation: "GetUsage", Err: err}
        }

        limit := quotas.GetLimit(resource)
        if limit == 0 {
                return 0, nil
        }

        currentUsage := int64(0)
        if usage != nil {
                currentUsage = usage.GetUsage(resource)
        }
        reserved := e.getReservedAmount(tenantID, resource)

        utilization := float64(currentUsage+reserved) / float64(limit)
        if utilization > 1.0 {
                utilization = 1.0
        }

        return utilization, nil
}

// CleanupExpiredReservations removes expired reservations.
func (e *Enforcer) CleanupExpiredReservations() int {
        count := 0

        e.pending.Range(func(key, value any) bool {
                reservation := value.(*Reservation)
                if reservation.IsExpired() && !reservation.Committed {
                        e.pending.Delete(key)
                        count++

                        e.logger.Debug("expired reservation cleaned up",
                                slog.String("reservation_id", reservation.ID),
                                slog.String("tenant_id", reservation.TenantID),
                        )
                }
                return true
        })

        return count
}

// GetReservation returns a reservation by ID.
func (e *Enforcer) GetReservation(reservationID string) (*Reservation, error) {
        value, ok := e.pending.Load(reservationID)
        if !ok {
                return nil, ErrReservationNotFound{ID: reservationID}
        }

        return value.(*Reservation), nil
}

// GetPendingReservations returns all pending reservations for a tenant.
func (e *Enforcer) GetPendingReservations(tenantID string) []*Reservation {
        var reservations []*Reservation

        e.pending.Range(func(_, value any) bool {
                reservation := value.(*Reservation)
                if reservation.TenantID == tenantID && !reservation.Committed && !reservation.IsExpired() {
                        reservations = append(reservations, reservation)
                }
                return true
        })

        return reservations
}

// Tracker monitors real-time resource usage with atomic operations.
type Tracker struct {
        store  QuotaStore
        logger *slog.Logger

        // cache holds in-memory usage per tenant.
        // Key is tenantID, value is *usageCache.
        cache sync.Map

        // syncInterval is the interval at which to sync usage to storage.
        syncInterval time.Duration
}

// usageCache holds atomic counters for tenant resource usage.
type usageCache struct {
        vms           atomic.Int64
        containers    atomic.Int64
        vcpus         atomic.Int64
        memoryMB      atomic.Int64
        diskGB        atomic.Int64
        bridges       atomic.Int64
        ports         atomic.Int64
        bandwidthMbps atomic.Int64
        volumes       atomic.Int64
        snapshots     atomic.Int64

        lastSync atomic.Value // time.Time
}

// newUsageCache creates a new usage cache initialized from existing usage.
func newUsageCache(usage *TenantUsage) *usageCache {
        c := &usageCache{}
        c.lastSync.Store(time.Now())

        if usage != nil {
                c.vms.Store(usage.VMs)
                c.containers.Store(usage.Containers)
                c.vcpus.Store(usage.VCPUs)
                c.memoryMB.Store(usage.MemoryMB)
                c.diskGB.Store(usage.DiskGB)
                c.bridges.Store(usage.Bridges)
                c.ports.Store(usage.Ports)
                c.bandwidthMbps.Store(usage.BandwidthMbps)
                c.volumes.Store(usage.Volumes)
                c.snapshots.Store(usage.Snapshots)
        }

        return c
}

// get returns the counter for a specific resource type.
func (c *usageCache) get(resource ResourceType) int64 {
        switch resource {
        case ResourceVMs:
                return c.vms.Load()
        case ResourceContainers:
                return c.containers.Load()
        case ResourceVCPUs:
                return c.vcpus.Load()
        case ResourceMemory:
                return c.memoryMB.Load()
        case ResourceDisk:
                return c.diskGB.Load()
        case ResourceBridges:
                return c.bridges.Load()
        case ResourcePorts:
                return c.ports.Load()
        case ResourceBandwidth:
                return c.bandwidthMbps.Load()
        case ResourceVolumes:
                return c.volumes.Load()
        case ResourceSnapshots:
                return c.snapshots.Load()
        default:
                return 0
        }
}

// add atomically adds to a resource counter, returning the new value.
func (c *usageCache) add(resource ResourceType, delta int64) int64 {
        switch resource {
        case ResourceVMs:
                return c.vms.Add(delta)
        case ResourceContainers:
                return c.containers.Add(delta)
        case ResourceVCPUs:
                return c.vcpus.Add(delta)
        case ResourceMemory:
                return c.memoryMB.Add(delta)
        case ResourceDisk:
                return c.diskGB.Add(delta)
        case ResourceBridges:
                return c.bridges.Add(delta)
        case ResourcePorts:
                return c.ports.Add(delta)
        case ResourceBandwidth:
                return c.bandwidthMbps.Add(delta)
        case ResourceVolumes:
                return c.volumes.Add(delta)
        case ResourceSnapshots:
                return c.snapshots.Add(delta)
        default:
                return 0
        }
}

// set atomically sets a resource counter value.
func (c *usageCache) set(resource ResourceType, value int64) {
        switch resource {
        case ResourceVMs:
                c.vms.Store(value)
        case ResourceContainers:
                c.containers.Store(value)
        case ResourceVCPUs:
                c.vcpus.Store(value)
        case ResourceMemory:
                c.memoryMB.Store(value)
        case ResourceDisk:
                c.diskGB.Store(value)
        case ResourceBridges:
                c.bridges.Store(value)
        case ResourcePorts:
                c.ports.Store(value)
        case ResourceBandwidth:
                c.bandwidthMbps.Store(value)
        case ResourceVolumes:
                c.volumes.Store(value)
        case ResourceSnapshots:
                c.snapshots.Store(value)
        }
}

// toTenantUsage converts the cache to a TenantUsage struct.
func (c *usageCache) toTenantUsage(tenantID string) *TenantUsage {
        lastSync, _ := c.lastSync.Load().(time.Time)
        return &TenantUsage{
                TenantID:      tenantID,
                VMs:           c.vms.Load(),
                Containers:    c.containers.Load(),
                VCPUs:         c.vcpus.Load(),
                MemoryMB:      c.memoryMB.Load(),
                DiskGB:        c.diskGB.Load(),
                Bridges:       c.bridges.Load(),
                Ports:         c.ports.Load(),
                BandwidthMbps: c.bandwidthMbps.Load(),
                Volumes:       c.volumes.Load(),
                Snapshots:     c.snapshots.Load(),
                Timestamp:     lastSync,
        }
}

// TrackerOption is a functional option for configuring the Tracker.
type TrackerOption func(*Tracker)

// WithSyncInterval sets the sync interval for the tracker.
func WithSyncInterval(interval time.Duration) TrackerOption {
        return func(t *Tracker) {
                t.syncInterval = interval
        }
}

// NewTracker creates a new usage tracker.
func NewTracker(store QuotaStore, logger *slog.Logger, opts ...TrackerOption) (*Tracker, error) {
        if store == nil {
                return nil, ErrNilStore{}
        }
        if logger == nil {
                return nil, ErrNilLogger{}
        }

        t := &Tracker{
                store:        store,
                logger:       logger,
                syncInterval: time.Minute,
        }

        for _, opt := range opts {
                opt(t)
        }

        return t, nil
}

// getOrCreateCache gets or creates a usage cache for a tenant.
func (t *Tracker) getOrCreateCache(ctx context.Context, tenantID string) (*usageCache, error) {
        if value, ok := t.cache.Load(tenantID); ok {
                return value.(*usageCache), nil
        }

        // Load from store.
        usage, err := t.store.GetUsage(ctx, tenantID)
        if err != nil {
                return nil, ErrStoreFailed{Operation: "GetUsage", Err: err}
        }

        cache := newUsageCache(usage)
        actual, _ := t.cache.LoadOrStore(tenantID, cache)
        return actual.(*usageCache), nil
}

// Increment adds to resource usage.
func (t *Tracker) Increment(ctx context.Context, tenantID string, resource ResourceType, amount int64) error {
        if !resource.IsValid() {
                return ErrInvalidResource{Resource: resource}
        }
        if amount < 0 {
                return ErrInvalidAmount{Resource: resource, Amount: amount}
        }

        cache, err := t.getOrCreateCache(ctx, tenantID)
        if err != nil {
                return err
        }

        newValue := cache.add(resource, amount)

        t.logger.Debug("usage incremented",
                slog.String("tenant_id", tenantID),
                slog.String("resource", string(resource)),
                slog.Int64("amount", amount),
                slog.Int64("new_value", newValue),
        )

        return nil
}

// Decrement subtracts from resource usage.
func (t *Tracker) Decrement(ctx context.Context, tenantID string, resource ResourceType, amount int64) error {
        if !resource.IsValid() {
                return ErrInvalidResource{Resource: resource}
        }
        if amount < 0 {
                return ErrInvalidAmount{Resource: resource, Amount: amount}
        }

        cache, err := t.getOrCreateCache(ctx, tenantID)
        if err != nil {
                return err
        }

        // Ensure we don't go below zero.
        for {
                current := cache.get(resource)
                newValue := current - amount
                if newValue < 0 {
                        newValue = 0
                }

                switch resource {
                case ResourceVMs:
                        if cache.vms.CompareAndSwap(current, newValue) {
                                t.logger.Debug("usage decremented",
                                        slog.String("tenant_id", tenantID),
                                        slog.String("resource", string(resource)),
                                        slog.Int64("amount", amount),
                                        slog.Int64("new_value", newValue),
                                )
                                return nil
                        }
                case ResourceContainers:
                        if cache.containers.CompareAndSwap(current, newValue) {
                                t.logger.Debug("usage decremented",
                                        slog.String("tenant_id", tenantID),
                                        slog.String("resource", string(resource)),
                                        slog.Int64("amount", amount),
                                        slog.Int64("new_value", newValue),
                                )
                                return nil
                        }

                                        slog.String("tenant_id", tenantID),

                                        slog.Int64("amount", amount),
                                        slog.Int64("new_value", newValue),
                                )
                                return nil
                        }
                case ResourceMemory:
                        if cache.memoryMB.CompareAndSwap(current, newValue) {
                                t.logger.Debug("usage decremented",
                                        slog.String("tenant_id", tenantID),
                                        slog.String("resource", string(resource)),
                                        slog.Int64("amount", amount),
                                        slog.Int64("new_value", newValue),

                        }
                case ResourceDisk:
                        if cache.diskGB.CompareAndSwap(current, newValue) {
                                t.logger.Debug("usage decremented",
                                        slog.String("tenant_id", tenantID),
                                        slog.String("resource", string(resource)),
                                        slog.Int64("amount", amount),
                                        slog.Int64("new_value", newValue),
                                )
                                return nil
                        }
                case ResourceBridges:
                        if cache.bridges.CompareAndSwap(current, newValue) {
                                t.logger.Debug("usage decremented",
                                        slog.String("tenant_id", tenantID),
                                        slog.String("resource", string(resource)),
                                        slog.Int64("amount", amount),
                                        slog.Int64("new_value", newValue),
                                )
                                return nil
                        }
                case ResourcePorts:
                        if cache.ports.CompareAndSwap(current, newValue) {
                                t.logger.Debug("usage decremented",
                                        slog.String("tenant_id", tenantID),
                                        slog.String("resource", string(resource)),
                                        slog.Int64("amount", amount),
                                        slog.Int64("new_value", newValue),
                                )
                                return nil
                        }
                case ResourceBandwidth:
                        if cache.bandwidthMbps.CompareAndSwap(current, newValue) {
                                t.logger.Debug("usage decremented",
                                        slog.String("tenant_id", tenantID),
                                        slog.String("resource", string(resource)),
                                        slog.Int64("amount", amount),
                                        slog.Int64("new_value", newValue),
                                )
                                return nil
                        }
                case ResourceVolumes:
                        if cache.volumes.CompareAndSwap(current, newValue) {
                                t.logger.Debug("usage decremented",
                                        slog.String("tenant_id", tenantID),
                                        slog.String("resource", string(resource)),
                                        slog.Int64("amount", amount),
                                        slog.Int64("new_value", newValue),
                                )
                                return nil
                        }
                case ResourceSnapshots:
                        if cache.snapshots.CompareAndSwap(current, newValue) {
                                t.logger.Debug("usage decremented",
                                        slog.String("tenant_id", tenantID),
                                        slog.String("resource", string(resource)),
                                        slog.Int64("amount", amount),
                                        slog.Int64("new_value", newValue),
                                )
                                return nil
                        }
                default:
                        return ErrInvalidResource{Resource: resource}
                }
                // CAS failed, retry.
        }
}

// GetUsage returns current usage for a resource.
func (t *Tracker) GetUsage(ctx context.Context, tenantID string, resource ResourceType) (int64, error) {
        if !resource.IsValid() {
                return 0, ErrInvalidResource{Resource: resource}
        }

        cache, err := t.getOrCreateCache(ctx, tenantID)
        if err != nil {
                return 0, err
        }

        return cache.get(resource), nil
}

// GetAllUsage returns all usage for a tenant.
func (t *Tracker) GetAllUsage(ctx context.Context, tenantID string) (*TenantUsage, error) {
        cache, err := t.getOrCreateCache(ctx, tenantID)
        if err != nil {
                return nil, err
        }

        return cache.toTenantUsage(tenantID), nil
}

// Sync persists in-memory usage to store.
func (t *Tracker) Sync(ctx context.Context, tenantID string) error {
        value, ok := t.cache.Load(tenantID)
        if !ok {
                return nil // Nothing to sync.
        }

        cache := value.(*usageCache)
        usage := cache.toTenantUsage(tenantID)
        usage.Timestamp = time.Now()

        if err := t.store.UpdateUsage(ctx, tenantID, usage); err != nil {
                return ErrStoreFailed{Operation: "UpdateUsage", Err: err}
        }

        cache.lastSync.Store(time.Now())

        t.logger.Debug("usage synced to store",
                slog.String("tenant_id", tenantID),
        )

        return nil
}

// SyncAll persists all tenant usage to store.
func (t *Tracker) SyncAll(ctx context.Context) error {
        var syncErrors []error

        t.cache.Range(func(key, value any) bool {
                tenantID := key.(string)
                if err := t.Sync(ctx, tenantID); err != nil {
                        syncErrors = append(syncErrors, err)
                        t.logger.Error("failed to sync tenant usage",
                                slog.String("tenant_id", tenantID),
                                slog.Any("error", err),
                        )
                }
                return true
        })

        if len(syncErrors) > 0 {
                return ErrStoreFailed{
                        Operation: "SyncAll",
                        Err:       syncErrors[0],
                }
        }

        return nil
}

// LoadFromStore loads usage from the store into cache for a tenant.
func (t *Tracker) LoadFromStore(ctx context.Context, tenantID string) error {
        usage, err := t.store.GetUsage(ctx, tenantID)
        if err != nil {
                return ErrStoreFailed{Operation: "GetUsage", Err: err}
        }

        cache := newUsageCache(usage)
        t.cache.Store(tenantID, cache)

        t.logger.Debug("usage loaded from store",
                slog.String("tenant_id", tenantID),
        )

        return nil
}

// Set directly sets the usage for a resource.
func (t *Tracker) Set(ctx context.Context, tenantID string, resource ResourceType, value int64) error {
        if !resource.IsValid() {
                return ErrInvalidResource{Resource: resource}
        }
        if value < 0 {
                return ErrInvalidAmount{Resource: resource, Amount: value}
        }

        cache, err := t.getOrCreateCache(ctx, tenantID)
        if err != nil {
                return err
        }

        cache.set(resource, value)

        t.logger.Debug("usage set",
                slog.String("tenant_id", tenantID),
                slog.String("resource", string(resource)),
                slog.Int64("value", value),
        )

        return nil
}

// Invalidate removes a tenant's cache, forcing reload on next access.
func (t *Tracker) Invalidate(tenantID string) {
        t.cache.Delete(tenantID)

        t.logger.Debug("usage cache invalidated",
                slog.String("tenant_id", tenantID),
        )
}

// InvalidateAll removes all cached data.
func (t *Tracker) InvalidateAll() {
        t.cache.Range(func(key, _ any) bool {
                t.cache.Delete(key)
                return true
        })

        t.logger.Debug("all usage caches invalidated")
}

// ComputeQuota defines compute resource quotas for a tenant.
type ComputeQuota struct {
        // Limits
        MaxVMs        int64 // Maximum number of VMs
        MaxContainers int64 // Maximum number of containers
        MaxVCPUs      int64 // Maximum number of virtual CPUs
        MaxMemoryMB   int64 // Maximum memory in megabytes
        MaxDiskGB     int64 // Maximum disk space in gigabytes

        // Usage tracking
        UsedVMs        int64 // Currently used VMs
        UsedContainers int64 // Currently used containers
        UsedVCPUs      int64 // Currently used vCPUs
        UsedMemoryMB   int64 // Currently used memory in MB
        UsedDiskGB     int64 // Currently used disk in GB
}

// HasCapacity checks if the quota has capacity for the requested resources.
func (q *ComputeQuota) HasCapacity(vms, containers, vcpus, memoryMB, diskGB int64) bool {
        if vms > 0 && q.UsedVMs+vms > q.MaxVMs {
                return false
        }
        if containers > 0 && q.UsedContainers+containers > q.MaxContainers {
                return false
        }
        if vcpus > 0 && q.UsedVCPUs+vcpus > q.MaxVCPUs {
                return false
        }
        if memoryMB > 0 && q.UsedMemoryMB+memoryMB > q.MaxMemoryMB {
                return false
        }
        if diskGB > 0 && q.UsedDiskGB+diskGB > q.MaxDiskGB {
                return false
        }
        return true
}

// Reserve reserves the specified resources from the quota.
// Returns an error if insufficient capacity.
func (q *ComputeQuota) Reserve(tenantID string, vms, containers, vcpus, memoryMB, diskGB int64) error {
        if vms > 0 && q.UsedVMs+vms > q.MaxVMs {
                return ErrQuotaExceeded{
                        TenantID:  tenantID,
                        Resource:  "VMs",
                        Requested: vms,
                        Available: q.MaxVMs - q.UsedVMs,
                }
        }
        if containers > 0 && q.UsedContainers+containers > q.MaxContainers {
                return ErrQuotaExceeded{
                        TenantID:  tenantID,
                        Resource:  "containers",
                        Requested: containers,
                        Available: q.MaxContainers - q.UsedContainers,
                }
        }
        if vcpus > 0 && q.UsedVCPUs+vcpus > q.MaxVCPUs {
                return ErrQuotaExceeded{
                        TenantID:  tenantID,
                        Resource:  "vCPUs",
                        Requested: vcpus,
                        Available: q.MaxVCPUs - q.UsedVCPUs,
                }
        }
        if memoryMB > 0 && q.UsedMemoryMB+memoryMB > q.MaxMemoryMB {
                return ErrQuotaExceeded{
                        TenantID:  tenantID,
                        Resource:  "memory",
                        Requested: memoryMB,
                        Available: q.MaxMemoryMB - q.UsedMemoryMB,
                }
        }
        if diskGB > 0 && q.UsedDiskGB+diskGB > q.MaxDiskGB {
                return ErrQuotaExceeded{
                        TenantID:  tenantID,
                        Resource:  "disk",
                        Requested: diskGB,
                        Available: q.MaxDiskGB - q.UsedDiskGB,
                }
        }

        q.UsedVMs += vms
        q.UsedContainers += containers
        q.UsedVCPUs += vcpus
        q.UsedMemoryMB += memoryMB
        q.UsedDiskGB += diskGB

        return nil
}

// Release releases the specified resources back to the quota.
func (q *ComputeQuota) Release(vms, containers, vcpus, memoryMB, diskGB int64) {
        q.UsedVMs = max(0, q.UsedVMs-vms)
        q.UsedContainers = max(0, q.UsedContainers-containers)
        q.UsedVCPUs = max(0, q.UsedVCPUs-vcpus)
        q.UsedMemoryMB = max(0, q.UsedMemoryMB-memoryMB)
        q.UsedDiskGB = max(0, q.UsedDiskGB-diskGB)
}

// GetUtilization returns the utilization percentage for each resource type.
func (q *ComputeQuota) GetUtilization() map[string]float64 {
        utilization := make(map[string]float64)

        utilization["vms"] = safePercentage(q.UsedVMs, q.MaxVMs)
        utilization["containers"] = safePercentage(q.UsedContainers, q.MaxContainers)
        utilization["vcpus"] = safePercentage(q.UsedVCPUs, q.MaxVCPUs)
        utilization["memory"] = safePercentage(q.UsedMemoryMB, q.MaxMemoryMB)
        utilization["disk"] = safePercentage(q.UsedDiskGB, q.MaxDiskGB)

        return utilization
}

// State represents the lifecycle state of a tenant.
type State string

const (
        // StateCreated indicates the tenant has been created but not yet activated.
        StateCreated State = "created"

        // StateActive indicates the tenant is fully operational.
        StateActive State = "active"

        // StateSuspended indicates the tenant is temporarily suspended.
        StateSuspended State = "suspended"

        // StateDeleting indicates the tenant is being deleted.
        StateDeleting State = "deleting"

        // StateDeleted indicates the tenant has been deleted.
        StateDeleted State = "deleted"
)

// ValidTransitions defines the allowed state transitions for tenant lifecycle.
// The map key is the current state and the value is a list of valid target states.
var ValidTransitions = map[State][]State{
        StateCreated: {
                StateActive,   // Tenant is activated after initial setup
                StateDeleting, // Tenant can be deleted before activation
        },
        StateActive: {
                StateSuspended, // Tenant can be suspended for various reasons
                StateDeleting,  // Tenant can be deleted when active
        },
        StateSuspended: {
                StateActive,   // Tenant can be reactivated
                StateDeleting, // Tenant can be deleted while suspended
        },
        StateDeleting: {
                StateDeleted, // Deletion completes
        },
        StateDeleted: {
                // No transitions allowed from deleted state
        },
}

// validStates contains all valid state values for quick lookup.
var validStates = map[State]struct{}{
        StateCreated:   {},
        StateActive:    {},
        StateSuspended: {},
        StateDeleting:  {},
        StateDeleted:   {},
}

// CanTransitionTo checks if transitioning from the current state to the target state is allowed.
func (s State) CanTransitionTo(target State) bool {
        // Same-state transitions are allowed for idempotency
        if s == target {
                return true
        }

        allowed, exists := ValidTransitions[s]
        if !exists {
                return false
        }

        for _, valid := range allowed {
                if valid == target {
                        return true
                }
        }

        return false
}

// String returns the string representation of the state.
func (s State) String() string {
        return string(s)
}

// IsValid checks if the state is a valid tenant state.
func (s State) IsValid() bool {
        _, exists := validStates[s]
        return exists
}

// IsTerminal checks if the state is a terminal state (no further transitions).
func (s State) IsTerminal() bool {
        return s == StateDeleted
}

// IsOperational checks if the tenant is in an operational state.
func (s State) IsOperational() bool {
        return s == StateActive
}

// AllowsResourceCreation checks if resources can be created in this state.
func (s State) AllowsResourceCreation() bool {
        return s == StateActive
}

// AllowsResourceDeletion checks if resources can be deleted in this state.
func (s State) AllowsResourceDeletion() bool {
        return s == StateActive || s == StateDeleting
}

// ParseState parses a string into a State value.
// Returns an error if the string does not represent a valid state.
func ParseState(s string) (State, error) {
        state := State(s)
        if !state.IsValid() {
                return "", ErrInvalidState{Current: state}
        }
        return state, nil
}

// AllStates returns all valid tenant states.
func AllStates() []State {
        return []State{
                StateCreated,
                StateActive,
                StateSuspended,
                StateDeleting,
                StateDeleted,
        }
}

// TransitionableStates returns states that can be transitioned from.
func TransitionableStates() []State {
        return []State{
                StateCreated,
                StateActive,
                StateSuspended,
                StateDeleting,
        }
}

// Allocator manages VLAN range allocation for tenants
type Allocator struct {
        store  Store
        logger *slog.Logger

        // Configuration
        globalRange Range  // e.g., 100-4000
        rangeSize   uint16 // Default size per tenant, e.g., 100

        // In-memory tracking
        mu        sync.RWMutex
        allocated map[string]Range // tenantID -> range
        usedVLANs *bitset          // Quick lookup of used VLANs
}

// NewAllocator creates a new VLAN allocator with the specified configuration
func NewAllocator(store Store, cfg *Config, logger *slog.Logger) (*Allocator, error) {
        if store == nil {
                return nil, &ErrNilStore{}
        }
        if cfg == nil {
                return nil, &ErrNilConfig{}
        }
        if err := cfg.Validate(); err != nil {
                return nil, err
        }
        if logger == nil {
                logger = slog.Default()
        }

        a := &Allocator{
                store:  store,
                logger: logger,
                globalRange: Range{
                        Start: cfg.GlobalRangeStart,
                        End:   cfg.GlobalRangeEnd,
                },
                rangeSize: cfg.DefaultRangeSize,
                allocated: make(map[string]Range),
                usedVLANs: newBitset(int(MaxVLANID) + 1),
        }

        // Load existing allocations from store
        if err := a.Reload(context.Background()); err != nil {
                return nil, err
        }

        return a, nil
}

// Allocate assigns a VLAN range to a tenant
// If requestedSize is 0, the default range size is used
func (a *Allocator) Allocate(ctx context.Context, tenantID string, requestedSize uint16) (*Allocation, error) {
        if tenantID == "" {
                return nil, &ErrEmptyTenantID{}
        }

        size := requestedSize
        if size == 0 {
                size = a.rangeSize
        }

        a.mu.Lock()
        defer a.mu.Unlock()

        // Check if tenant already has an allocation
        if existing, ok := a.allocated[tenantID]; ok {
                return nil, &ErrTenantAlreadyAllocated{
                        TenantID: tenantID,
                        Range:    existing,
                }
        }

        // Find an available range
        start, found := a.usedVLANs.FindClearRange(a.globalRange.Start, a.globalRange.End, size)
        if !found {
                return nil, &ErrVLANRangeExhausted{
                        Start: a.globalRange.Start,
                        End:   a.globalRange.End,
                }
        }

        allocRange := Range{
                Start: start,
                End:   start + size - 1,
        }

        // Validate the range is within global bounds
        if allocRange.End > a.globalRange.End {
                return nil, &ErrVLANRangeExhausted{
                        Start: a.globalRange.Start,
                        End:   a.globalRange.End,
                }
        }

        alloc := &Allocation{
                TenantID:    tenantID,
                Range:       allocRange,
                AllocatedAt: time.Now().UTC(),
        }

        // Persist to store
        if err := a.store.SaveAllocation(ctx, alloc); err != nil {
                return nil, &ErrStoreFailed{
                        Operation: "SaveAllocation",
                        Err:       err,
                }
        }

        // Update in-memory state
        a.allocated[tenantID] = allocRange
        a.usedVLANs.SetRange(allocRange.Start, allocRange.End)

        a.logger.Info("allocated VLAN range",
                slog.String("tenant_id", tenantID),
                slog.String("range", allocRange.String()),
        )

        return alloc, nil
}

// AllocateSpecific assigns a specific VLAN range to a tenant
func (a *Allocator) AllocateSpecific(ctx context.Context, tenantID string, requested Range) (*Allocation, error) {
        if tenantID == "" {
                return nil, &ErrEmptyTenantID{}
        }

        if err := requested.Validate(); err != nil {
                return nil, err
        }

        // Check requested range is within global bounds
        if requested.Start < a.globalRange.Start || requested.End > a.globalRange.End {
                return nil, &ErrVLANOutOfRange{
                        VLAN: requested.Start,
                        Min:  a.globalRange.Start,
                        Max:  a.globalRange.End,
                }
        }

        a.mu.Lock()
        defer a.mu.Unlock()

        // Check if tenant already has an allocation
        if existing, ok := a.allocated[tenantID]; ok {
                return nil, &ErrTenantAlreadyAllocated{
                        TenantID: tenantID,
                        Range:    existing,
                }
        }

        // Check for overlap with existing allocations
        for tid, existingRange := range a.allocated {
                if requested.Overlaps(existingRange) {
                        return nil, &ErrVLANRangeOverlap{
                                Existing:  existingRange,
                                Requested: requested,
                        }
                }
                // Also check via bitset for any partial overlaps
                _ = tid // Using tid in future logging if needed
        }

        // Verify the range is actually available in the bitset
        if !a.usedVLANs.TestRangeClear(requested.Start, requested.End) {
                // Find which allocation owns the overlapping VLAN
                for tid, existingRange := range a.allocated {
                        if requested.Overlaps(existingRange) {
                                return nil, &ErrVLANRangeOverlap{
                                        Existing:  existingRange,
                                        Requested: requested,
                                }
                        }
                        _ = tid
                }
                // If we can't find the owner, return a generic overlap error
                return nil, &ErrVLANRangeOverlap{
                        Existing:  Range{}, // Unknown
                        Requested: requested,
                }
        }

        alloc := &Allocation{
                TenantID:    tenantID,
                Range:       requested,
                AllocatedAt: time.Now().UTC(),
        }

        // Persist to store
        if err := a.store.SaveAllocation(ctx, alloc); err != nil {
                return nil, &ErrStoreFailed{
                        Operation: "SaveAllocation",
                        Err:       err,
                }
        }

        // Update in-memory state
        a.allocated[tenantID] = requested
        a.usedVLANs.SetRange(requested.Start, requested.End)

        a.logger.Info("allocated specific VLAN range",
                slog.String("tenant_id", tenantID),
                slog.String("range", requested.String()),
        )

        return alloc, nil
}

// Release frees a tenant's VLAN allocation
func (a *Allocator) Release(ctx context.Context, tenantID string) error {
        if tenantID == "" {
                return &ErrEmptyTenantID{}
        }

        a.mu.Lock()
        defer a.mu.Unlock()

        // Check if tenant has an allocation
        allocRange, ok := a.allocated[tenantID]
        if !ok {
                return &ErrTenantVLANNotFound{TenantID: tenantID}
        }

        // Delete from store
        if err := a.store.DeleteAllocation(ctx, tenantID); err != nil {
                return &ErrStoreFailed{
                        Operation: "DeleteAllocation",
                        Err:       err,
                }
        }

        // Update in-memory state
        delete(a.allocated, tenantID)
        a.usedVLANs.ClearRange(allocRange.Start, allocRange.End)

        a.logger.Info("released VLAN range",
                slog.String("tenant_id", tenantID),
                slog.String("range", allocRange.String()),
        )

        return nil
}

// Get returns a tenant's current allocation
func (a *Allocator) Get(ctx context.Context, tenantID string) (*Allocation, error) {
        if tenantID == "" {
                return nil, &ErrEmptyTenantID{}
        }

        a.mu.RLock()
        allocRange, ok := a.allocated[tenantID]
        a.mu.RUnlock()

        if ok {
                // Return from cache
                return &Allocation{
                        TenantID: tenantID,
                        Range:    allocRange,
                }, nil
        }

        // Try to load from store
        alloc, err := a.store.GetAllocation(ctx, tenantID)
        if err != nil {
                return nil, &ErrStoreFailed{
                        Operation: "GetAllocation",
                        Err:       err,
                }
        }
        if alloc == nil {
                return nil, &ErrTenantVLANNotFound{TenantID: tenantID}
        }

        // Update cache
        a.mu.Lock()
        a.allocated[tenantID] = alloc.Range
        a.usedVLANs.SetRange(alloc.Range.Start, alloc.Range.End)
        a.mu.Unlock()

        return alloc, nil
}

// GetByVLAN finds which tenant owns a VLAN
// Returns the tenant ID if found, or empty string with an error if not found
func (a *Allocator) GetByVLAN(ctx context.Context, vlan uint16) (string, error) {
        // Validate VLAN is in valid range
        if vlan < MinVLANID || vlan > MaxVLANID {
                return "", &ErrVLANOutOfRange{
                        VLAN: vlan,
                        Min:  MinVLANID,
                        Max:  MaxVLANID,
                }
        }

        a.mu.RLock()
        defer a.mu.RUnlock()

        // Check if VLAN is within global range
        if vlan < a.globalRange.Start || vlan > a.globalRange.End {
                return "", &ErrVLANOutOfRange{
                        VLAN: vlan,
                        Min:  a.globalRange.Start,
                        Max:  a.globalRange.End,
                }
        }

        // Quick check if VLAN is allocated at all
        if !a.usedVLANs.Test(vlan) {
                return "", nil // Not allocated
        }

        // Find the owning tenant
        for tenantID, allocRange := range a.allocated {
                if allocRange.Contains(vlan) {
                        return tenantID, nil
                }
        }

        // Should not reach here if state is consistent
        return "", nil
}

// IsAvailable checks if a VLAN range is available
func (a *Allocator) IsAvailable(ctx context.Context, r Range) bool {
        if err := r.Validate(); err != nil {
                return false
        }

        a.mu.RLock()
        defer a.mu.RUnlock()

        // Check if range is within global bounds
        if r.Start < a.globalRange.Start || r.End > a.globalRange.End {
                return false
        }

        // Check if all VLANs in range are clear
        return a.usedVLANs.TestRangeClear(r.Start, r.End)
}

// ListAllocations returns all VLAN allocations
func (a *Allocator) ListAllocations(ctx context.Context) ([]*Allocation, error) {
        a.mu.RLock()
        defer a.mu.RUnlock()

        allocations := make([]*Allocation, 0, len(a.allocated))
        for tenantID, allocRange := range a.allocated {
                allocations = append(allocations, &Allocation{
                        TenantID: tenantID,
                        Range:    allocRange,
                })
        }

        return allocations, nil
}

// Reload refreshes in-memory state from store
func (a *Allocator) Reload(ctx context.Context) error {
        allocations, err := a.store.ListAllocations(ctx)
        if err != nil {
                return &ErrStoreFailed{
                        Operation: "ListAllocations",
                        Err:       err,
                }
        }

        a.mu.Lock()
        defer a.mu.Unlock()

        // Reset in-memory state
        a.allocated = make(map[string]Range)
        a.usedVLANs.Reset()

        // Rebuild from store
        for _, alloc := range allocations {
                if alloc == nil || alloc.TenantID == "" {
                        continue
                }
                a.allocated[alloc.TenantID] = alloc.Range
                a.usedVLANs.SetRange(alloc.Range.Start, alloc.Range.End)
        }

        a.logger.Info("reloaded VLAN allocations",
                slog.Int("count", len(a.allocated)),
        )

        return nil
}

// GetGlobalRange returns the configured global VLAN range
func (a *Allocator) GetGlobalRange() Range {
        return a.globalRange
}

// GetDefaultRangeSize returns the default allocation size
func (a *Allocator) GetDefaultRangeSize() uint16 {
        return a.rangeSize
}

// GetAvailableCapacity returns the number of unallocated VLANs
func (a *Allocator) GetAvailableCapacity() int {
        a.mu.RLock()
        defer a.mu.RUnlock()

        globalSize := int(a.globalRange.End-a.globalRange.Start) + 1
        usedInRange := a.usedVLANs.CountSetInRange(a.globalRange.Start, a.globalRange.End)
        return globalSize - usedInRange
}

// GetAllocationCount returns the number of active allocations
func (a *Allocator) GetAllocationCount() int {
        a.mu.RLock()
        defer a.mu.RUnlock()
        return len(a.allocated)
}

// bitset provides efficient storage for tracking used VLANs
// VLANs 1-4094 fit in ~64 uint64 words (512 bytes)
// This implementation is lock-free using atomic operations
type bitset struct {
        bits []uint64
        size int
}

// newBitset creates a new bitset capable of tracking n bits
func newBitset(size int) *bitset {
        if size <= 0 {
                size = int(MaxVLANID) + 1
        }
        // Calculate number of uint64 words needed (64 bits per word)
        words := (size + 63) / 64
        return &bitset{
                bits: make([]uint64, words),
                size: size,
        }
}

// wordIndex returns the word index and bit offset for a given bit position
func (b *bitset) wordIndex(n uint16) (word int, bit uint64) {
        word = int(n) / 64
        bit = uint64(1) << (uint(n) % 64)
        return
}

// Set atomically sets the bit at position n
func (b *bitset) Set(n uint16) {
        if int(n) >= b.size {
                return
        }
        word, bit := b.wordIndex(n)
        addr := (*uint64)(unsafe.Pointer(&b.bits[word]))
        for {
                old := atomic.LoadUint64(addr)
                if old&bit != 0 {
                        return // Already set
                }
                if atomic.CompareAndSwapUint64(addr, old, old|bit) {
                        return
                }
        }
}

// Clear atomically clears the bit at position n
func (b *bitset) Clear(n uint16) {
        if int(n) >= b.size {
                return
        }
        word, bit := b.wordIndex(n)
        addr := (*uint64)(unsafe.Pointer(&b.bits[word]))
        for {
                old := atomic.LoadUint64(addr)
                if old&bit == 0 {
                        return // Already clear
                }
                if atomic.CompareAndSwapUint64(addr, old, old&^bit) {
                        return
                }
        }
}

// Test atomically checks if the bit at position n is set
func (b *bitset) Test(n uint16) bool {
        if int(n) >= b.size {
                return false
        }
        word, bit := b.wordIndex(n)
        addr := (*uint64)(unsafe.Pointer(&b.bits[word]))
        return atomic.LoadUint64(addr)&bit != 0
}

// SetRange atomically sets all bits in the range [start, end]
func (b *bitset) SetRange(start, end uint16) {
        for i := start; i <= end; i++ {
                b.Set(i)
                // Prevent overflow if end is at max value
                if i == end {
                        break
                }
        }
}

// ClearRange atomically clears all bits in the range [start, end]
func (b *bitset) ClearRange(start, end uint16) {
        for i := start; i <= end; i++ {
                b.Clear(i)
                // Prevent overflow if end is at max value
                if i == end {
                        break
                }
        }
}

// FindFirstClear finds the first clear bit in the range [start, end]
// Returns the bit position and true if found, or 0 and false if not found
func (b *bitset) FindFirstClear(start, end uint16) (uint16, bool) {
        if int(start) >= b.size || start > end {
                return 0, false
        }
        if int(end) >= b.size {
                end = uint16(b.size - 1)
        }

        for i := start; i <= end; i++ {
                if !b.Test(i) {
                        return i, true
                }
                // Prevent overflow if end is at max value
                if i == end {
                        break
                }
        }
        return 0, false
}

// FindClearRange finds a contiguous range of clear bits of the specified size
// within the range [start, end]
// Returns the starting position and true if found, or 0 and false if not found
func (b *bitset) FindClearRange(start, end uint16, size uint16) (uint16, bool) {
        if size == 0 || int(start) >= b.size || start > end {
                return 0, false
        }
        if int(end) >= b.size {
                end = uint16(b.size - 1)
        }

        // Need at least 'size' bits available in the search range
        availableRange := end - start + 1
        if availableRange < size {
                return 0, false
        }

        rangeStart := start
        count := uint16(0)

        for i := start; i <= end; i++ {
                if !b.Test(i) {
                        if count == 0 {
                                rangeStart = i
                        }
                        count++
                        if count >= size {
                                return rangeStart, true
                        }
                } else {
                        count = 0
                }
                // Prevent overflow if end is at max value
                if i == end {
                        break
                }
        }
        return 0, false
}

// CountSet returns the number of set bits
func (b *bitset) CountSet() int {
        count := 0
        for i := 0; i < len(b.bits); i++ {
                addr := (*uint64)(unsafe.Pointer(&b.bits[i]))
                word := atomic.LoadUint64(addr)
                count += popcount(word)
        }
        return count
}

// CountClear returns the number of clear bits within the valid size
func (b *bitset) CountClear() int {
        return b.size - b.CountSet()
}

// CountSetInRange returns the number of set bits in the range [start, end]
func (b *bitset) CountSetInRange(start, end uint16) int {
        if start > end {
                return 0
        }
        count := 0
        for i := start; i <= end; i++ {
                if b.Test(i) {
                        count++
                }
                // Prevent overflow if end is at max value
                if i == end {
                        break
                }
        }
        return count
}

// TestRange returns true if all bits in the range [start, end] are set
func (b *bitset) TestRange(start, end uint16) bool {
        for i := start; i <= end; i++ {
                if !b.Test(i) {
                        return false
                }
                // Prevent overflow if end is at max value
                if i == end {
                        break
                }
        }
        return true
}

// TestRangeClear returns true if all bits in the range [start, end] are clear
func (b *bitset) TestRangeClear(start, end uint16) bool {
        for i := start; i <= end; i++ {
                if b.Test(i) {
                        return false
                }
                // Prevent overflow if end is at max value
                if i == end {
                        break
                }
        }
        return true
}

// Reset clears all bits in the bitset
func (b *bitset) Reset() {
        for i := 0; i < len(b.bits); i++ {
                addr := (*uint64)(unsafe.Pointer(&b.bits[i]))
                atomic.StoreUint64(addr, 0)
        }
}

// Clone returns a copy of the bitset
// Note: This is not atomic across the entire bitset
func (b *bitset) Clone() *bitset {
        newBits := make([]uint64, len(b.bits))
        for i := 0; i < len(b.bits); i++ {
                addr := (*uint64)(unsafe.Pointer(&b.bits[i]))
                newBits[i] = atomic.LoadUint64(addr)
        }
        return &bitset{
                bits: newBits,
                size: b.size,
        }
}

// popcount returns the number of set bits in a uint64 using Brian Kernighan's algorithm
func popcount(x uint64) int {
        count := 0
        for x != 0 {
                x &= x - 1
                count++
        }
        return count
}

import (
        "context"
        "fmt"

const (
        // MinVLANID is the minimum valid VLAN ID (VLAN 0 is reserved)
        MinVLANID uint16 = 1

        // MaxVLANID is the maximum valid VLAN ID (4095 is reserved)
        MaxVLANID uint16 = 4094

        // DefaultRangeSize is the default number of VLANs allocated per tenant
        DefaultRangeSize uint16 = 100
)

// Range represents a VLAN ID range
type Range struct {
        Start uint16
        End   uint16
}

// Contains returns true if the specified VLAN is within this range
func (r Range) Contains(vlan uint16) bool {
        return vlan >= r.Start && vlan <= r.End
}

// Size returns the number of VLANs in this range
func (r Range) Size() int {
        if r.End < r.Start {
                return 0
        }
        return int(r.End-r.Start) + 1
}

// Validate checks if the range is valid according to VLAN specifications
func (r Range) Validate() error {
        if r.Start < MinVLANID {
                return &ErrInvalidVLANRange{
                        Start:  r.Start,
                        End:    r.End,
                        Reason: fmt.Sprintf("start VLAN %d is below minimum %d", r.Start, MinVLANID),
                }
        }
        if r.End > MaxVLANID {
                return &ErrInvalidVLANRange{
                        Start:  r.Start,
                        End:    r.End,
                        Reason: fmt.Sprintf("end VLAN %d exceeds maximum %d", r.End, MaxVLANID),
                }
        }
        if r.Start > r.End {
                return &ErrInvalidVLANRange{
                        Start:  r.Start,
                        End:    r.End,
                        Reason: "start VLAN exceeds end VLAN",
                }
        }
        return nil
}

// Overlaps returns true if this range overlaps with another range
func (r Range) Overlaps(other Range) bool {
        return r.Start <= other.End && other.Start <= r.End
}

// String returns a string representation of the range
func (r Range) String() string {
        return fmt.Sprintf("%d-%d", r.Start, r.End)
}

// IsEmpty returns true if the range has no VLANs
func (r Range) IsEmpty() bool {
        return r.Size() == 0
}

// Allocation represents a tenant's VLAN allocation
type Allocation struct {
        TenantID    string
        Range       Range
        AllocatedAt time.Time
}

// IsValid checks if the allocation is valid
func (a *Allocation) IsValid() bool {
        if a == nil {
                return false
        }
        if a.TenantID == "" {
                return false
        }
        return a.Range.Validate() == nil
}

// Contains returns true if this allocation contains the specified VLAN
func (a *Allocation) Contains(vlan uint16) bool {
        if a == nil {
                return false
        }
        return a.Range.Contains(vlan)
}

// Store is the interface for persisting VLAN allocations
type Store interface {
        // GetAllocation retrieves a tenant's VLAN allocation
        GetAllocation(ctx context.Context, tenantID string) (*Allocation, error)

        // SaveAllocation persists a VLAN allocation
        SaveAllocation(ctx context.Context, alloc *Allocation) error

        // DeleteAllocation removes a tenant's VLAN allocation
        DeleteAllocation(ctx context.Context, tenantID string) error

        // ListAllocations returns all VLAN allocations
        ListAllocations(ctx context.Context) ([]*Allocation, error)
}

// Config holds the allocator configuration
type Config struct {
        GlobalRangeStart uint16
        GlobalRangeEnd   uint16
        DefaultRangeSize uint16
}

// Validate checks if the configuration is valid
func (c *Config) Validate() error {
        if c.GlobalRangeStart < MinVLANID {
                return &ErrInvalidVLANRange{
                        Start:  c.GlobalRangeStart,
                        End:    c.GlobalRangeEnd,
                        Reason: fmt.Sprintf("global range start %d is below minimum %d", c.GlobalRangeStart, MinVLANID),
                }
        }
        if c.GlobalRangeEnd > MaxVLANID {
                return &ErrInvalidVLANRange{
                        Start:  c.GlobalRangeStart,
                        End:    c.GlobalRangeEnd,
                        Reason: fmt.Sprintf("global range end %d exceeds maximum %d", c.GlobalRangeEnd, MaxVLANID),
                }
        }
        if c.GlobalRangeStart > c.GlobalRangeEnd {
                return &ErrInvalidVLANRange{
                        Start:  c.GlobalRangeStart,
                        End:    c.GlobalRangeEnd,
                        Reason: "global range start exceeds end",
                }
        }
        if c.DefaultRangeSize == 0 {
                return &ErrInvalidVLANRange{
                        Start:  c.GlobalRangeStart,
                        End:    c.GlobalRangeEnd,
                        Reason: "default range size cannot be zero",
                }
        }
        globalSize := int(c.GlobalRangeEnd-c.GlobalRangeStart) + 1
        if int(c.DefaultRangeSize) > globalSize {
                return &ErrInvalidVLANRange{
                        Start:  c.GlobalRangeStart,
                        End:    c.GlobalRangeEnd,
                        Reason: fmt.Sprintf("default range size %d exceeds global range size %d", c.DefaultRangeSize, globalSize),
                }
        }
        return nil
}

// DefaultConfig returns a default configuration
func DefaultConfig() *Config {
        return &Config{
                GlobalRangeStart: 100,
                GlobalRangeEnd:   4000,
                DefaultRangeSize: DefaultRangeSize,
        }
}

// Validator validates VLAN usage and access
type Validator struct {
        allocator *Allocator
        logger    *slog.Logger
}

// NewValidator creates a new VLAN validator
func NewValidator(allocator *Allocator, logger *slog.Logger) *Validator {
        if logger == nil {
                logger = slog.Default()
        }
        return &Validator{
                allocator: allocator,
                logger:    logger,
        }
}

// ValidateTenantVLAN checks if a tenant can use a specific VLAN
// Returns nil if the tenant owns the VLAN, otherwise returns an error
func (v *Validator) ValidateTenantVLAN(ctx context.Context, tenantID string, vlan uint16) error {
        if tenantID == "" {
                return &ErrEmptyTenantID{}
        }

        // Validate VLAN is in valid IEEE 802.1Q range
        if vlan < MinVLANID || vlan > MaxVLANID {
                return &ErrVLANOutOfRange{
                        VLAN: vlan,
                        Min:  MinVLANID,
                        Max:  MaxVLANID,
                }
        }

        // Get the tenant's allocation
        alloc, err := v.allocator.Get(ctx, tenantID)
        if err != nil {
                return err
        }

        // Check if VLAN is within the tenant's allocated range
        if !alloc.Range.Contains(vlan) {
                // Find who owns this VLAN
                owner, _ := v.allocator.GetByVLAN(ctx, vlan)
                if owner != "" {
                        return &ErrVLANInUse{
                                VLAN:     vlan,
                                TenantID: owner,
                        }
                }
                return &ErrVLANOutOfRange{
                        VLAN: vlan,
                        Min:  alloc.Range.Start,
                        Max:  alloc.Range.End,
                }
        }

        return nil
}

// ValidatePortVLAN validates VLAN assignment for a port
// This is a convenience method that combines VLAN validation with port context
func (v *Validator) ValidatePortVLAN(ctx context.Context, tenantID string, portVLAN uint16) error {
        if tenantID == "" {
                return &ErrEmptyTenantID{}
        }

        // Validate the VLAN ID is in valid range
        if portVLAN < MinVLANID || portVLAN > MaxVLANID {
                return &ErrVLANOutOfRange{
                        VLAN: portVLAN,
                        Min:  MinVLANID,
                        Max:  MaxVLANID,
                }
        }

        // Check if tenant has a VLAN allocation
        alloc, err := v.allocator.Get(ctx, tenantID)
        if err != nil {
                return err
        }

        // Verify the port VLAN is within tenant's allocation
        if !alloc.Range.Contains(portVLAN) {
                return &ErrVLANOutOfRange{
                        VLAN: portVLAN,
                        Min:  alloc.Range.Start,
                        Max:  alloc.Range.End,
                }
        }

        v.logger.Debug("validated port VLAN",
                slog.String("tenant_id", tenantID),
                slog.Uint64("vlan", uint64(portVLAN)),
        )

        return nil
}

// ValidateRange validates a VLAN range configuration
func (v *Validator) ValidateRange(ctx context.Context, r Range) error {
        // Validate basic range properties
        if err := r.Validate(); err != nil {
                return err
        }

        // Check if range is within allocator's global range
        globalRange := v.allocator.GetGlobalRange()
        if r.Start < globalRange.Start {
                return &ErrVLANOutOfRange{
                        VLAN: r.Start,
                        Min:  globalRange.Start,
                        Max:  globalRange.End,
                }
        }
        if r.End > globalRange.End {
                return &ErrVLANOutOfRange{
                        VLAN: r.End,
                        Min:  globalRange.Start,
                        Max:  globalRange.End,
                }
        }

        return nil
}

// CheckOverlap checks if a range would overlap with existing allocations
// The tenantID parameter allows excluding a specific tenant's existing allocation
// from the overlap check (useful when expanding a tenant's range)
func (v *Validator) CheckOverlap(ctx context.Context, tenantID string, r Range) error {
        if err := r.Validate(); err != nil {
                return err
        }

        // Get all allocations
        allocations, err := v.allocator.ListAllocations(ctx)
        if err != nil {
                return err
        }

        // Check for overlaps with other tenants
        for _, alloc := range allocations {
                // Skip the tenant we're checking for (allows expanding existing allocation)
                if alloc.TenantID == tenantID {
                        continue
                }

                if r.Overlaps(alloc.Range) {
                        return &ErrVLANRangeOverlap{
                                Existing:  alloc.Range,
                                Requested: r,
                        }
                }
        }

        return nil
}

// IsVLANAvailable checks if a specific VLAN is available for allocation
func (v *Validator) IsVLANAvailable(ctx context.Context, vlan uint16) bool {
        if vlan < MinVLANID || vlan > MaxVLANID {
                return false
        }

        owner, _ := v.allocator.GetByVLAN(ctx, vlan)
        return owner == ""
}

// IsRangeAvailable checks if a VLAN range is available for allocation
func (v *Validator) IsRangeAvailable(ctx context.Context, r Range) bool {
        return v.allocator.IsAvailable(ctx, r)
}

// GetTenantForVLAN returns the tenant ID that owns a specific VLAN
// Returns empty string if the VLAN is not allocated
func (v *Validator) GetTenantForVLAN(ctx context.Context, vlan uint16) (string, error) {
        return v.allocator.GetByVLAN(ctx, vlan)
}

// ValidateVLANBoundary checks if a VLAN crosses tenant boundaries
// This is useful for validating VLAN trunk configurations
func (v *Validator) ValidateVLANBoundary(ctx context.Context, tenantID string, vlans []uint16) error {
        if tenantID == "" {
                return &ErrEmptyTenantID{}
        }

        alloc, err := v.allocator.Get(ctx, tenantID)
        if err != nil {
                return err
        }

        for _, vlan := range vlans {
                if !alloc.Range.Contains(vlan) {
                        // Check if it belongs to another tenant
                        owner, _ := v.allocator.GetByVLAN(ctx, vlan)
                        if owner != "" && owner != tenantID {
                                return &ErrVLANInUse{
                                        VLAN:     vlan,
                                        TenantID: owner,
                                }
                        }
                        return &ErrVLANOutOfRange{
                                VLAN: vlan,
                                Min:  alloc.Range.Start,
                                Max:  alloc.Range.End,
                        }
                }
        }

        return nil
}
