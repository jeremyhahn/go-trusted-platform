const (
        bridgeTable = "Bridge"
        ovsTable    = "Open_vSwitch"
)

// CreateBridge creates a new OVS bridge
// This creates the bridge along with the required internal port and interface
// (same as ovs-vsctl add-br)
func (c *Client) CreateBridge(ctx context.Context, name string) (*Bridge, error) {
        if name == "" {
                return nil, ErrInvalidBridgeName
        }

        // Check if bridge already exists
        existing, err := c.GetBridge(ctx, name)
        if err == nil && existing != nil {
                return nil, ErrBridgeExists
        }

        // Get OVS UUID first
        ovsUUID, err := c.getOpenVSwitchUUID(ctx)
        if err != nil {
                return nil, fmt.Errorf("failed to get Open_vSwitch UUID: %w", err)
        }

        // Create internal interface (required for bridge to work properly)
        interfaceRow := map[string]interface{}{
                "name": name,
                "type": "internal",
        }
        interfaceInsert := NewInsertOperation(interfaceTable, interfaceRow, "new_interface")

        // Create internal port referencing the interface
        portRow := map[string]interface{}{
                "name":       name,
                "interfaces": NamedUUID{NamedUUID: "new_interface"},
        }
        portInsert := NewInsertOperation(portTable, portRow, "new_port")

        // Create bridge row with reference to the port
        bridgeRow := map[string]interface{}{
                "name":          name,
                "datapath_type": "",
                "ports":         NamedUUID{NamedUUID: "new_port"},
        }
        bridgeInsert := NewInsertOperation(bridgeTable, bridgeRow, "new_bridge")

        // Mutate OVS table to add bridge reference
        ovsMutate := NewMutateOperation(
                ovsTable,
                []Condition{NewCondition("_uuid", "==", []interface{}{"uuid", ovsUUID})},
                []Mutation{NewMutation("bridges", "insert", NamedUUID{NamedUUID: "new_bridge"})},
        )

        // Execute transaction: interface -> port -> bridge -> OVS mutation
        results, err := c.Transact(ctx, interfaceInsert, portInsert, bridgeInsert, ovsMutate)
        if err != nil {
                return nil, fmt.Errorf("failed to create bridge: %w", err)
        }

        if len(results) < 3 || results[2].UUID == nil {
                return nil, ErrTransactionFailed
        }

        // Retrieve the created bridge
        return c.GetBridge(ctx, name)
}

// DeleteBridge deletes an OVS bridge
func (c *Client) DeleteBridge(ctx context.Context, name string) error {
        if name == "" {
                return ErrInvalidBridgeName
        }

        // Get bridge to ensure it exists
        bridge, err := c.GetBridge(ctx, name)
        if err != nil {
                return err
        }
        if bridge == nil {
                return ErrBridgeNotFound
        }

        // Get OVS UUID
        ovsUUID, err := c.getOpenVSwitchUUID(ctx)
        if err != nil {
                return fmt.Errorf("failed to get Open_vSwitch UUID: %w", err)
        }

        // Remove bridge from OVS table
        ovsMutate := NewMutateOperation(
                ovsTable,
                []Condition{NewCondition("_uuid", "==", []interface{}{"uuid", ovsUUID})},
                []Mutation{NewMutation("bridges", "delete", []interface{}{"uuid", bridge.UUID})},
        )

        // Delete bridge
        bridgeDelete := NewDeleteOperation(
                bridgeTable,
                []Condition{NewCondition("_uuid", "==", []interface{}{"uuid", bridge.UUID})},
        )

        // Execute transaction
        _, err = c.Transact(ctx, ovsMutate, bridgeDelete)
        if err != nil {
                return fmt.Errorf("failed to delete bridge: %w", err)
        }

        return nil
}

// GetBridge retrieves a bridge by name
func (c *Client) GetBridge(ctx context.Context, name string) (*Bridge, error) {
        if name == "" {
                return nil, ErrInvalidBridgeName
        }

        rows, err := c.Select(ctx, bridgeTable, []Condition{NewCondition("name", "==", name)}, nil)
        if err != nil {
                return nil, err
        }

        if len(rows) == 0 {
                return nil, ErrBridgeNotFound
        }

        return parseBridge(rows[0])
}

// ListBridges returns all bridges
func (c *Client) ListBridges(ctx context.Context) ([]*Bridge, error) {
        rows, err := c.Select(ctx, bridgeTable, nil, nil)
        if err != nil {
                return nil, err
        }

        bridges := make([]*Bridge, 0, len(rows))
        for _, row := range rows {
                bridge, err := parseBridge(row)
                if err != nil {
                        continue // Skip unparseable bridges
                }
                bridges = append(bridges, bridge)
        }

        return bridges, nil
}

// UpdateBridge updates bridge properties
func (c *Client) UpdateBridge(ctx context.Context, name string, updates map[string]interface{}) error {
        if name == "" {
                return ErrInvalidBridgeName
        }

        bridge, err := c.GetBridge(ctx, name)
        if err != nil {
                return err
        }

        updateOp := NewUpdateOperation(
                bridgeTable,
                []Condition{NewCondition("_uuid", "==", []interface{}{"uuid", bridge.UUID})},
                updates,
        )

        _, err = c.Transact(ctx, updateOp)
        if err != nil {
                return fmt.Errorf("failed to update bridge: %w", err)
        }

        return nil
}

// SetBridgeFailMode sets the fail mode for a bridge
func (c *Client) SetBridgeFailMode(ctx context.Context, name string, failMode string) error {
        if failMode != "standalone" && failMode != "secure" && failMode != "" {
                return fmt.Errorf("invalid fail mode: %s (must be 'standalone', 'secure', or empty)", failMode)
        }

        var value interface{}
        if failMode == "" {
                value = Set{Set: []interface{}{}}
        } else {
                value = failMode
        }

        return c.UpdateBridge(ctx, name, map[string]interface{}{
                "fail_mode": value,
        })
}

// SetBridgeProtocols sets the OpenFlow protocols for a bridge
func (c *Client) SetBridgeProtocols(ctx context.Context, name string, protocols []string) error {
        for _, proto := range protocols {
                if !isValidProtocol(proto) {
                        return ErrInvalidProtocol
                }
        }

        value := Set{Set: make([]interface{}, len(protocols))}
        for i, p := range protocols {
                value.Set[i] = p
        }

        return c.UpdateBridge(ctx, name, map[string]interface{}{
                "protocols": value,
        })
}

// parseBridge parses a bridge row from OVSDB
func parseBridge(row map[string]interface{}) (*Bridge, error) {
        bridge := &Bridge{
                ExternalIDs: make(map[string]string),
                OtherConfig: make(map[string]string),
                Status:      make(map[string]string),
        }

        // Parse UUID
        if uuidVal, ok := row["_uuid"]; ok {
                if uuidArr, ok := uuidVal.([]interface{}); ok && len(uuidArr) == 2 {
                        if uuidStr, ok := uuidArr[1].(string); ok {
                                bridge.UUID = uuidStr
                        }
                }
        }

        // Parse name
        if name, ok := row["name"].(string); ok {
                bridge.Name = name
        }

        // Parse datapath_type
        if dt, ok := row["datapath_type"].(string); ok {
                bridge.DatapathType = dt
        }

        // Parse fail_mode
        if fm, ok := row["fail_mode"].(string); ok {
                bridge.FailMode = fm
        } else if fmArr, ok := row["fail_mode"].([]interface{}); ok && len(fmArr) == 2 {
                if fmStr, ok := fmArr[1].(string); ok {
                        bridge.FailMode = fmStr
                }
        }

        // Parse ports
        bridge.Ports = parseUUIDSet(row["ports"])

        // Parse protocols
        bridge.FlowProtocols = parseStringSet(row["protocols"])

        // Parse STP enable
        if stp, ok := row["stp_enable"].(bool); ok {
                bridge.STP = stp
        }

        // Parse RSTP enable
        if rstp, ok := row["rstp_enable"].(bool); ok {
                bridge.RSTP = rstp
        }

        // Parse mcast_snooping_enable
        if mcast, ok := row["mcast_snooping_enable"].(bool); ok {
                bridge.McastSnooping = mcast
        }

        // Parse external_ids
        bridge.ExternalIDs = parseMap(row["external_ids"])

        // Parse other_config
        bridge.OtherConfig = parseMap(row["other_config"])

        // Parse status
        bridge.Status = parseMap(row["status"])

        return bridge, nil
}

// parseUUIDSet parses an OVSDB set of UUIDs
func parseUUIDSet(val interface{}) []string {
        if val == nil {
                return nil
        }

        // Handle set format: ["set", [["uuid", "xxx"], ["uuid", "yyy"]]]
        if arr, ok := val.([]interface{}); ok {
                if len(arr) == 2 && arr[0] == "set" {
                        if items, ok := arr[1].([]interface{}); ok {
                                result := make([]string, 0, len(items))
                                for _, item := range items {
                                        if uuidArr, ok := item.([]interface{}); ok && len(uuidArr) == 2 {
                                                if uuidStr, ok := uuidArr[1].(string); ok {
                                                        result = append(result, uuidStr)
                                                }
                                        }
                                }
                                return result
                        }
                }
                // Handle single UUID: ["uuid", "xxx"]
                if len(arr) == 2 && arr[0] == "uuid" {
                        if uuidStr, ok := arr[1].(string); ok {
                                return []string{uuidStr}
                        }
                }
        }

        return nil
}

// parseStringSet parses an OVSDB set of strings
func parseStringSet(val interface{}) []string {
        if val == nil {
                return nil
        }

        // Handle set format: ["set", ["str1", "str2"]]
        if arr, ok := val.([]interface{}); ok {
                if len(arr) == 2 && arr[0] == "set" {
                        if items, ok := arr[1].([]interface{}); ok {
                                result := make([]string, 0, len(items))
                                for _, item := range items {
                                        if s, ok := item.(string); ok {
                                                result = append(result, s)
                                        }
                                }
                                return result
                        }
                }
        }

        // Handle single string
        if s, ok := val.(string); ok {
                return []string{s}
        }

        return nil
}

// parseMap parses an OVSDB map
func parseMap(val interface{}) map[string]string {
        result := make(map[string]string)

        if val == nil {
                return result
        }

        // Handle map format: ["map", [["key1", "val1"], ["key2", "val2"]]]
        if arr, ok := val.([]interface{}); ok && len(arr) == 2 && arr[0] == "map" {
                if items, ok := arr[1].([]interface{}); ok {
                        for _, item := range items {
                                if pair, ok := item.([]interface{}); ok && len(pair) == 2 {
                                        if key, ok := pair[0].(string); ok {
                                                if value, ok := pair[1].(string); ok {
                                                        result[key] = value
                                                }
                                        }
                                }
                        }
                }
        }

        return result
}

// isValidProtocol checks if an OpenFlow protocol version is valid
func isValidProtocol(protocol string) bool {
        validProtocols := []string{
                "OpenFlow10", "OpenFlow11", "OpenFlow12",
                "OpenFlow13", "OpenFlow14", "OpenFlow15",
        }

        for _, valid := range validProtocols {
                if strings.EqualFold(protocol, valid) {
                        return true
                }
        }

        return false
}

// Client represents an OVSDB client connection
type Client struct {
        socketPath string
        database   string
        conn       net.Conn
        encoder    *json.Encoder
        decoder    *json.Decoder
        mu         sync.RWMutex
        connected  bool
        timeout    time.Duration

        // Response handling
        pending   map[uint64]chan *JSONRPCResponse
        pendingMu sync.Mutex

        // Shutdown
        done   chan struct{}
        wg     sync.WaitGroup
        closed bool

        // OpenFlow clients per bridge (pure Go implementation)
        ofClients   map[string]*openflow.Client
        ofClientsMu sync.RWMutex
        ofSocketDir string // Directory for OpenFlow sockets
}

// ClientOption is a functional option for configuring the Client
type ClientOption func(*Client)

// WithDatabase sets the OVSDB database name
func WithDatabase(database string) ClientOption {
        return func(c *Client) {
                c.database = database
        }
}

// WithTimeout sets the operation timeout
func WithTimeout(timeout time.Duration) ClientOption {
        return func(c *Client) {
                c.timeout = timeout
        }
}

// WithOpenFlowSocketDir sets the directory for OpenFlow management sockets
func WithOpenFlowSocketDir(dir string) ClientOption {
        return func(c *Client) {
                c.ofSocketDir = dir
        }
}

// NewClient creates a new OVSDB client
func NewClient(socketPath string, opts ...ClientOption) (*Client, error) {
        if socketPath == "" {
                return nil, ErrInvalidSocket
        }

        client := &Client{
                socketPath:  socketPath,
                database:    DefaultDatabase,
                timeout:     30 * time.Second,
                pending:     make(map[uint64]chan *JSONRPCResponse),
                done:        make(chan struct{}),
                ofClients:   make(map[string]*openflow.Client),
                ofSocketDir: "/var/run/openvswitch",
        }

        for _, opt := range opts {
                opt(client)
        }

        return client, nil
}

// Connect establishes a connection to the OVSDB server
func (c *Client) Connect(ctx context.Context) error {
        c.mu.Lock()
        defer c.mu.Unlock()

        if c.connected {
                return nil
        }

        // Connect to the Unix socket
        var d net.Dialer
        conn, err := d.DialContext(ctx, "unix", c.socketPath)
        if err != nil {
                return fmt.Errorf("%w: %v", ErrConnectionFailed, err)
        }

        c.conn = conn
        c.encoder = json.NewEncoder(conn)
        c.decoder = json.NewDecoder(bufio.NewReader(conn))
        c.connected = true
        c.closed = false
        c.done = make(chan struct{})

        // Start the response listener
        c.wg.Add(1)
        go c.responseListener()

        return nil
}

// Disconnect closes the connection to the OVSDB server
func (c *Client) Disconnect() error {
        c.mu.Lock()
        defer c.mu.Unlock()

        if !c.connected {
                return nil
        }

        c.connected = false
        c.closed = true

        // Signal shutdown
        close(c.done)

        // Close connection
        if c.conn != nil {
                c.conn.Close()
        }

        // Wait for listener to exit
        c.wg.Wait()

        // Clean up pending requests
        c.pendingMu.Lock()
        for id, ch := range c.pending {
                close(ch)
                delete(c.pending, id)
        }
        c.pendingMu.Unlock()

        // Close all OpenFlow clients
        c.ofClientsMu.Lock()
        for name, ofClient := range c.ofClients {
                ofClient.Close()
                delete(c.ofClients, name)
        }
        c.ofClientsMu.Unlock()

        return nil
}

// IsConnected returns whether the client is connected
func (c *Client) IsConnected() bool {
        c.mu.Lock()
        defer c.mu.Unlock()
        return c.connected
}

// Close is an alias for Disconnect
func (c *Client) Close() error {
        return c.Disconnect()
}

// responseListener reads responses from the socket and dispatches them
func (c *Client) responseListener() {
        defer c.wg.Done()

        for {
                select {
                case <-c.done:
                        return
                default:
                }

                // Set read deadline to allow periodic checking of done channel
                if c.conn != nil {
                        c.conn.SetReadDeadline(time.Now().Add(100 * time.Millisecond))
                }

                var response JSONRPCResponse
                if err := c.decoder.Decode(&response); err != nil {
                        // Check if we're shutting down
                        select {
                        case <-c.done:
                                return
                        default:
                        }

                        // Timeout is expected, continue
                        if netErr, ok := err.(net.Error); ok && netErr.Timeout() {
                                continue
                        }

                        // Other error - connection may be broken
                        c.mu.Lock()
                        if c.connected && !c.closed {
                                c.connected = false
                        }
                        c.mu.Unlock()
                        return
                }

                // Clear deadline for next read
                if c.conn != nil {
                        c.conn.SetReadDeadline(time.Time{})
                }

                // Dispatch response to waiting caller
                c.pendingMu.Lock()
                if ch, ok := c.pending[response.ID]; ok {
                        select {
                        case ch <- &response:
                        default:
                        }
                        delete(c.pending, response.ID)
                }
                c.pendingMu.Unlock()
        }
}

// call sends a JSON-RPC request and waits for response
func (c *Client) call(ctx context.Context, method string, params []interface{}) (*JSONRPCResponse, error) {
        if !c.IsConnected() {
                return nil, ErrNotConnected
        }

        // Create request
        req := NewJSONRPCRequest(method, params)

        // Create response channel
        respCh := make(chan *JSONRPCResponse, 1)

        // Register pending request
        c.pendingMu.Lock()
        c.pending[req.ID] = respCh
        c.pendingMu.Unlock()

        // Cleanup on exit
        defer func() {
                c.pendingMu.Lock()
                delete(c.pending, req.ID)
                c.pendingMu.Unlock()
        }()

        // Send request
        c.mu.Lock()
        err := c.encoder.Encode(req)
        c.mu.Unlock()

        if err != nil {
                return nil, fmt.Errorf("failed to send request: %w", err)
        }

        // Wait for response with timeout
        select {
        case <-ctx.Done():
                return nil, ctx.Err()
        case resp := <-respCh:
                if resp == nil {
                        return nil, ErrConnectionClosed
                }
                if resp.Error != nil {
                        return nil, fmt.Errorf("OVSDB error: %s", resp.Error.Message)
                }
                return resp, nil
        }
}

// Transact executes a transaction on the OVSDB server
func (c *Client) Transact(ctx context.Context, operations ...Operation) (TransactResponse, error) {
        // Build params: database name followed by operations
        params := make([]interface{}, 0, len(operations)+1)
        params = append(params, c.database)
        for _, op := range operations {
                params = append(params, op)
        }

        resp, err := c.call(ctx, "transact", params)
        if err != nil {
                return nil, err
        }

        var result TransactResponse
        if err := json.Unmarshal(resp.Result, &result); err != nil {
                return nil, fmt.Errorf("failed to parse transact result: %w", err)
        }

        // Check for operation errors
        for i, opResult := range result {
                if opResult.Error != "" {
                        return result, fmt.Errorf("operation %d failed: %s - %s", i, opResult.Error, opResult.Details)
                }
        }

        return result, nil
}

// Select executes a select operation and returns the matching rows
func (c *Client) Select(ctx context.Context, table string, where []Condition, columns []string) ([]map[string]interface{}, error) {
        op := NewSelectOperation(table, where, columns)
        results, err := c.Transact(ctx, op)
        if err != nil {
                return nil, err
        }

        if len(results) == 0 {
                return nil, ErrNoResult
        }

        return results[0].Rows, nil
}

// getOpenVSwitchUUID retrieves the UUID of the Open_vSwitch table row
func (c *Client) getOpenVSwitchUUID(ctx context.Context) (string, error) {
        rows, err := c.Select(ctx, "Open_vSwitch", nil, []string{"_uuid"})
        if err != nil {
                return "", err
        }

        if len(rows) == 0 {
                return "", fmt.Errorf("no Open_vSwitch row found")
        }

        // Parse UUID from result
        uuidVal, ok := rows[0]["_uuid"]
        if !ok {
                return "", fmt.Errorf("_uuid field not found")
        }

        // UUID is returned as ["uuid", "actual-uuid-string"]
        uuidArr, ok := uuidVal.([]interface{})
        if !ok || len(uuidArr) != 2 {
                return "", fmt.Errorf("invalid UUID format: %v", uuidVal)
        }

        uuidStr, ok := uuidArr[1].(string)
        if !ok {
                return "", fmt.Errorf("invalid UUID string: %v", uuidArr[1])
        }

        return uuidStr, nil
}

// getOpenFlowClient returns an OpenFlow client for the specified bridge.
// Creates and connects a new client if one doesn't exist.
func (c *Client) getOpenFlowClient(ctx context.Context, bridgeName string) (*openflow.Client, error) {
        c.ofClientsMu.RLock()
        if ofClient, ok := c.ofClients[bridgeName]; ok {
                if ofClient.IsConnected() {
                        c.ofClientsMu.RUnlock()
                        return ofClient, nil
                }
        }
        c.ofClientsMu.RUnlock()

        // Need to create/reconnect
        c.ofClientsMu.Lock()
        defer c.ofClientsMu.Unlock()

        // Double-check after acquiring write lock
        if ofClient, ok := c.ofClients[bridgeName]; ok {
                if ofClient.IsConnected() {
                        return ofClient, nil
                }
                // Disconnect stale client
                ofClient.Close()
        }

        // Create new OpenFlow client
        ofClient, err := openflow.NewClient(bridgeName,
                openflow.WithSocketDir(c.ofSocketDir),
                openflow.WithTimeout(c.timeout),
        )
        if err != nil {
                return nil, fmt.Errorf("failed to create OpenFlow client: %w", err)
        }

        if err := ofClient.Connect(ctx); err != nil {
                return nil, fmt.Errorf("failed to connect OpenFlow client: %w", err)
        }

        c.ofClients[bridgeName] = ofClient
        return ofClient, nil
}

// closeOpenFlowClient closes the OpenFlow client for a specific bridge
func (c *Client) closeOpenFlowClient(bridgeName string) {
        c.ofClientsMu.Lock()
        defer c.ofClientsMu.Unlock()

        if ofClient, ok := c.ofClients[bridgeName]; ok {
                ofClient.Close()
                delete(c.ofClients, bridgeName)
        }
}

// AddFlow adds an OpenFlow rule to a bridge using native OpenFlow protocol
func (c *Client) AddFlow(ctx context.Context, bridgeName string, flow *FlowMod) error {
        if bridgeName == "" {
                return ErrInvalidBridgeName
        }
        if flow == nil {
                return ErrInvalidFlow
        }

        if !c.IsConnected() {
                return ErrNotConnected
        }

        // Get or create OpenFlow client for this bridge
        ofClient, err := c.getOpenFlowClient(ctx, bridgeName)
        if err != nil {
                return fmt.Errorf("failed to get OpenFlow client: %w", err)
        }

        // Convert OVS FlowMod to OpenFlow FlowMod
        ofFlow, err := convertToOpenFlow(flow)
        if err != nil {
                return fmt.Errorf("failed to convert flow: %w", err)
        }

        return ofClient.AddFlow(ctx, ofFlow)
}

// DeleteFlow deletes an OpenFlow rule from a bridge using native OpenFlow protocol
func (c *Client) DeleteFlow(ctx context.Context, bridgeName string, match string) error {
        if bridgeName == "" {
                return ErrInvalidBridgeName
        }
        if match == "" {
                return ErrInvalidFlow
        }

        if !c.IsConnected() {
                return ErrNotConnected
        }

        // Get or create OpenFlow client for this bridge
        ofClient, err := c.getOpenFlowClient(ctx, bridgeName)
        if err != nil {
                return fmt.Errorf("failed to get OpenFlow client: %w", err)
        }

        // Build OpenFlow FlowMod for delete
        ofFlow := openflow.NewFlowMod(0)
        ofFlow.Command = openflow.OFPFCDelete
        ofFlow.TableID = openflow.OFPTTAll
        ofFlow.OutPort = openflow.OFPPAny
        ofFlow.OutGroup = 0xffffffff

        // Parse match string and add to flow
        if err := parseMatchString(ofFlow.Match, match); err != nil {
                return fmt.Errorf("failed to parse match: %w", err)
        }

        return ofClient.DeleteFlow(ctx, ofFlow)
}

// ListFlows returns all OpenFlow rules on a bridge using native OpenFlow protocol
func (c *Client) ListFlows(ctx context.Context, bridgeName string) ([]*Flow, error) {
        if bridgeName == "" {

        }

// GetFlowStats returns statistics for flows matching the criteria
func (c *Client) GetFlowStats(ctx context.Context, bridgeName string, match string) ([]*Flow, error) {
        if bridgeName == "" {
                return nil, ErrInvalidBridgeName
        }

        if !c.IsConnected() {

                }

        if err != nil {

                        ByteCount:   int64(stat.ByteCount),
                }

// convertToOpenFlow converts an OVS FlowMod to an OpenFlow FlowMod
func convertToOpenFlow(flow *FlowMod) (*openflow.FlowMod, error) {
        ofFlow := openflow.NewFlowMod(0)
        ofFlow.TableID = uint8(flow.Table)
        ofFlow.Priority = uint16(flow.Priority)
        ofFlow.IdleTimeout = uint16(flow.IdleTimeout)
        ofFlow.HardTimeout = uint16(flow.HardTimeout)

        // Parse cookie
        if flow.Cookie != "" {
                cookie, err := parseCookie(flow.Cookie)
                if err == nil {
                        ofFlow.Cookie = cookie
                }
        }

        // Parse match string
        if flow.Match != "" {
                if err := parseMatchString(ofFlow.Match, flow.Match); err != nil {
                        return nil, fmt.Errorf("failed to parse match: %w", err)
                }
        }

        // Parse actions string
        if flow.Actions != "" {
                actions, err := parseActionsString(flow.Actions)
                if err != nil {
                        return nil, fmt.Errorf("failed to parse actions: %w", err)
                }
                if len(actions) > 0 {
                        ofFlow.Instructions = append(ofFlow.Instructions,
                                openflow.NewInstructionApplyActions(actions...))
                }
        }

        return ofFlow, nil
}

// parseMatchString parses ovs-ofctl style match string into OpenFlow match
func parseMatchString(match *openflow.Match, matchStr string) error {
        // Parse key=value pairs from match string
        parts := strings.Split(matchStr, ",")
        for _, part := range parts {
                part = strings.TrimSpace(part)
                if part == "" {
                        continue
                }

                kv := strings.SplitN(part, "=", 2)
                if len(kv) != 2 {
                        continue
                }

                key := strings.TrimSpace(kv[0])
                value := strings.TrimSpace(kv[1])

                switch key {
                case "in_port":
                        port, err := strconv.ParseUint(value, 10, 32)
                        if err != nil {
                                return fmt.Errorf("invalid in_port: %s", value)
                        }
                        match.AddInPort(uint32(port))

                case "dl_src":
                        mac, err := net.ParseMAC(value)
                        if err != nil {
                                return fmt.Errorf("invalid dl_src: %s", value)
                        }
                        match.AddEthSrc(mac)

                case "dl_dst":
                        mac, err := net.ParseMAC(value)
                        if err != nil {
                                return fmt.Errorf("invalid dl_dst: %s", value)
                        }
                        match.AddEthDst(mac)

                case "dl_type":
                        ethType, err := parseHexOrDecimal(value)
                        if err != nil {
                                return fmt.Errorf("invalid dl_type: %s", value)
                        }
                        match.AddEthType(uint16(ethType))

                case "dl_vlan":
                        vlan, err := strconv.ParseUint(value, 10, 16)
                        if err != nil {
                                return fmt.Errorf("invalid dl_vlan: %s", value)
                        }
                        match.AddVLANVid(uint16(vlan))

                case "nw_src":
                        ip := net.ParseIP(value)
                        if ip == nil {
                                return fmt.Errorf("invalid nw_src: %s", value)
                        }
                        match.AddIPv4Src(ip)

                case "nw_dst":
                        ip := net.ParseIP(value)
                        if ip == nil {
                                return fmt.Errorf("invalid nw_dst: %s", value)
                        }
                        match.AddIPv4Dst(ip)

                case "nw_proto":
                        proto, err := strconv.ParseUint(value, 10, 8)
                        if err != nil {
                                return fmt.Errorf("invalid nw_proto: %s", value)
                        }
                        match.AddIPProto(uint8(proto))

                case "tp_src":
                        port, err := strconv.ParseUint(value, 10, 16)
                        if err != nil {
                                return fmt.Errorf("invalid tp_src: %s", value)
                        }
                        match.AddTCPSrc(uint16(port))

                case "tp_dst":
                        port, err := strconv.ParseUint(value, 10, 16)
                        if err != nil {
                                return fmt.Errorf("invalid tp_dst: %s", value)
                        }
                        match.AddTCPDst(uint16(port))
                }
        }

        return nil
}

// parseActionsString parses ovs-ofctl style actions string into OpenFlow actions
func parseActionsString(actionsStr string) ([]openflow.Action, error) {
        var actions []openflow.Action

        parts := strings.Split(actionsStr, ",")
        for _, part := range parts {
                part = strings.TrimSpace(part)
                if part == "" {
                        continue
                }

                switch {
                case part == "drop":
                        // DROP means no actions
                        return []openflow.Action{}, nil

                case part == "NORMAL":
                        actions = append(actions, openflow.NewActionOutput(openflow.OFPPNormal, 0))

                case part == "FLOOD":
                        actions = append(actions, openflow.NewActionOutput(openflow.OFPPFlood, 0))

                case part == "ALL":
                        actions = append(actions, openflow.NewActionOutput(openflow.OFPPAll, 0))

                case part == "CONTROLLER":
                        actions = append(actions, openflow.NewActionOutput(openflow.OFPPController, 65535))

                case part == "LOCAL":
                        actions = append(actions, openflow.NewActionOutput(openflow.OFPPLocal, 0))

                case part == "IN_PORT":
                        actions = append(actions, openflow.NewActionOutput(openflow.OFPPInPort, 0))

                case strings.HasPrefix(part, "output:"):
                        portStr := strings.TrimPrefix(part, "output:")
                        port, err := strconv.ParseUint(portStr, 10, 32)
                        if err != nil {
                                return nil, fmt.Errorf("invalid output port: %s", portStr)
                        }
                        actions = append(actions, openflow.NewActionOutput(uint32(port), 0))

                case strings.HasPrefix(part, "mod_vlan_vid:"):
                        vlanStr := strings.TrimPrefix(part, "mod_vlan_vid:")
                        vlan, err := strconv.ParseUint(vlanStr, 10, 16)
                        if err != nil {
                                return nil, fmt.Errorf("invalid VLAN ID: %s", vlanStr)
                        }
                        // Push VLAN if not already present, then set VID
                        actions = append(actions, openflow.NewActionPushVLAN(0x8100))
                        value := make([]byte, 2)
                        openflow.ByteOrder.PutUint16(value, uint16(vlan)|0x1000) // Set OFPVID_PRESENT
                        actions = append(actions, openflow.NewActionSetField(
                                openflow.OFPXMCOpenflowBasic, openflow.OFPXMTOFBVlanVid, value))

                case part == "strip_vlan":
                        actions = append(actions, openflow.NewActionPopVLAN())

                case strings.HasPrefix(part, "mod_dl_src:"):
                        macStr := strings.TrimPrefix(part, "mod_dl_src:")
                        mac, err := net.ParseMAC(macStr)
                        if err != nil {
                                return nil, fmt.Errorf("invalid MAC: %s", macStr)
                        }
                        actions = append(actions, openflow.NewActionSetField(
                                openflow.OFPXMCOpenflowBasic, openflow.OFPXMTOFBEthSrc, mac))

                case strings.HasPrefix(part, "mod_dl_dst:"):
                        macStr := strings.TrimPrefix(part, "mod_dl_dst:")
                        mac, err := net.ParseMAC(macStr)
                        if err != nil {
                                return nil, fmt.Errorf("invalid MAC: %s", macStr)
                        }
                        actions = append(actions, openflow.NewActionSetField(
                                openflow.OFPXMCOpenflowBasic, openflow.OFPXMTOFBEthDst, mac))

                case strings.HasPrefix(part, "mod_nw_src:"):
                        ipStr := strings.TrimPrefix(part, "mod_nw_src:")
                        ip := net.ParseIP(ipStr)
                        if ip == nil {
                                return nil, fmt.Errorf("invalid IP: %s", ipStr)
                        }
                        actions = append(actions, openflow.NewActionSetField(
                                openflow.OFPXMCOpenflowBasic, openflow.OFPXMTOFBIPv4Src, ip.To4()))

                case strings.HasPrefix(part, "mod_nw_dst:"):
                        ipStr := strings.TrimPrefix(part, "mod_nw_dst:")
                        ip := net.ParseIP(ipStr)
                        if ip == nil {
                                return nil, fmt.Errorf("invalid IP: %s", ipStr)
                        }
                        actions = append(actions, openflow.NewActionSetField(
                                openflow.OFPXMCOpenflowBasic, openflow.OFPXMTOFBIPv4Dst, ip.To4()))
                }
        }

        return actions, nil
}

// parseCookie parses a cookie string (hex or decimal) into uint64
func parseCookie(s string) (uint64, error) {
        return parseHexOrDecimal(s)
}

// parseHexOrDecimal parses a string as hex (if prefixed with 0x) or decimal
func parseHexOrDecimal(s string) (uint64, error) {
        s = strings.TrimSpace(s)
        if strings.HasPrefix(s, "0x") || strings.HasPrefix(s, "0X") {
                return strconv.ParseUint(s[2:], 16, 64)
        }
        return strconv.ParseUint(s, 10, 64)
}

// AddPort adds a port to a bridge
func (c *Client) AddPort(ctx context.Context, bridgeName, portName string) (*Port, error) {
        if bridgeName == "" {
                return nil, ErrInvalidBridgeName
        }
        if portName == "" {
                return nil, ErrInvalidPortName
        }

        // Check if bridge exists
        bridge, err := c.GetBridge(ctx, bridgeName)
        if err != nil {
                return nil, err
        }

        // Check if port already exists
        existing, _ := c.GetPort(ctx, portName)
        if existing != nil {
                return nil, ErrPortExists
        }

        // Create interface row
        interfaceRow := map[string]interface{}{
                "name": portName,
                "type": "",
        }
        interfaceInsert := NewInsertOperation(interfaceTable, interfaceRow, "new_interface")

        // Create port row with interface reference
        portRow := map[string]interface{}{
                "name":       portName,
                "interfaces": NamedUUID{NamedUUID: "new_interface"},
        }
        portInsert := NewInsertOperation(portTable, portRow, "new_port")

        // Mutate bridge to add port reference
        bridgeMutate := NewMutateOperation(
                bridgeTable,
                []Condition{NewCondition("_uuid", "==", []interface{}{"uuid", bridge.UUID})},
                []Mutation{NewMutation("ports", "insert", NamedUUID{NamedUUID: "new_port"})},
        )

        // Execute transaction
        results, err := c.Transact(ctx, interfaceInsert, portInsert, bridgeMutate)
        if err != nil {
                return nil, fmt.Errorf("failed to add port: %w", err)
        }

        if len(results) < 2 || results[1].UUID == nil {
                return nil, ErrTransactionFailed
        }

        // Retrieve the created port
        return c.GetPort(ctx, portName)
}

// DeletePort removes a port from a bridge
func (c *Client) DeletePort(ctx context.Context, bridgeName, portName string) error {
        if bridgeName == "" {
                return ErrInvalidBridgeName
        }
        if portName == "" {
                return ErrInvalidPortName
        }

        // Check if bridge exists
        bridge, err := c.GetBridge(ctx, bridgeName)
        if err != nil {
                return err
        }

        // Check if port exists
        port, err := c.GetPort(ctx, portName)
        if err != nil {
                return err
        }

        // Remove port from bridge
        bridgeMutate := NewMutateOperation(
                bridgeTable,
                []Condition{NewCondition("_uuid", "==", []interface{}{"uuid", bridge.UUID})},
                []Mutation{NewMutation("ports", "delete", []interface{}{"uuid", port.UUID})},
        )

        // Delete port
        portDelete := NewDeleteOperation(
                portTable,
                []Condition{NewCondition("_uuid", "==", []interface{}{"uuid", port.UUID})},
        )

        // Delete associated interfaces
        var interfaceDeletes []Operation
        for _, ifaceUUID := range port.Interfaces {
                interfaceDeletes = append(interfaceDeletes, NewDeleteOperation(
                        interfaceTable,
                        []Condition{NewCondition("_uuid", "==", []interface{}{"uuid", ifaceUUID})},
                ))
        }

        // Execute transaction
        ops := []Operation{bridgeMutate, portDelete}
        ops = append(ops, interfaceDeletes...)
        _, err = c.Transact(ctx, ops...)
        if err != nil {
                return fmt.Errorf("failed to delete port: %w", err)
        }

        return nil
}

        bridge, err := c.GetBridge(ctx, bridgeName)
        if err != nil {
                return nil, err
        }

        ports := make([]*Port, 0, len(bridge.Ports))
        for _, portUUID := range bridge.Ports {

                rows, err := c.Select(ctx, portTable, []Condition{NewCondition("_uuid", "==", []interface{}{"uuid", portUUID})}, nil)

                        continue
                }
                port, err := parsePort(rows[0])

                        continue
                }
                ports = append(ports, port)
        }

        return ports, nil

// SetPortVLAN sets the VLAN tag for a port
func (c *Client) SetPortVLAN(ctx context.Context, portName string, tag int64) error {
        if tag < 0 || tag > 4095 {
                return fmt.Errorf("invalid VLAN tag: %d (must be 0-4095)", tag)
        }

        var value interface{}
        if tag == 0 {
                value = Set{Set: []interface{}{}}
        } else {
                value = tag
        }

        return c.UpdatePort(ctx, portName, map[string]interface{}{
                "tag": value,
        })
}

// SetPortTrunks sets the trunk VLANs for a port
func (c *Client) SetPortTrunks(ctx context.Context, portName string, trunks []int64) error {
        for _, vlan := range trunks {
                if vlan < 0 || vlan > 4095 {
                        return fmt.Errorf("invalid VLAN tag: %d (must be 0-4095)", vlan)
                }
        }

        value := Set{Set: make([]interface{}, len(trunks))}
        for i, vlan := range trunks {
                value.Set[i] = vlan
        }

        return c.UpdatePort(ctx, portName, map[string]interface{}{
                "trunks": value,
        })
}

// SetPortState sets the admin state of a port using OVSDB

                return ErrInvalidPortName
        }

        // Check if port exists
        _, err := c.GetPort(ctx, portName)
        if err != nil {
                return err
        }

        adminState := "up"
        if state == PortStateDown {
                adminState = "down"
        }

        // Set interface admin_state via OVSDB transaction
        updateOp := NewUpdateOperation(
                interfaceTable,
                []Condition{NewCondition("name", "==", portName)},
                map[string]interface{}{
                        "admin_state": adminState,
                },
        )

        _, err = c.Transact(ctx, updateOp)
        if err != nil {
                return fmt.Errorf("failed to set interface state: %w", err)
        }

        return nil
}

// parsePort parses a port row from OVSDB
func parsePort(row map[string]interface{}) (*Port, error) {
        port := &Port{
                ExternalIDs: make(map[string]string),
                OtherConfig: make(map[string]string),
                Statistics:  make(map[string]int64),
        }

        // Parse UUID
        if uuidVal, ok := row["_uuid"]; ok {
                if uuidArr, ok := uuidVal.([]interface{}); ok && len(uuidArr) == 2 {
                        if uuidStr, ok := uuidArr[1].(string); ok {
                                port.UUID = uuidStr
                        }
                }
        }

        // Parse name
        if name, ok := row["name"].(string); ok {
                port.Name = name
        }

        // Parse tag
        if tagVal, ok := row["tag"]; ok {
                switch v := tagVal.(type) {
                case float64:
                        port.Tag = int64(v)
                case int64:
                        port.Tag = v
                case []interface{}:
                        // Empty set: ["set", []]
                        if len(v) == 2 && v[0] == "set" {
                                port.Tag = 0
                        }
                }
        }

        // Parse interfaces
        port.Interfaces = parseUUIDSet(row["interfaces"])

        // Parse trunks
        if trunksVal, ok := row["trunks"]; ok {
                if arr, ok := trunksVal.([]interface{}); ok && len(arr) == 2 && arr[0] == "set" {
                        if items, ok := arr[1].([]interface{}); ok {
                                for _, item := range items {
                                        if num, ok := item.(float64); ok {
                                                port.Trunks = append(port.Trunks, int64(num))
                                        }
                                }
                        }
                }
        }

        // Parse external_ids
        port.ExternalIDs = parseMap(row["external_ids"])

        // Parse other_config
        port.OtherConfig = parseMap(row["other_config"])

        return port, nil
}

// parseInterface parses an interface row from OVSDB
func parseInterface(row map[string]interface{}) (*Interface, error) {
        iface := &Interface{
                ExternalIDs: make(map[string]string),
                OtherConfig: make(map[string]string),
                Statistics:  make(map[string]int64),
        }

        // Parse UUID
        if uuidVal, ok := row["_uuid"]; ok {
                if uuidArr, ok := uuidVal.([]interface{}); ok && len(uuidArr) == 2 {
                        if uuidStr, ok := uuidArr[1].(string); ok {
                                iface.UUID = uuidStr
                        }
                }
        }

        // Parse name
        if name, ok := row["name"].(string); ok {
                iface.Name = name
        }

        // Parse type
        if typ, ok := row["type"].(string); ok {
                iface.Type = typ
        }

        // Parse MAC
        if mac, ok := row["mac_in_use"].(string); ok {
                iface.MAC = mac
        }

        // Parse MTU
        if mtuVal, ok := row["mtu"]; ok {
                switch v := mtuVal.(type) {
                case float64:
                        iface.MTU = int64(v)
                case int64:
                        iface.MTU = v
                }
        }

        // Parse admin_state
        if state, ok := row["admin_state"].(string); ok {
                iface.AdminState = state
        }

        // Parse link_state
        if state, ok := row["link_state"].(string); ok {
                iface.LinkState = state
        }

        // Parse external_ids
        iface.ExternalIDs = parseMap(row["external_ids"])

        // Parse other_config
        iface.OtherConfig = parseMap(row["other_config"])

        return iface, nil
}

// OVSDBValue represents a value in OVSDB format
type OVSDBValue interface{}

// MarshalSet marshals a slice into an OVSDB set
func MarshalSet(values []interface{}) OVSDBValue {
        if len(values) == 0 {
                return Set{Set: []interface{}{}}
        }
        if len(values) == 1 {
                return values[0]
        }
        return Set{Set: values}
}

// MarshalMap marshals a map into an OVSDB map
func MarshalMap(m map[string]string) OVSDBValue {
        if len(m) == 0 {
                return Map{Map: [][]interface{}{}}
        }
        pairs := make([][]interface{}, 0, len(m))
        for k, v := range m {
                pairs = append(pairs, []interface{}{k, v})
        }
        return Map{Map: pairs}
}

// UnmarshalSet unmarshals an OVSDB set
func UnmarshalSet(value interface{}) ([]interface{}, error) {
        switch v := value.(type) {
        case []interface{}:
                return v, nil
        case map[string]interface{}:
                if set, ok := v["set"].([]interface{}); ok {
                        return set, nil
                }
        }
        return nil, fmt.Errorf("cannot unmarshal set from %T", value)
}

// UnmarshalMap unmarshals an OVSDB map
func UnmarshalMap(value interface{}) (map[string]string, error) {
        result := make(map[string]string)

        switch v := value.(type) {
        case map[string]interface{}:
                if mapData, ok := v["map"].([]interface{}); ok {
                        for _, pair := range mapData {
                                if p, ok := pair.([]interface{}); ok && len(p) == 2 {
                                        if key, ok := p[0].(string); ok {
                                                if val, ok := p[1].(string); ok {
                                                        result[key] = val
                                                }
                                        }
                                }
                        }
                        return result, nil
                }
        }

        return result, nil
}

// UnmarshalUUID extracts a UUID from an OVSDB value
func UnmarshalUUID(value interface{}) (string, error) {
        switch v := value.(type) {
        case string:
                return v, nil
        case []interface{}:
                if len(v) == 2 && v[0] == "uuid" {
                        if uuid, ok := v[1].(string); ok {
                                return uuid, nil
                        }
                }
        case map[string]interface{}:
                if uuid, ok := v["uuid"].(string); ok {
                        return uuid, nil
                }
        }
        return "", ErrInvalidUUID
}

// UnmarshalJSON implements json.Unmarshaler for UUID
func (u *UUID) UnmarshalJSON(data []byte) error {
        // Try to unmarshal as array first ["uuid", "actual-uuid"]
        var arr []string
        if err := json.Unmarshal(data, &arr); err == nil {
                if len(arr) == 2 && arr[0] == "uuid" {
                        u.UUID = arr[1]
                        return nil
                }
        }

        // Try to unmarshal as simple object {"uuid": "actual-uuid"}
        type plainUUID UUID
        var plain plainUUID
        if err := json.Unmarshal(data, &plain); err == nil {
                *u = UUID(plain)
                return nil
        }

        return ErrInvalidUUID
}
