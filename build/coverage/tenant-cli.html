// NewCreateCommand creates the tenant create subcommand
func NewCreateCommand(getFlags func() *cli.CommonFlags) *cobra.Command {
        var (
                name             string
                displayName      string
                vlanStart        uint16
                vlanEnd          uint16
                maxVMs           int64
                maxContainers    int64
                maxVCPUs         int64
                maxMemoryGB      int64
                maxDiskGB        int64
                maxBridges       int64
                maxPorts         int64
                maxBandwidthMbps int64
                maxVolumesGB     int64
                maxSnapshots     int64
        )

        cmd := &cobra.Command{
                Use:   "create",
                Short: "Create a new tenant",
                Long: `Create a new tenant with the specified configuration and quotas.

The tenant will be created with a unique ID and provisioned with:
  - A dedicated VLAN range (auto-allocated or specified)
  - An OVS bridge for network isolation
  - Resource quotas for compute, network, and storage

Examples:
  # Create a basic tenant
  platformctl tenant create --name my-tenant

  # Create with display name
  platformctl tenant create --name my-tenant --display-name "My Tenant"

  # Create with specific VLAN range
  platformctl tenant create --name my-tenant --vlan-start 100 --vlan-end 199

  # Create with compute quotas
  platformctl tenant create --name my-tenant --max-vms 100 --max-vcpus 400 --max-memory-gb 512

  # Create with all quotas
  platformctl tenant create --name my-tenant \
    --max-vms 100 --max-containers 200 --max-vcpus 400 --max-memory-gb 512 --max-disk-gb 1000 \
    --max-bridges 10 --max-ports 1000 --max-bandwidth-mbps 10000 \
    --max-volumes-gb 5000 --max-snapshots 100`,
                RunE: func(cmd *cobra.Command, args []string) error {
                        if name == "" {
                                return ErrMissingTenantName{}
                        }

                        // Validate VLAN range if specified
                        if vlanStart > 0 || vlanEnd > 0 {
                                if vlanStart < 1 || vlanEnd > 4094 || vlanStart > vlanEnd {
                                        return ErrInvalidVLANRange{
                                                Start:  vlanStart,
                                                End:    vlanEnd,
                                                Reason: "VLAN range must be 1-4094 and start <= end",
                                        }
                                }
                        }

                        flags := getFlags()
                        return runTenantCreate(flags, &createTenantOptions{
                                Name:             name,
                                DisplayName:      displayName,
                                VLANStart:        vlanStart,
                                VLANEnd:          vlanEnd,
                                MaxVMs:           maxVMs,
                                MaxContainers:    maxContainers,
                                MaxVCPUs:         maxVCPUs,
                                MaxMemoryGB:      maxMemoryGB,
                                MaxDiskGB:        maxDiskGB,
                                MaxBridges:       maxBridges,
                                MaxPorts:         maxPorts,
                                MaxBandwidthMbps: maxBandwidthMbps,
                                MaxVolumesGB:     maxVolumesGB,
                                MaxSnapshots:     maxSnapshots,
                        })
                },
        }

        // Required flags
        cmd.Flags().StringVar(&name, "name", "", "unique tenant name (required)")

        // Optional flags
        cmd.Flags().StringVar(&displayName, "display-name", "", "human-readable display name")
        cmd.Flags().Uint16Var(&vlanStart, "vlan-start", 0, "start of VLAN range (auto-allocate if not set)")
        cmd.Flags().Uint16Var(&vlanEnd, "vlan-end", 0, "end of VLAN range (auto-allocate if not set)")

        // Compute quota flags
        cmd.Flags().Int64Var(&maxVMs, "max-vms", 0, "maximum number of VMs (0 = use default)")
        cmd.Flags().Int64Var(&maxContainers, "max-containers", 0, "maximum number of containers (0 = use default)")
        cmd.Flags().Int64Var(&maxVCPUs, "max-vcpus", 0, "maximum number of vCPUs (0 = use default)")
        cmd.Flags().Int64Var(&maxMemoryGB, "max-memory-gb", 0, "maximum memory in GB (0 = use default)")
        cmd.Flags().Int64Var(&maxDiskGB, "max-disk-gb", 0, "maximum disk space in GB (0 = use default)")

        // Network quota flags
        cmd.Flags().Int64Var(&maxBridges, "max-bridges", 0, "maximum number of bridges (0 = use default)")
        cmd.Flags().Int64Var(&maxPorts, "max-ports", 0, "maximum number of network ports (0 = use default)")
        cmd.Flags().Int64Var(&maxBandwidthMbps, "max-bandwidth-mbps", 0, "maximum bandwidth in Mbps (0 = use default)")

        // Storage quota flags
        cmd.Flags().Int64Var(&maxVolumesGB, "max-volumes-gb", 0, "maximum volume storage in GB (0 = use default)")
        cmd.Flags().Int64Var(&maxSnapshots, "max-snapshots", 0, "maximum number of snapshots (0 = use default)")

        return cmd
}

// formatTimestamp formats a protobuf timestamp for display
func formatTimestamp(ts *timestamppb.Timestamp) string {
        if ts == nil {
                return "N/A"
        }
        return ts.AsTime().UTC().Format(time.RFC3339)
}

// formatTime formats a time.Time for display
func formatTime(t time.Time) string {
        if t.IsZero() {
                return "N/A"
        }
        return t.UTC().Format(time.RFC3339)
}

// formatBool formats a boolean for display
func formatBool(b bool) string {
        if b {
                return "Yes"
        }
        return "No"
}

// formatVLANRange formats a VLAN range for display
func formatVLANRange(start, end uint16) string {
        if start == 0 && end == 0 {
                return "N/A"
        }
        return fmt.Sprintf("%d-%d", start, end)
}

// formatVLANSize formats the size of a VLAN range
func formatVLANSize(start, end uint16) string {
        if start == 0 && end == 0 {
                return "0"
        }
        size := int(end-start) + 1
        return fmt.Sprintf("%d", size)
}

// formatQuotaValue formats a quota value for display
func formatQuotaValue(value int64, suffix string) string {
        if value <= 0 {
                return "unlimited"
        }
        if suffix != "" {
                return fmt.Sprintf("%d %s", value, suffix)
        }
        return fmt.Sprintf("%d", value)
}

// formatUsageValue formats a usage value with its limit
func formatUsageValue(used, max int64, suffix string) string {
        usedStr := fmt.Sprintf("%d", used)
        maxStr := "unlimited"
        if max > 0 {
                maxStr = fmt.Sprintf("%d", max)
        }
        if suffix != "" {
                return fmt.Sprintf("%s / %s %s", usedStr, maxStr, suffix)
        }
        return fmt.Sprintf("%s / %s", usedStr, maxStr)
}

// formatPercentage formats a percentage value
func formatPercentage(used, max int64) string {
        if max <= 0 {
                return "N/A"
        }
        pct := float64(used) / float64(max) * 100
        return fmt.Sprintf("%.1f%%", pct)
}

// formatState formats a tenant state for display
func formatState(state tenant.State) string {
        return string(state)
}

// formatStateWithColor returns state with color hints
func formatStateWithColor(state tenant.State) string {
        switch state {
        case tenant.StateActive:
                return "active"
        case tenant.StateSuspended:
                return "suspended"
        case tenant.StateCreated:
                return "created"
        case tenant.StateDeleting:
                return "deleting"
        case tenant.StateDeleted:
                return "deleted"
        default:
                return string(state)
        }
}

// formatLabels formats a label map for display
func formatLabels(labels map[string]string) string {
        if len(labels) == 0 {
                return "none"
        }
        parts := make([]string, 0, len(labels))
        for k, v := range labels {
                parts = append(parts, fmt.Sprintf("%s=%s", k, v))
        }
        return strings.Join(parts, ", ")
}

// formatAnnotations formats an annotation map for display
func formatAnnotations(annotations map[string]string) string {
        return formatLabels(annotations)
}

// NewCommand creates the tenant root command for tenant management operations
func NewCommand(getFlags func() *cli.CommonFlags) *cobra.Command {
        cmd := &cobra.Command{
                Use:   "tenant",
                Short: "Tenant management operations",
                Long: `Manage tenants, quotas, VLANs, and network bridges.

The tenant subsystem provides comprehensive tenant management including:
  - Tenant lifecycle (create, suspend, resume, delete)
  - Quota management (compute, network, storage)
  - Usage monitoring
  - VLAN allocation information
  - OVS bridge status

Tenants provide isolated environments with:
  - Dedicated VLAN ranges for network isolation
  - Per-tenant OVS bridges for SDN
  - Resource quotas for compute, network, and storage
  - TPM-backed data encryption keys (DEK)

Examples:
  # List all tenants
  platformctl tenant list

  # Show tenant details
  platformctl tenant show my-tenant

  # Create a new tenant with quotas
  platformctl tenant create --name my-tenant --max-vms 100 --max-vcpus 400

  # Get tenant quotas
  platformctl tenant quotas get my-tenant

  # Update compute quota
  platformctl tenant quotas set my-tenant --max-vms 200 --max-vcpus 800

  # Get tenant usage
  platformctl tenant usage my-tenant

  # Get tenant VLAN info
  platformctl tenant vlan my-tenant

  # Get tenant bridge status
  platformctl tenant bridge my-tenant

  # Suspend a tenant
  platformctl tenant suspend my-tenant

  # Resume a suspended tenant
  platformctl tenant resume my-tenant`,
        }

        // Add subcommands
        cmd.AddCommand(NewListCommand(getFlags))
        cmd.AddCommand(NewShowCommand(getFlags))
        cmd.AddCommand(NewCreateCommand(getFlags))
        cmd.AddCommand(NewDeleteCommand(getFlags))
        cmd.AddCommand(NewSuspendCommand(getFlags))
        cmd.AddCommand(NewResumeCommand(getFlags))
        cmd.AddCommand(NewQuotasCommand(getFlags))
        cmd.AddCommand(NewUsageCommand(getFlags))
        cmd.AddCommand(NewVLANCommand(getFlags))
        cmd.AddCommand(NewBridgeCommand(getFlags))

        return cmd
}

// BaseCommandInterface defines the interface for BaseCommand to enable testing
type BaseCommandInterface interface {
        Client() IAMClientProvider
        Printer() *cli.Printer
        Context() context.Context
        Close() error
        Flags() *cli.CommonFlags
}

// IAMClientProvider provides access to the IAM client
type IAMClientProvider interface {
        IAM() iampb.IAMServiceClient
}

// baseCommandAdapter wraps cli.BaseCommand to implement BaseCommandInterface
type baseCommandAdapter struct {
        base *cli.BaseCommand
}

func (b *baseCommandAdapter) Client() IAMClientProvider {
        return &clientAdapter{client: b.base.Client()}
}

func (b *baseCommandAdapter) Printer() *cli.Printer {
        return b.base.Printer()
}

func (b *baseCommandAdapter) Context() context.Context {
        return b.base.Context()
}

func (b *baseCommandAdapter) Close() error {
        return b.base.Close()
}

func (b *baseCommandAdapter) Flags() *cli.CommonFlags {
        return b.base.Flags()
}

// clientAdapter wraps cli.Client to implement IAMClientProvider
type clientAdapter struct {
        client *cli.Client
}

func (c *clientAdapter) IAM() iampb.IAMServiceClient {
        return c.client.IAM()
}

// runTenantListWithClient runs tenant list with a testable interface
func runTenantListWithClient(base BaseCommandInterface, limit, offset int32) error {
        defer base.Close()

        req := &iampb.ListTenantsRequest{
                Limit:  limit,
                Offset: offset,
        }

        ctx := base.Context()
        resp, err := base.Client().IAM().ListTenants(ctx, req)
        if err != nil {
                return ErrOperationFailed{
                        Operation: "list",
                        TenantID:  "",
                        Err:       err,
                }
        }

        printer := base.Printer()
        flags := base.Flags()

        switch flags.OutputFormat {
        case cli.OutputFormatJSON, cli.OutputFormatYAML:
                summaries := make([]*TenantListItem, 0, len(resp.Tenants))
                for _, t := range resp.Tenants {
                        summaries = append(summaries, &TenantListItem{
                                ID:        t.Id,
                                Name:      t.Name,
                                CreatedAt: formatTimestamp(t.CreatedAt),
                        })
                }

                output := &TenantListResponse{
                        Tenants: summaries,
                        Total:   resp.Total,
                        Limit:   limit,
                        Offset:  offset,
                }

                return printer.Print(output)

        case cli.OutputFormatTable:
                if len(resp.Tenants) == 0 {
                        printer.PrintInfo("No tenants found")
                        return nil
                }

                table := cli.NewTableData("NAME", "ID", "CREATED AT")
                for _, t := range resp.Tenants {
                        table.AddRow(
                                t.Name,
                                truncateID(t.Id),
                                formatTimestamp(t.CreatedAt),
                        )
                }

                if err := printer.Print(*table); err != nil {
                        return err
                }

                fmt.Println()
                printer.PrintKeyValue("Total Count", fmt.Sprintf("%d", resp.Total))
                return nil

        default:
                return cli.ErrInvalidOutputFormat{Format: string(flags.OutputFormat)}
        }
}

// runTenantCreateWithClient runs tenant create with a testable interface
func runTenantCreateWithClient(base BaseCommandInterface, opts *createTenantOptions) error {
        defer base.Close()

        req := &iampb.CreateTenantRequest{
                Id:   "",
                Name: opts.Name,
        }

        ctx := base.Context()
        resp, err := base.Client().IAM().CreateTenant(ctx, req)
        if err != nil {
                return ErrOperationFailed{
                        Operation: "create",
                        TenantID:  opts.Name,
                        Err:       err,
                }
        }

        entity := tenant.FromProto(resp)
        if opts.DisplayName != "" {
                entity.DisplayName = opts.DisplayName
        } else {
                entity.DisplayName = opts.Name
        }

        if opts.VLANStart > 0 && opts.VLANEnd > 0 {
                entity.VLANRange = tenant.VLANRange{
                        Start: opts.VLANStart,
                        End:   opts.VLANEnd,
                }
        }

        if opts.MaxVMs > 0 {
                entity.ComputeQuota.MaxVMs = opts.MaxVMs
        }
        if opts.MaxContainers > 0 {
                entity.ComputeQuota.MaxContainers = opts.MaxContainers
        }
        if opts.MaxVCPUs > 0 {
                entity.ComputeQuota.MaxVCPUs = opts.MaxVCPUs
        }
        if opts.MaxMemoryGB > 0 {
                entity.ComputeQuota.MaxMemoryMB = opts.MaxMemoryGB * 1024
        }
        if opts.MaxDiskGB > 0 {
                entity.ComputeQuota.MaxDiskGB = opts.MaxDiskGB
        }
        if opts.MaxBridges > 0 {
                entity.NetworkQuota.MaxBridges = opts.MaxBridges
        }
        if opts.MaxPorts > 0 {
                entity.NetworkQuota.MaxPorts = opts.MaxPorts
        }
        if opts.MaxBandwidthMbps > 0 {
                entity.NetworkQuota.MaxBandwidthMbps = opts.MaxBandwidthMbps
        }
        if opts.MaxVolumesGB > 0 {
                entity.StorageQuota.MaxVolumesGB = opts.MaxVolumesGB
        }
        if opts.MaxSnapshots > 0 {
                entity.StorageQuota.MaxSnapshots = opts.MaxSnapshots
        }

        printer := base.Printer()
        flags := base.Flags()

        switch flags.OutputFormat {
        case cli.OutputFormatJSON, cli.OutputFormatYAML:
                return printer.Print(toTenantDetail(entity))

        case cli.OutputFormatTable:
                printer.PrintSuccess(fmt.Sprintf("Tenant created successfully: %s", entity.Name))
                fmt.Println()
                printTenantDetails(printer, entity)
                return nil

        default:
                return cli.ErrInvalidOutputFormat{Format: string(flags.OutputFormat)}
        }
}

        if foundTenant == nil {

        }

        entity := tenant.FromProto(foundTenant)

        if entity.OVSBridge == "" {

        }

        printer := base.Printer()
        flags := base.Flags()

        switch flags.OutputFormat {
        case cli.OutputFormatJSON, cli.OutputFormatYAML:

                fmt.Println()
                printer.PrintKeyValue("Bridge Name", entity.OVSBridge)
                printer.PrintKeyValue("State", formatState(entity.State))

                if entity.VLANRange.Start > 0 && entity.VLANRange.End > 0 {
                        fmt.Println()

                case tenant.StateCreated:
                        printer.PrintInfo("Bridge is created but not yet active")
                case tenant.StateDeleting:
                        printer.PrintWarning("Bridge is being deleted")
                default:
                        printer.PrintKeyValue("Bridge State", string(entity.State))
                }

                return nil

        default:
                return cli.ErrInvalidOutputFormat{Format: string(flags.OutputFormat)}
        }
}

// printComputeUsageTable prints compute usage as a table

        fmt.Println("Compute Usage:")

        table := cli.NewTableData("RESOURCE", "USED", "MAX", "UTILIZATION")

                formatQuotaLimit(quota.MaxVMs), formatPercentage(quota.UsedVMs, quota.MaxVMs))
        table.AddRow("Containers", fmt.Sprintf("%d", quota.UsedContainers),
                formatQuotaLimit(quota.MaxContainers), formatPercentage(quota.UsedContainers, quota.MaxContainers))
        table.AddRow("vCPUs", fmt.Sprintf("%d", quota.UsedVCPUs),
                formatQuotaLimit(quota.MaxVCPUs), formatPercentage(quota.UsedVCPUs, quota.MaxVCPUs))
        table.AddRow("Memory (MB)", fmt.Sprintf("%d", quota.UsedMemoryMB),

        table.AddRow("Disk (GB)", fmt.Sprintf("%d", quota.UsedDiskGB),
                formatQuotaLimit(quota.MaxDiskGB), formatPercentage(quota.UsedDiskGB, quota.MaxDiskGB))

        printer.Print(*table)
}

// formatQuotaLimit formats a quota limit value
func formatQuotaLimit(limit int64) string {
        if limit <= 0 {
                return "unlimited"
        }
        return fmt.Sprintf("%d", limit)
}
