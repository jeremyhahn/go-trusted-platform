// OrchestratorConfig holds configuration for the compute orchestrator
type OrchestratorConfig struct {
        // Type is the orchestrator type (vm, container)
        Type ComputeType `json:"type" yaml:"type"`

        // WorkDir is the base working directory for all instances
        WorkDir string `json:"work_dir" yaml:"work_dir"`

        // DefaultMemory is the default memory allocation in MB
        DefaultMemory int `json:"default_memory" yaml:"default_memory"`

        // DefaultCPUs is the default number of CPU cores
        DefaultCPUs int `json:"default_cpus" yaml:"default_cpus"`

        // DefaultDiskSize is the default disk size
        DefaultDiskSize string `json:"default_disk_size" yaml:"default_disk_size"`

        // MaxInstances is the maximum number of instances allowed
        MaxInstances int `json:"max_instances" yaml:"max_instances"`

        // EnableKVM enables KVM acceleration by default (VMs only)
        EnableKVM bool `json:"enable_kvm" yaml:"enable_kvm"`

        // NetworkBridge is the default network bridge name
        NetworkBridge string `json:"network_bridge" yaml:"network_bridge"`

        // OVSSocketPath is the path to the OVSDB socket
        OVSSocketPath string `json:"ovs_socket_path" yaml:"ovs_socket_path"`

        // QEMUBinary is the path to the QEMU binary (VMs only)
        QEMUBinary string `json:"qemu_binary,omitempty" yaml:"qemu_binary,omitempty"`

        // OVMFCode is the path to OVMF firmware code
        OVMFCode string `json:"ovmf_code,omitempty" yaml:"ovmf_code,omitempty"`

        // OVMFCodeSecure is the path to OVMF firmware with Secure Boot
        OVMFCodeSecure string `json:"ovmf_code_secure,omitempty" yaml:"ovmf_code_secure,omitempty"`

        // OVMFVars is the path to OVMF NVRAM template
        OVMFVars string `json:"ovmf_vars,omitempty" yaml:"ovmf_vars,omitempty"`

        // AutoCleanup enables automatic cleanup of stopped instances
        AutoCleanup bool `json:"auto_cleanup" yaml:"auto_cleanup"`

        // Debug enables debug logging
        Debug bool `json:"debug" yaml:"debug"`
}

// DefaultOrchestratorConfig returns a configuration with sensible defaults
func DefaultOrchestratorConfig(computeType ComputeType) *OrchestratorConfig {
        config := &OrchestratorConfig{
                Type:            computeType,
                WorkDir:         "/var/lib/trusted-platform/compute",
                DefaultMemory:   4096,
                DefaultCPUs:     runtime.NumCPU(),
                DefaultDiskSize: "20G",
                MaxInstances:    100,
                EnableKVM:       true,
                NetworkBridge:   "trusted0",
                OVSSocketPath:   "/var/run/openvswitch/db.sock",
                AutoCleanup:     false,
                Debug:           false,
        }

        // VM-specific defaults
        if computeType == ComputeTypeVM {
                config.QEMUBinary = "qemu-system-x86_64"
                config.OVMFCode = "/usr/share/OVMF/OVMF_CODE_4M.fd"
                config.OVMFCodeSecure = "/usr/share/OVMF/OVMF_CODE_4M.secboot.fd"
                config.OVMFVars = "/usr/share/OVMF/OVMF_VARS_4M.fd"
        }

        return config
}

// Validate validates the orchestrator configuration
func (c *OrchestratorConfig) Validate() error {
        if !c.Type.IsValid() {
                return fmt.Errorf("%w: invalid compute type: %s", ErrInvalidConfiguration, c.Type)
        }

        if c.WorkDir == "" {
                return fmt.Errorf("%w: work directory is required", ErrInvalidConfiguration)
        }

        if c.DefaultMemory <= 0 {
                return fmt.Errorf("%w: default memory must be greater than 0", ErrInvalidConfiguration)
        }

        if c.DefaultCPUs <= 0 {
                return fmt.Errorf("%w: default CPUs must be greater than 0", ErrInvalidConfiguration)
        }

        if c.DefaultDiskSize == "" {
                return fmt.Errorf("%w: default disk size is required", ErrInvalidConfiguration)
        }

        if c.MaxInstances <= 0 {
                return fmt.Errorf("%w: max instances must be greater than 0", ErrInvalidConfiguration)
        }

        // VM-specific validation
        if c.Type == ComputeTypeVM {
                if c.QEMUBinary == "" {
                        return fmt.Errorf("%w: QEMU binary path is required for VM orchestrator", ErrInvalidConfiguration)
                }
                if c.OVMFCode == "" {
                        return fmt.Errorf("%w: OVMF code path is required for VM orchestrator", ErrInvalidConfiguration)
                }
                if c.OVMFVars == "" {
                        return fmt.Errorf("%w: OVMF vars path is required for VM orchestrator", ErrInvalidConfiguration)
                }
        }

        return nil
}

// InstanceState represents the lifecycle state of a compute instance
type InstanceState string

const (
        // StateUnknown represents an unknown or uninitialized state
        StateUnknown InstanceState = "unknown"

        // StateCreating represents an instance being created
        StateCreating InstanceState = "creating"

        // StateStopped represents a stopped instance
        StateStopped InstanceState = "stopped"

        // StateStarting represents an instance in the process of starting
        StateStarting InstanceState = "starting"

        // StateRunning represents a running instance
        StateRunning InstanceState = "running"

        // StateStopping represents an instance in the process of stopping
        StateStopping InstanceState = "stopping"

        // StateError represents an instance in an error state
        StateError InstanceState = "error"

        // StateDeleting represents an instance being deleted
        StateDeleting InstanceState = "deleting"
)

// String returns the string representation of InstanceState
func (s InstanceState) String() string {
        return string(s)
}

// IsValid checks if the InstanceState is valid
func (s InstanceState) IsValid() bool {
        switch s {
        case StateUnknown, StateCreating, StateStopped, StateStarting,
                StateRunning, StateStopping, StateError, StateDeleting:
                return true
        default:
                return false
        }
}

// IsTerminal returns true if the state is terminal (no further transitions expected)
func (s InstanceState) IsTerminal() bool {
        return s == StateError || s == StateDeleting
}

// CanTransitionTo checks if the current state can transition to the target state
func (s InstanceState) CanTransitionTo(target InstanceState) bool {
        // Terminal states can't transition
        if s.IsTerminal() {
                return false
        }

        // Valid state transitions
        validTransitions := map[InstanceState][]InstanceState{
                StateUnknown:  {StateCreating, StateError},
                StateCreating: {StateStopped, StateError},
                StateStopped:  {StateStarting, StateDeleting, StateError},
                StateStarting: {StateRunning, StateError},
                StateRunning:  {StateStopping, StateError},
                StateStopping: {StateStopped, StateError},
                StateError:    {}, // Terminal state
                StateDeleting: {}, // Terminal state
        }

        allowed, exists := validTransitions[s]
        if !exists {
                return false
        }

        for _, allowedState := range allowed {
                if allowedState == target {
                        return true
                }
        }

        return false
}

// LifecycleManager manages instance state transitions
type LifecycleManager struct {
        mu         sync.RWMutex
        states     map[string]InstanceState
        timestamps map[string]time.Time
}

// NewLifecycleManager creates a new lifecycle manager
func NewLifecycleManager() *LifecycleManager {
        return &LifecycleManager{
                states:     make(map[string]InstanceState),
                timestamps: make(map[string]time.Time),
        }
}

// SetState sets the state for an instance
func (lm *LifecycleManager) SetState(instanceID string, state InstanceState) error {
        lm.mu.Lock()
        defer lm.mu.Unlock()

        if instanceID == "" {
                return ErrInvalidInstanceID
        }

        if !state.IsValid() {
                return fmt.Errorf("%w: %s", ErrInvalidState, state)
        }

        // Check if transition is valid
        currentState, exists := lm.states[instanceID]
        if exists && !currentState.CanTransitionTo(state) {
                return fmt.Errorf("%w: cannot transition from %s to %s", ErrInvalidState, currentState, state)
        }

        lm.states[instanceID] = state
        lm.timestamps[instanceID] = time.Now()

        return nil
}

// GetState returns the current state of an instance
func (lm *LifecycleManager) GetState(instanceID string) (InstanceState, error) {
        lm.mu.RLock()
        defer lm.mu.RUnlock()

        if instanceID == "" {
                return StateUnknown, ErrInvalidInstanceID
        }

        state, exists := lm.states[instanceID]
        if !exists {
                return StateUnknown, ErrInstanceNotFound
        }

        return state, nil
}

// GetStateTimestamp returns when the instance last changed state
func (lm *LifecycleManager) GetStateTimestamp(instanceID string) (time.Time, error) {
        lm.mu.RLock()
        defer lm.mu.RUnlock()

        if instanceID == "" {
                return time.Time{}, ErrInvalidInstanceID
        }

        ts, exists := lm.timestamps[instanceID]
        if !exists {
                return time.Time{}, ErrInstanceNotFound
        }

        return ts, nil
}

// DeleteInstance removes an instance from the lifecycle manager
func (lm *LifecycleManager) DeleteInstance(instanceID string) error {
        lm.mu.Lock()
        defer lm.mu.Unlock()

        if instanceID == "" {
                return ErrInvalidInstanceID
        }

        if _, exists := lm.states[instanceID]; !exists {
                return ErrInstanceNotFound
        }

        delete(lm.states, instanceID)
        delete(lm.timestamps, instanceID)

        return nil
}

// ListInstances returns all instance IDs and their states
func (lm *LifecycleManager) ListInstances() map[string]InstanceState {
        lm.mu.RLock()
        defer lm.mu.RUnlock()

        result := make(map[string]InstanceState, len(lm.states))
        for id, state := range lm.states {
                result[id] = state
        }

        return result
}

// WaitForState waits for an instance to reach a specific state or timeout
func (lm *LifecycleManager) WaitForState(instanceID string, targetState InstanceState, timeout time.Duration) error {
        if instanceID == "" {
                return ErrInvalidInstanceID
        }

        if !targetState.IsValid() {
                return fmt.Errorf("%w: %s", ErrInvalidState, targetState)
        }

        ticker := time.NewTicker(100 * time.Millisecond)
        defer ticker.Stop()

        timeoutCh := time.After(timeout)

        for {
                select {
                case <-ticker.C:
                        state, err := lm.GetState(instanceID)
                        if err != nil {
                                return err
                        }
                        if state == targetState {
                                return nil
                        }
                        if state == StateError {
                                return fmt.Errorf("%w: instance entered error state while waiting for %s", ErrInvalidState, targetState)
                        }
                case <-timeoutCh:
                        return ErrTimeout
                }
        }
}

// QEMUOrchestrator implements the Orchestrator interface for QEMU virtual machines
type QEMUOrchestrator struct {
        config    *OrchestratorConfig
        logger    *logging.Logger
        lifecycle *LifecycleManager
        ovsClient *ovs.Client
        mu        sync.RWMutex
        instances map[string]*qemuInstance
}

// qemuInstance holds the VM instance and its metadata
type qemuInstance struct {
        vm                *qemu.VM
        instance          *Instance
        ovsNetwork        *qemu.OVSNetwork
        mu                sync.Mutex
        cancelFunc        context.CancelFunc
        waitGroup         sync.WaitGroup
        networkAttached   bool
        primaryPortName   string
        primaryBridgeName string
}

// NewQEMUOrchestrator creates a new QEMU orchestrator
func NewQEMUOrchestrator(config *OrchestratorConfig, logger *logging.Logger) (*QEMUOrchestrator, error) {
        if config == nil {
                return nil, fmt.Errorf("%w: config is required", ErrInvalidConfiguration)
        }

        if logger == nil {
                logger = logging.DefaultLogger()
        }

        // Validate configuration
        if err := config.Validate(); err != nil {
                return nil, err
        }

        if config.Type != ComputeTypeVM {
                return nil, fmt.Errorf("%w: QEMU orchestrator requires VM compute type", ErrInvalidConfiguration)
        }

        // Create work directory
        if err := os.MkdirAll(config.WorkDir, 0755); err != nil {
                return nil, fmt.Errorf("failed to create work directory: %w", err)
        }

        // Create OVS client
        ovsClient, err := ovs.NewClient(config.OVSSocketPath)
        if err != nil {
                logger.Warnf("Failed to create OVS client: %v (OVS networking will be unavailable)", err)
        }

        o := &QEMUOrchestrator{
                config:    config,
                logger:    logger,
                lifecycle: NewLifecycleManager(),
                ovsClient: ovsClient,
                instances: make(map[string]*qemuInstance),
        }

        // Connect to OVS if available
        if o.ovsClient != nil {
                ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
                defer cancel()
                if err := o.ovsClient.Connect(ctx); err != nil {
                        logger.Warnf("Failed to connect to OVS: %v (OVS networking will be unavailable)", err)
                        o.ovsClient = nil
                }
        }

        return o, nil
}

// generateInstanceID generates a unique instance ID
func generateInstanceID() string {
        b := make([]byte, 16)
        rand.Read(b)
        return hex.EncodeToString(b)
}

// Create creates a new VM instance from a device profile
func (o *QEMUOrchestrator) Create(ctx context.Context, profile *entities.DeviceProfile) (*Instance, error) {
        if profile == nil {
                return nil, ErrInvalidProfile
        }

        o.mu.Lock()
        defer o.mu.Unlock()

        // Check instance limit
        if len(o.instances) >= o.config.MaxInstances {
                return nil, ErrResourceExhausted
        }

        // Generate unique instance ID
        instanceID := generateInstanceID()

        // Check if already exists
        if _, exists := o.instances[instanceID]; exists {
                return nil, ErrInstanceAlreadyExists
        }

        // Set initial state
        if err := o.lifecycle.SetState(instanceID, StateCreating); err != nil {
                return nil, err
        }

        // Create instance configuration
        instanceConfig := &InstanceConfig{
                Memory:      o.config.DefaultMemory,
                CPUs:        o.config.DefaultCPUs,
                DiskSize:    o.config.DefaultDiskSize,
                WorkDir:     filepath.Join(o.config.WorkDir, instanceID),
                SecureBoot:  false,
                EnableKVM:   o.config.EnableKVM,
                TPMEnabled:  true,
                NetworkMode: "bridge",
                BootMode:    "boot",
        }

        // Override with profile-specific settings if available
        if profile.PlatformdConfig != nil {
                // Profile configuration can override defaults
                // This would be implemented based on actual profile structure
        }

        // Create disk path
        instanceConfig.DiskPath = filepath.Join(instanceConfig.WorkDir, "disk.qcow2")

        // Create work directory
        if err := os.MkdirAll(instanceConfig.WorkDir, 0755); err != nil {
                o.lifecycle.SetState(instanceID, StateError)
                return nil, fmt.Errorf("failed to create instance work directory: %w", err)
        }

        // Create QEMU configuration
        qemuConfig := &qemu.Config{
                QEMUBinary:     o.config.QEMUBinary,
                DiskPath:       instanceConfig.DiskPath,
                DiskSize:       instanceConfig.DiskSize,
                Memory:         instanceConfig.Memory,
                CPUs:           instanceConfig.CPUs,
                WorkDir:        instanceConfig.WorkDir,
                SecureBoot:     instanceConfig.SecureBoot,
                Debug:          o.config.Debug,
                OVMFCode:       o.config.OVMFCode,
                OVMFCodeSecure: o.config.OVMFCodeSecure,
                OVMFVars:       o.config.OVMFVars,
                EnableKVM:      instanceConfig.EnableKVM,
                SerialConsole:  false,
                NoGraphic:      true,
                BootOrder:      "c",
                NetworkMode:    instanceConfig.NetworkMode,
                BridgeName:     o.config.NetworkBridge,
                MACAddress:     instanceConfig.MACAddress,
        }

        // Create VM
        vm, err := qemu.NewVM(qemuConfig, o.logger)
        if err != nil {
                o.lifecycle.SetState(instanceID, StateError)
                return nil, fmt.Errorf("failed to create VM: %w", err)
        }

        // Create disk
        if err := vm.CreateDisk(); err != nil {
                o.lifecycle.SetState(instanceID, StateError)
                return nil, fmt.Errorf("failed to create VM disk: %w", err)
        }

        // Create instance metadata
        instance := &Instance{
                ID:                 instanceID,
                Name:               profile.Name,
                Type:               ComputeTypeVM,
                State:              StateCreating,
                Profile:            profile,
                Config:             instanceConfig,
                NetworkAttachments: []NetworkAttachment{},
                CreatedAt:          time.Now(),
                UpdatedAt:          time.Now(),
                Metadata:           make(map[string]string),
        }

        // Store instance
        o.instances[instanceID] = &qemuInstance{
                vm:       vm,
                instance: instance,
        }

        // Update state to stopped
        if err := o.lifecycle.SetState(instanceID, StateStopped); err != nil {
                delete(o.instances, instanceID)
                return nil, err
        }
        instance.State = StateStopped
        instance.UpdatedAt = time.Now()

        o.logger.Infof("Created VM instance: %s (profile: %s)", instanceID, profile.Name)

        return instance, nil
}

// Start starts a stopped VM instance
func (o *QEMUOrchestrator) Start(ctx context.Context, instanceID string) error {
        if instanceID == "" {
                return ErrInvalidInstanceID
        }

        o.mu.RLock()
        qemuInst, exists := o.instances[instanceID]
        o.mu.RUnlock()

        if !exists {
                return ErrInstanceNotFound
        }

        qemuInst.mu.Lock()
        defer qemuInst.mu.Unlock()

        // Check current state
        currentState, err := o.lifecycle.GetState(instanceID)
        if err != nil {
                return err
        }

        if currentState == StateRunning {
                return ErrInstanceAlreadyRunning
        }

        if currentState != StateStopped {
                return fmt.Errorf("%w: cannot start instance in state %s", ErrInvalidState, currentState)
        }

        // Update state to starting
        if err := o.lifecycle.SetState(instanceID, StateStarting); err != nil {
                return err
        }
        qemuInst.instance.State = StateStarting
        qemuInst.instance.UpdatedAt = time.Now()

        // Determine boot mode
        bootMode := qemuInst.instance.Config.BootMode
        if bootMode == "" {
                bootMode = "boot"
        }

        // Create context for VM lifecycle
        vmCtx, cancel := context.WithCancel(context.Background())
        qemuInst.cancelFunc = cancel

        // Start VM
        if err := qemuInst.vm.Start(vmCtx, bootMode); err != nil {
                o.lifecycle.SetState(instanceID, StateError)
                qemuInst.instance.State = StateError
                qemuInst.instance.UpdatedAt = time.Now()
                return fmt.Errorf("failed to start VM: %w", err)
        }

        // Update instance with PID
        qemuInst.instance.PID = qemuInst.vm.GetPID()

        // Update state to running
        if err := o.lifecycle.SetState(instanceID, StateRunning); err != nil {
                // VM is running but state update failed - try to stop it
                qemuInst.vm.Stop()
                return err
        }
        qemuInst.instance.State = StateRunning
        qemuInst.instance.UpdatedAt = time.Now()

        // Start wait goroutine
        qemuInst.waitGroup.Add(1)
        go func() {
                defer qemuInst.waitGroup.Done()
                if err := qemuInst.vm.Wait(); err != nil {
                        o.logger.Errorf("VM %s exited with error: %v", instanceID, err)
                }
                // Update state to stopped
                o.lifecycle.SetState(instanceID, StateStopped)
                qemuInst.mu.Lock()
                qemuInst.instance.State = StateStopped
                qemuInst.instance.UpdatedAt = time.Now()
                qemuInst.instance.PID = 0
                qemuInst.mu.Unlock()
                o.logger.Infof("VM %s stopped", instanceID)
        }()

        o.logger.Infof("Started VM instance: %s (PID: %d)", instanceID, qemuInst.instance.PID)

        return nil
}

// Stop stops a running VM instance
func (o *QEMUOrchestrator) Stop(ctx context.Context, instanceID string) error {
        if instanceID == "" {
                return ErrInvalidInstanceID
        }

        o.mu.RLock()
        qemuInst, exists := o.instances[instanceID]
        o.mu.RUnlock()

        if !exists {
                return ErrInstanceNotFound
        }

        qemuInst.mu.Lock()
        defer qemuInst.mu.Unlock()

        // Check current state
        currentState, err := o.lifecycle.GetState(instanceID)
        if err != nil {
                return err
        }

        if currentState != StateRunning {
                return ErrInstanceNotRunning
        }

        // Update state to stopping
        if err := o.lifecycle.SetState(instanceID, StateStopping); err != nil {
                return err
        }
        qemuInst.instance.State = StateStopping
        qemuInst.instance.UpdatedAt = time.Now()

        // Stop VM
        if err := qemuInst.vm.Stop(); err != nil {
                o.lifecycle.SetState(instanceID, StateError)
                qemuInst.instance.State = StateError
                qemuInst.instance.UpdatedAt = time.Now()
                return fmt.Errorf("failed to stop VM: %w", err)
        }

        // Cancel context
        if qemuInst.cancelFunc != nil {
                qemuInst.cancelFunc()
        }

        // Wait for VM to exit
        qemuInst.waitGroup.Wait()

        o.logger.Infof("Stopped VM instance: %s", instanceID)

        return nil
}

// Delete deletes a VM instance and cleans up resources
func (o *QEMUOrchestrator) Delete(ctx context.Context, instanceID string) error {
        if instanceID == "" {
                return ErrInvalidInstanceID
        }

        o.mu.Lock()
        defer o.mu.Unlock()

        qemuInst, exists := o.instances[instanceID]
        if !exists {
                return ErrInstanceNotFound
        }

        qemuInst.mu.Lock()
        defer qemuInst.mu.Unlock()

        // Check current state
        currentState, err := o.lifecycle.GetState(instanceID)
        if err != nil {
                return err
        }

        // Stop if running
        if currentState == StateRunning {
                qemuInst.mu.Unlock()
                o.mu.Unlock()
                if err := o.Stop(ctx, instanceID); err != nil {
                        o.mu.Lock()
                        qemuInst.mu.Lock()
                        return err
                }
                o.mu.Lock()
                qemuInst.mu.Lock()
        }

        // Update state to deleting
        if err := o.lifecycle.SetState(instanceID, StateDeleting); err != nil {
                return err
        }
        qemuInst.instance.State = StateDeleting
        qemuInst.instance.UpdatedAt = time.Now()

        // Cleanup OVS network if attached
        if qemuInst.ovsNetwork != nil {
                if err := qemuInst.ovsNetwork.Cleanup(ctx); err != nil {
                        o.logger.Warnf("Failed to cleanup OVS network for instance %s: %v", instanceID, err)
                }
        }

        // Remove work directory
        if err := os.RemoveAll(qemuInst.instance.Config.WorkDir); err != nil {
                o.logger.Warnf("Failed to remove work directory for instance %s: %v", instanceID, err)
        }

        // Remove from lifecycle manager
        if err := o.lifecycle.DeleteInstance(instanceID); err != nil {
                o.logger.Warnf("Failed to remove instance from lifecycle manager: %v", err)
        }

        // Remove from instances map
        delete(o.instances, instanceID)

        o.logger.Infof("Deleted VM instance: %s", instanceID)

        return nil
}

// List returns all VM instances, optionally filtered
func (o *QEMUOrchestrator) List(ctx context.Context, filter *InstanceFilter) ([]*Instance, error) {
        o.mu.RLock()
        defer o.mu.RUnlock()

        result := make([]*Instance, 0, len(o.instances))

        for _, qemuInst := range o.instances {
                qemuInst.mu.Lock()
                instance := qemuInst.instance
                qemuInst.mu.Unlock()

                // Apply filter if provided
                if filter != nil && !filter.Matches(instance) {
                        continue
                }

                // Create a copy to avoid external modifications
                instanceCopy := *instance
                result = append(result, &instanceCopy)
        }

        return result, nil
}

// Get retrieves a specific VM instance by ID
func (o *QEMUOrchestrator) Get(ctx context.Context, instanceID string) (*Instance, error) {
        if instanceID == "" {
                return nil, ErrInvalidInstanceID
        }

        o.mu.RLock()
        qemuInst, exists := o.instances[instanceID]
        o.mu.RUnlock()

        if !exists {
                return nil, ErrInstanceNotFound
        }

        qemuInst.mu.Lock()
        defer qemuInst.mu.Unlock()

        // Return a copy
        instanceCopy := *qemuInst.instance
        return &instanceCopy, nil
}

// AttachNetwork attaches a VM instance to an OVS network port
func (o *QEMUOrchestrator) AttachNetwork(ctx context.Context, instanceID, portName string) error {
        if instanceID == "" {
                return ErrInvalidInstanceID
        }

        if portName == "" {
                return fmt.Errorf("%w: port name is required", ErrNetworkAttachmentFailed)
        }


        o.mu.RLock()
        qemuInst, exists := o.instances[instanceID]
        o.mu.RUnlock()

        if !exists {
                return ErrInstanceNotFound
        }

        if o.ovsClient == nil {
                return fmt.Errorf("%w: OVS client not available", ErrNetworkAttachmentFailed)
        }

        qemuInst.mu.Lock()
        defer qemuInst.mu.Unlock()

        // Check if already attached
        if qemuInst.networkAttached {
                return fmt.Errorf("%w: network already attached", ErrNetworkAttachmentFailed)
        }

        // Get or create OVS network
        vmConfig := qemuInst.vm.GetConfig()
        if vmConfig.OVS == nil {
                vmConfig.OVS = &qemu.OVSConfig{
                        Enable:     true,
                        BridgeName: o.config.NetworkBridge,
                        SocketPath: o.config.OVSSocketPath,
                        PortName:   portName,
                }
        }

        ovsNetwork, err := qemu.NewOVSNetwork(vmConfig, o.logger)
        if err != nil {
                return fmt.Errorf("%w: %v", ErrNetworkAttachmentFailed, err)
        }

        // Setup network (creates TAP and attaches to OVS)
        if err := ovsNetwork.Setup(ctx); err != nil {
                return fmt.Errorf("%w: %v", ErrNetworkAttachmentFailed, err)
        }

        // Update instance
        qemuInst.ovsNetwork = ovsNetwork
        qemuInst.networkAttached = true
        qemuInst.primaryPortName = portName
        qemuInst.primaryBridgeName = o.config.NetworkBridge

        // Add network attachment to instance metadata
        attachment := NetworkAttachment{
                PortName:   portName,
                BridgeName: o.config.NetworkBridge,
                AttachedAt: time.Now(),
        }
        qemuInst.instance.NetworkAttachments = append(qemuInst.instance.NetworkAttachments, attachment)
        qemuInst.instance.UpdatedAt = time.Now()

        o.logger.Infof("Attached network port %s to instance %s", portName, instanceID)

        return nil
}

// DetachNetwork detaches a VM instance from an OVS network port
func (o *QEMUOrchestrator) DetachNetwork(ctx context.Context, instanceID, portName string) error {
        if instanceID == "" {
                return ErrInvalidInstanceID
        }

        if portName == "" {
                return fmt.Errorf("%w: port name is required", ErrNetworkDetachmentFailed)
        }

        o.mu.RLock()
        qemuInst, exists := o.instances[instanceID]
        o.mu.RUnlock()

        if !exists {
                return ErrInstanceNotFound
        }

        qemuInst.mu.Lock()
        defer qemuInst.mu.Unlock()

        if !qemuInst.networkAttached {
                return fmt.Errorf("%w: no network attached", ErrNetworkDetachmentFailed)
        }

        // Cleanup OVS network
        if qemuInst.ovsNetwork != nil {
                if err := qemuInst.ovsNetwork.Cleanup(ctx); err != nil {
                        return fmt.Errorf("%w: %v", ErrNetworkDetachmentFailed, err)
                }
        }

        // Update instance
        qemuInst.ovsNetwork = nil
        qemuInst.networkAttached = false
        qemuInst.primaryPortName = ""
        qemuInst.primaryBridgeName = ""

        // Remove network attachment from instance metadata
        newAttachments := make([]NetworkAttachment, 0)
        for _, att := range qemuInst.instance.NetworkAttachments {
                if att.PortName != portName {
                        newAttachments = append(newAttachments, att)
                }
        }
        qemuInst.instance.NetworkAttachments = newAttachments
        qemuInst.instance.UpdatedAt = time.Now()

        o.logger.Infof("Detached network port %s from instance %s", portName, instanceID)

        return nil
}

// Wait waits for a VM instance to exit
func (o *QEMUOrchestrator) Wait(ctx context.Context, instanceID string) error {
        if instanceID == "" {
                return ErrInvalidInstanceID
        }

        o.mu.RLock()
        qemuInst, exists := o.instances[instanceID]
        o.mu.RUnlock()

        if !exists {
                return ErrInstanceNotFound
        }

        // Wait for the VM to exit
        qemuInst.waitGroup.Wait()

        return nil
}

// GetState returns the current state of a VM instance
func (o *QEMUOrchestrator) GetState(ctx context.Context, instanceID string) (InstanceState, error) {
        if instanceID == "" {
                return StateUnknown, ErrInvalidInstanceID
        }

        return o.lifecycle.GetState(instanceID)
}

// Close closes the orchestrator and releases resources
func (o *QEMUOrchestrator) Close() error {
        o.mu.Lock()
        defer o.mu.Unlock()

        // Stop all running instances
        for instanceID, qemuInst := range o.instances {
                if qemuInst.vm.IsRunning() {
                        o.logger.Infof("Stopping instance %s during shutdown", instanceID)
                        if err := qemuInst.vm.Stop(); err != nil {
                                o.logger.Errorf("Failed to stop instance %s: %v", instanceID, err)
                        }
                }
        }

        // Close OVS client
        if o.ovsClient != nil {
                if err := o.ovsClient.Close(); err != nil {
                        o.logger.Warnf("Failed to close OVS client: %v", err)
                }
        }

        o.logger.Info("QEMU orchestrator closed")

        return nil
}

// ComputeType represents the type of compute instance
type ComputeType string

const (
        // ComputeTypeVM represents a virtual machine
        ComputeTypeVM ComputeType = "vm"

        // ComputeTypeContainer represents a container
        ComputeTypeContainer ComputeType = "container"
)

// String returns the string representation of ComputeType
func (ct ComputeType) String() string {
        return string(ct)
}

// IsValid checks if the ComputeType is valid
func (ct ComputeType) IsValid() bool {
        return ct == ComputeTypeVM || ct == ComputeTypeContainer
}

// NetworkAttachment represents a network attachment point
type NetworkAttachment struct {
        // PortName is the OVS port name
        PortName string `json:"port_name" yaml:"port_name"`

        // BridgeName is the OVS bridge name
        BridgeName string `json:"bridge_name" yaml:"bridge_name"`

        // VLAN is the VLAN ID (0 for untagged)
        VLAN uint16 `json:"vlan" yaml:"vlan"`

        // MACAddress is the MAC address assigned to this attachment
        MACAddress string `json:"mac_address,omitempty" yaml:"mac_address,omitempty"`

        // IPAddress is the IP address assigned to this attachment
        IPAddress string `json:"ip_address,omitempty" yaml:"ip_address,omitempty"`

        // AttachedAt is the timestamp when the attachment was created
        AttachedAt time.Time `json:"attached_at" yaml:"attached_at"`
}

// Instance represents a compute instance (VM or container)
type Instance struct {
        // ID is the unique identifier for this instance
        ID string `json:"id" yaml:"id"`

        // Name is a human-readable name for the instance
        Name string `json:"name" yaml:"name"`

        // Type is the compute type (VM or container)
        Type ComputeType `json:"type" yaml:"type"`

        // State is the current lifecycle state
        State InstanceState `json:"state" yaml:"state"`

        // Profile is the device profile used to create this instance
        Profile *entities.DeviceProfile `json:"profile,omitempty" yaml:"profile,omitempty"`

        // Config is the instance-specific configuration
        Config *InstanceConfig `json:"config,omitempty" yaml:"config,omitempty"`

        // NetworkAttachments are the network ports attached to this instance
        NetworkAttachments []NetworkAttachment `json:"network_attachments,omitempty" yaml:"network_attachments,omitempty"`

        // CreatedAt is the timestamp when the instance was created
        CreatedAt time.Time `json:"created_at" yaml:"created_at"`

        // UpdatedAt is the timestamp when the instance was last updated
        UpdatedAt time.Time `json:"updated_at" yaml:"updated_at"`

        // Metadata is a map of arbitrary key-value pairs
        Metadata map[string]string `json:"metadata,omitempty" yaml:"metadata,omitempty"`

        // PID is the process ID of the running instance (VM or container)
        PID int `json:"pid,omitempty" yaml:"pid,omitempty"`
}

// InstanceConfig holds instance-specific configuration
type InstanceConfig struct {
        // Memory is the amount of RAM in MB
        Memory int `json:"memory" yaml:"memory"`

        // CPUs is the number of CPU cores
        CPUs int `json:"cpus" yaml:"cpus"`

        // DiskPath is the path to the instance disk
        DiskPath string `json:"disk_path" yaml:"disk_path"`

        // DiskSize is the size of the disk (e.g., "10G", "20G")
        DiskSize string `json:"disk_size,omitempty" yaml:"disk_size,omitempty"`

        // WorkDir is the working directory for instance artifacts
        WorkDir string `json:"work_dir" yaml:"work_dir"`

        // SecureBoot enables UEFI Secure Boot (VMs only)
        SecureBoot bool `json:"secure_boot" yaml:"secure_boot"`

        // EnableKVM enables KVM acceleration (VMs only)
        EnableKVM bool `json:"enable_kvm" yaml:"enable_kvm"`

        // TPMEnabled enables virtual TPM (VMs only)
        TPMEnabled bool `json:"tpm_enabled" yaml:"tpm_enabled"`

        // NetworkMode specifies the network configuration
        NetworkMode string `json:"network_mode,omitempty" yaml:"network_mode,omitempty"`

        // MACAddress is the MAC address for the primary network interface
        MACAddress string `json:"mac_address,omitempty" yaml:"mac_address,omitempty"`

        // BootMode specifies how to boot the instance (install, boot, pxe)
        BootMode string `json:"boot_mode,omitempty" yaml:"boot_mode,omitempty"`

        // ISOPath is the path to the installation ISO (for install mode)
        ISOPath string `json:"iso_path,omitempty" yaml:"iso_path,omitempty"`

        // AdditionalDisks are additional disk images to attach
        AdditionalDisks []string `json:"additional_disks,omitempty" yaml:"additional_disks,omitempty"`

        // Environment variables for the instance
        Environment map[string]string `json:"environment,omitempty" yaml:"environment,omitempty"`
}

// InstanceFilter is used to filter instances in List operations
type InstanceFilter struct {
        // Type filters by compute type
        Type *ComputeType `json:"type,omitempty" yaml:"type,omitempty"`

        // State filters by instance state
        State *InstanceState `json:"state,omitempty" yaml:"state,omitempty"`

        // Name filters by instance name (substring match)
        Name string `json:"name,omitempty" yaml:"name,omitempty"`

        // ProfileID filters by device profile ID
        ProfileID *uint64 `json:"profile_id,omitempty" yaml:"profile_id,omitempty"`

        // CreatedAfter filters instances created after this time
        CreatedAfter *time.Time `json:"created_after,omitempty" yaml:"created_after,omitempty"`

        // CreatedBefore filters instances created before this time
        CreatedBefore *time.Time `json:"created_before,omitempty" yaml:"created_before,omitempty"`
}

// Matches returns true if the instance matches the filter
func (f *InstanceFilter) Matches(instance *Instance) bool {
        if instance == nil {
                return false
        }

        if f.Type != nil && instance.Type != *f.Type {
                return false
        }

        if f.State != nil && instance.State != *f.State {
                return false
        }

        if f.Name != "" && instance.Name != f.Name {
                return false
        }

        if f.ProfileID != nil && (instance.Profile == nil || instance.Profile.ID != *f.ProfileID) {
                return false
        }

        if f.CreatedAfter != nil && instance.CreatedAt.Before(*f.CreatedAfter) {
                return false
        }

        if f.CreatedBefore != nil && instance.CreatedAt.After(*f.CreatedBefore) {
                return false
        }

        return true
}
